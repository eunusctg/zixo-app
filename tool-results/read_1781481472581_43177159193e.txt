   101→   101→ *
   102→   102→ * Features:
   103→   103→ * - Group-specific header: group name, avatar, participant count

   106→   106→ * - Admin badges on participant rows

   109→   109→ * - Participant avatars in the top bar
   110→   110→ * - Group calling: audio/video call buttons start LiveKit Room sessions
   111→   111→ * - Ephemeral timer indicator

   184→   184→                    participantCount = participantCount,
   185→   185→                    participants = participants,
   186→   186→                    onBackClick = onBack,
   187→   187→                    onHeaderClick = { showGroupInfoPanel = true },
   188→   188→                    onAudioCallClick = { viewModel.startGroupAudioCall() },

   191→   191→
   192→   192→                // ── Ephemeral Timer Indicator ─────────────────────────────

   194→   194→                    Row(
   195→   195→                        modifier = Modifier
   196→   196→                            .fillMaxWidth()
   197→   197→                            .background(NeonMintAlpha15)
   198→   198→                            .padding(horizontal = 16.dp, vertical = 4.dp),
   199→   199→                        verticalAlignment = Alignment.CenterVertically,
   200→   200→                        horizontalArrangement = Arrangement.Center
   201→   201→                    ) {
   202→   202→                        Text(
   203→   203→                            text = "⏱ Messages disappear in ${formatEphemeralTimer(ephemeralSeconds)}",
   204→   204→                            color = NeonMint,
   205→   205→                            fontSize = 12.sp,
   206→   206→                            fontWeight = FontWeight.Medium
   207→   207→                        )
   208→   208→                    }
   209→   209→                }
   210→   210→
   211→   211→                // ── Message List ───────────────────────────────────────────
   212→   212→                LazyColumn(
   213→   213→                    state = listState,
   214→   214→                    reverseLayout = true,
   215→   215→                    modifier = Modifier
   216→   216→                        .weight(1f)

   223→   223→                        key = { index -> messages[index].id }
   224→   224→                    ) { index ->
   225→   225→                        val message = messages[index]
   226→   226→                        val isOwnMessage = message.senderUid == currentUserId
   227→   227→
   228→   228→                        GroupMessageBubble(
   229→   229→                            message = message,
   230→   230→                            isOwnMessage = isOwnMessage,
   231→   231→                            participants = participants,
   232→   232→                            onLongClick = { viewModel.showActionMenu.value = message }
   233→   233→                        )
   234→   234→                    }
   235→   235→                }
   236→   236→
   237→   237→                // ── Reply Preview Bar ──────────────────────────────────────
   238→   238→                AnimatedVisibility(
   239→   239→                    visible = replyToMessage != null,
   240→   240→                    enter = slideInVertically() + fadeIn(),
   241→   241→                    exit = slideOutVertically() + fadeOut()
   242→   242→                ) {
   243→   243→                    replyToMessage?.let { replied ->
   244→   244→                        ReplyPreviewBar(
   245→   245→                            senderName = replied.replyToSenderName ?: replied.senderDisplayName,
   246→   246→                            previewText = replied.replyToPreview ?: replied.content,
   247→   247→                            onDismiss = { viewModel.clearReply() }
   248→   248→                        )
   249→   249→                    }
   250→   250→                }
   251→   251→
   252→   252→                // ── Input Tray (74dp Liquid Glass Pill) ────────────────────
   253→   253→                GroupInputTray(
   254→   254→                    value = inputText,
   255→   255→                    onValueChange = { viewModel.inputText.value = it },
   256→   256→                    onSendClick = { viewModel.sendMessage() },
   257→   257→                    isSending = isSending,
   258→   258→                    modifier = Modifier.fillMaxWidth()
   259→   259→                )
   260→   260→            }
   261→   261→
   262→   262→            // ── Group Info Panel (Slide-up Sheet) ─────────────────────────
   263→   263→            if (showGroupInfoPanel) {
   264→   264→                GroupInfoPanel(
   265→   265→                    groupName = groupName,
   266→   266→                    groupAvatar = groupAvatar,
   267→   267→                    groupDescription = thread?.groupDescription ?: "",
   268→   268→                    participants = participants,
   269→   269→                    currentUserId = currentUserId,
   270→   270→                    isAdmin = isAdmin,
   271→   271→                    onClose = { showGroupInfoPanel = false }
   272→   272→                )
   273→   273→            }
   274→   274→
   275→   275→            // ── Action Menu Overlay ────────────────────────────────────────
   276→   276→            showActionMenu?.let { actionMessage ->
   277→   277→                GroupActionMenuOverlay(
   278→   278→                    message = actionMessage,
   279→   279→                    isOwnMessage = actionMessage.senderUid == currentUserId,
   280→   280→                    onReact = { emoji, isThreeD ->
   281→   281→                        viewModel.addReaction(actionMessage.id, emoji, isThreeD)
   282→   282→                    },
   283→   283→                    onReply = {
   284→   284→                        viewModel.replyToMessage.value = actionMessage
   285→   285→                        viewModel.dismissActionMenu()
   286→   286→                    },
   287→   287→                    onForward = {
   288→   288→                        viewModel.forwardMessage(actionMessage.id, emptyList())
   289→   289→                    },
   290→   290→                    onDeleteForMe = {
   291→   291→                        viewModel.deleteForMe(actionMessage.id)
   292→   292→                    },
   293→   293→                    onDeleteForEveryone = {
   294→   294→                        viewModel.deleteForEveryone(actionMessage.id)
   295→   295→                    },
   296→   296→                    onDismiss = { viewModel.dismissActionMenu() }
   297→   297→                )
   298→   298→            }
   299→   299→
   300→   300→            // ── Call State Overlay ─────────────────────────────────────────
   301→   301→            if (callState !is CallState.IDLE) {
   302→   302→                GroupCallOverlay(
   303→   303→                    callState = callState,
   304→   304→                    groupName = groupName,
   305→   305→                    groupAvatar = groupAvatar
   306→   306→                )
   307→   307→            }
   308→   308→        }
   309→   309→    }
   310→   310→}

   412→   412→@OptIn(ExperimentalFoundationApi::class)
   413→   413→@Composable
   414→   414→private fun GroupMessageBubble(
   415→   415→    message: MessageModel,
   416→   416→    isOwnMessage: Boolean,
   417→   417→    participants: List<ThreadParticipant>,
   418→   418→    onLongClick: () -> Unit
   419→   419→) {
   420→   420→    if (message.isDeletedForEveryone) {
   421→   421→        Row(
   422→   422→            modifier = Modifier
   423→   423→                .fillMaxWidth()
   424→   424→                .padding(start = 48.dp, end = 48.dp),
   425→   425→            horizontalArrangement = if (isOwnMessage) Arrangement.End else Arrangement.Start
   426→   426→        ) {
   427→   427→            Box(
   428→   428→                modifier = Modifier
   429→   429→                    .clip(RoundedCornerShape(BubbleCornerSize))
   430→   430→                    .background(IncomingBubbleColor)
   431→   431→                    .padding(horizontal = 14.dp, vertical = 10.dp)
   432→   432→            ) {
   433→   433→                Text(
   434→   434→                    text = "This message was deleted",
   435→   435→                    color = TextSecondary,
   436→   436→                    fontSize = 14.sp,
   437→   437→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   438→   438→                )
   439→   439→            }
   440→   440→        }
   441→   441→        return
   442→   442→    }
   443→   443→
   444→   444→    if (message.isDeletedForMe) return
   445→   445→
   446→   446→    val bubbleColor = if (isOwnMessage) NeonMintAlpha15 else IncomingBubbleColor
   447→   447→    val alignment = if (isOwnMessage) Arrangement.End else Arrangement.Start
   448→   448→    val shape = RoundedCornerShape(
   449→   449→        topStart = BubbleCornerSize,
   450→   450→        topEnd = BubbleCornerSize,
   451→   451→        bottomStart = if (isOwnMessage) BubbleCornerSize else 4.dp,
   452→   452→        bottomEnd = if (isOwnMessage) 4.dp else BubbleCornerSize
   453→   453→    )
   454→   454→
   455→   455→    Row(
   456→   456→        modifier = Modifier
   457→   457→            .fillMaxWidth()
   458→   458→            .padding(start = if (isOwnMessage) 48.dp else 0.dp, end = if (!isOwnMessage) 48.dp else 0.dp),
   459→   459→        horizontalArrangement = alignment
   460→   460→    ) {
   461→   461→        Column(
   462→   462→            modifier = Modifier
   463→   463→                .clip(shape)
   464→   464→                .background(bubbleColor)
   465→   465→                .combinedClickable(
   466→   466→                    interactionSource = remember { MutableInteractionSource() },
   467→   467→                    indication = null,
   468→   468→                    onClick = {},
   469→   469→                    onLongClick = onLongClick
   470→   470→                )
   471→   471→                .padding(horizontal = 14.dp, vertical = 10.dp)
   472→   472→        ) {
   473→   473→            // Sender name label (group-specific: show for incoming messages)
   474→   474→            if (!isOwnMessage) {
   475→   475→                val sender = participants.firstOrNull { it.uid == message.senderUid }

  1157→  1157→/** Formats epoch milliseconds to a readable time string. */
  1158→  1158→private fun formatTimestamp(epochMs: Long): String {
  1159→  1159→    if (epochMs == 0L) return ""
  1160→  1160→    val sdf = java.text.SimpleDateFormat("HH:mm", java.util.Locale.getDefault())
  1161→  1161→    return sdf.format(java.util.Date(epochMs))
  1162→  1162→}

  1164→  1164→/** Formats ephemeral timer seconds into a human-readable string. */
  1165→  1165→private fun formatEphemeralTimer(seconds: Int): String {
  1166→  1166→    return when {
  1167→  1167→        seconds >= 86400 -> "${seconds / 86400} day${if (seconds / 86400 > 1) "s" else ""}"
  1168→  1168→        seconds >= 3600 -> "${seconds / 3600} hour${if (seconds / 3600 > 1) "s" else ""}"
  1169→  1169→        seconds >= 60 -> "${seconds / 60} minute${if (seconds / 60 > 1) "s" else ""}"
  1170→  1170→        else -> "$seconds second${if (seconds > 1) "s" else ""}"
  1171→  1171→    }
  1172→  1172→}
