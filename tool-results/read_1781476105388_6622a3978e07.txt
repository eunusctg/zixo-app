   439→// ──────────────────────────────────────────────────────────────────────────────
   440→// QR Code composable using ZXing
   441→// ──────────────────────────────────────────────────────────────────────────────
   442→
   443→@Composable
   444→private fun ZixoQrCode(
   445→    data: String,
   446→    size: androidx.compose.ui.unit.Dp,
   447→    modifier: Modifier = Modifier
   448→) {
   449→    if (data.isBlank()) return
   450→
   451→    val qrBitmap = remember(data) {
   452→        generateQrBitmap(data, size.value.toInt())
   453→    }
   454→
   455→    Card(
   456→        modifier = modifier.padding(2.dp),
   457→        shape = RoundedCornerShape(8.dp),
   458→        colors = CardDefaults.cardColors(containerColor = Color.White),
   459→        elevation = CardDefaults.cardElevation(defaultElevation = 0.dp),
   460→    ) {
   461→        androidx.compose.foundation.Image(
   462→            bitmap = qrBitmap.asImageBitmap(),
   463→            contentDescription = "QR Code for Zixo number",
   464→            modifier = Modifier
   465→                .size(size)
   466→                .padding(8.dp),
   467→            contentScale = ContentScale.Fit
   468→        )
   469→    }
   470→}
   471→
   472→/**
   473→ * Generates a QR code bitmap using ZXing core library.
   474→ */
   475→private fun generateQrBitmap(data: String, sizePx: Int): Bitmap {
   476→    val writer = QRCodeWriter()
   477→    val bitMatrix = writer.encode(
   478→        data,
   479→        BarcodeFormat.QR_CODE,
   480→        sizePx,
   481→        sizePx
   482→    )
   483→    val width = bitMatrix.width
   484→    val height = bitMatrix.height
   485→    val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.RGB_565)
   486→    for (x in 0 until width) {
   487→        for (y in 0 until height) {
   488→            bitmap.setPixel(
   489→                x, y,
   490→                if (bitMatrix[x, y]) AndroidColor.BLACK else AndroidColor.WHITE
   491→            )
   492→        }
   493→    }
   494→    return bitmap
   495→}

   767→/**
   768→ * Formats an 8-digit number as two 4-digit blocks separated by a space.
   769→ * e.g., "12345678" -> "1234 5678"
   770→ */
   771→private fun formatZixoNumber(number: String): String {
   772→    val digits = number.filter { it.isDigit() }
   773→    return if (digits.length >= 8) {
   774→        "${digits.substring(0, 4)} ${digits.substring(4, 8)}"
   775→    } else {
   776→        digits
   777→    }
   778→}
