    97→/**
    98→ * Full chat message screen with keyboard avoidance and Liquid Glass design.
    99→ *
   100→ * Features:
   101→ * - Wrapped in [Scaffold] with `Modifier.imePadding()` for keyboard avoidance
   102→ * - [LazyColumn] with `reverseLayout = true` for messages — newest at bottom
   103→ * - Auto-scroll on new messages via [LaunchedEffect]
   104→ * - 74dp liquid glass pill-shaped input tray
   105→ * - Outgoing messages: glowing neon mint transparency background
   106→ * - Incoming messages: translucent dark petrol grey curves
   107→ * - Long-press action menu with 3D reactions, reply, forward, delete, copy
   108→ * - Reply preview bar above input tray
   109→ * - Call state overlay when active
   110→ * - Top bar with contact avatar, name, online status, and call icons
   111→ * - Communication gate check — blocked overlay if not mutual contact
   112→ * - NavController for navigation
   113→ *
   114→ * @param threadId The ID of the chat thread to display.
   115→ * @param navController Navigation controller for screen transitions.
   116→ * @param onBack Callback for the back navigation button.
   117→ */
   118→@OptIn(ExperimentalFoundationApi::class)
   119→@Composable
   120→fun ChatMessageScreen(
   121→    threadId: String,
   122→    navController: NavController,
   123→    onBack: () -> Unit = {},
   124→    viewModel: ChatViewModel = hiltViewModel()
   125→) {
   126→    val messages by viewModel.messages.collectAsState()
   127→    val thread by viewModel.thread.collectAsState()
   128→    val inputText by viewModel.inputText.collectAsState()
   129→    val isSending by viewModel.isSending.collectAsState()
   130→    val callState by viewModel.callState.collectAsState()
   131→    val replyToMessage by viewModel.replyToMessage.collectAsState()
   132→    val showActionMenu by viewModel.showActionMenu.collectAsState()
   133→    val currentUserId by viewModel.currentUserId.collectAsState()
   134→    val errorMessage by viewModel.errorMessage.collectAsState()
   135→    val communicationGate by viewModel.communicationGate.collectAsState()
   136→
   137→    val snackbarHostState = remember { SnackbarHostState() }
   138→    val listState = rememberLazyListState()
   139→
   140→    // Auto-scroll to bottom on new messages
   141→    LaunchedEffect(messages.size) {
   142→        if (messages.isNotEmpty()) {
   143→            listState.animateScrollToItem(0)
   144→        }
   145→    }
   146→
   147→    // Display error messages as snackbars
   148→    LaunchedEffect(errorMessage) {
   149→        errorMessage?.let {
   150→            snackbarHostState.showSnackbar(it)
   151→            viewModel.clearError()
   152→        }
   153→    }
   154→
   155→    // Determine contact info from thread
   156→    val otherParticipant = thread?.participantProfiles?.values?.firstOrNull {
   157→        it.uid != currentUserId
   158→    }
   159→    val contactName = otherParticipant?.displayName ?: "Chat"
   160→    val contactAvatar = otherParticipant?.avatarUrl ?: ""
   161→    val isOnline = otherParticipant?.isOnline ?: false
   162→
   163→    // Check if communication is blocked
   164→    val isBlocked = communicationGate is CommunicationGate.Blocked
   165→
   166→    Scaffold(
   167→        snackbarHost = { SnackbarHost(snackbarHostState) },
   168→        containerColor = Color.Transparent,
   169→        modifier = Modifier.fillMaxSize()
   170→    ) { innerPadding ->
   171→        Box(modifier = Modifier.fillMaxSize()) {
   172→            // Animated glass background
   173→            ZixoGlassBackground()
   174→
   175→            Column(
   176→                modifier = Modifier
   177→                    .fillMaxSize()
   178→                    .imePadding()
   179→                    .padding(innerPadding)
   180→            ) {
   181→                // ── Top Bar ────────────────────────────────────────────────
   182→                ChatTopBar(
   183→                    contactName = contactName,
   184→                    contactAvatar = contactAvatar,
   185→                    isOnline = isOnline,
   186→                    onBackClick = onBack,
   187→                    onAudioCallClick = { viewModel.startAudioCall() },
   188→                    onVideoCallClick = { viewModel.startVideoCall() }
   189→                )
   190→
   191→                // ── Message List ───────────────────────────────────────────
   192→                LazyColumn(
   193→                    state = listState,
   194→                    reverseLayout = true,
   195→                    modifier = Modifier
   196→                        .weight(1f)
   197→                        .fillMaxWidth()
   198→                        .padding(horizontal = 12.dp),
   199→                    verticalArrangement = Arrangement.spacedBy(6.dp)
   200→                ) {
   201→                    items(
   202→                        count = messages.size,
   203→                        key = { index -> messages[index].id }
   204→                    ) { index ->
   205→                        val message = messages[index]
   206→                        val isOwnMessage = message.senderUid == currentUserId
   207→
   208→                        MessageBubble(
   209→                            message = message,
   210→                            isOwnMessage = isOwnMessage,
   211→                            onLongClick = { viewModel.showActionMenu.value = message }
   212→                        )
   213→                    }
   214→                }
   215→
   216→                // ── Reply Preview Bar ──────────────────────────────────────
   217→                AnimatedVisibility(
   218→                    visible = replyToMessage != null,
   219→                    enter = slideInVertically() + fadeIn(),
   220→                    exit = slideOutVertically() + fadeOut()
   221→                ) {
   222→                    replyToMessage?.let { replied ->
   223→                        ReplyPreviewBar(
   224→                            senderName = replied.replyToSenderName ?: replied.senderDisplayName,
   225→                            previewText = replied.replyToPreview ?: replied.content,
   226→                            onDismiss = { viewModel.clearReplyTo() }
   227→                        )
   228→                    }
   229→                }
   230→
   231→                // ── Input Tray (74dp Liquid Glass Pill) ────────────────────
   232→                InputTray(
   233→                    value = inputText,
   234→                    onValueChange = { viewModel.onInputTextChanged(it) },
   235→                    onSendClick = { viewModel.sendMessage() },
   236→                    isSending = isSending,
   237→                    enabled = !isBlocked,
   238→                    modifier = Modifier.fillMaxWidth()
   239→                )
   240→            }
   241→
   242→            // ── Communication Gate Blocked Overlay ────────────────────────
   243→            if (isBlocked) {
   244→                CommunicationGateOverlay(
   245→                    reason = (communicationGate as? CommunicationGate.Blocked)?.reason
   246→                        ?: "You can only message mutual contacts"
   247→                )
   248→            }
   249→
   250→            // ── Action Menu Overlay ────────────────────────────────────────
   251→            showActionMenu?.let { actionMessage ->
   252→                ActionMenuOverlay(
   253→                    message = actionMessage,
   254→                    isOwnMessage = actionMessage.senderUid == currentUserId,
   255→                    onReact = { emoji, isThreeD ->
   256→                        viewModel.addReaction(actionMessage.id, emoji, isThreeD)
   257→                    },
   258→                    onReply = {
   259→                        viewModel.setReplyTo(actionMessage.id)
   260→                    },
   261→                    onForward = {
   262→                        viewModel.forwardMessage(actionMessage.id, emptyList())
   263→                    },
   264→                    onDeleteForMe = {
   265→                        viewModel.deleteForMe(actionMessage.id)
   266→                    },
   267→                    onDeleteForEveryone = {
   268→                        viewModel.deleteForEveryone(actionMessage.id)
   269→                    },
   270→                    onCopy = {
   271→                        // Handled inside the overlay
   272→                    },
   273→                    onDismiss = { viewModel.dismissActionMenu() }
   274→                )
   275→            }
   276→
   277→            // ── Call State Overlay ─────────────────────────────────────────
   278→            if (callState !is CallState.IDLE) {
   279→                CallStateOverlay(
   280→                    callState = callState,
   281→                    contactName = contactName,
   282→                    contactAvatar = contactAvatar
   283→                )
   284→            }
   285→        }
   286→    }
   287→}

   367→@OptIn(ExperimentalFoundationApi::class)
   368→@Composable
   369→private fun MessageBubble(
   370→    message: MessageModel,
   371→    isOwnMessage: Boolean,
   372→    onLongClick: () -> Unit
   373→) {
   374→    // Deleted placeholder
   375→    if (message.isDeletedForEveryone) {
   376→        Row(
   377→            modifier = Modifier
   378→                .fillMaxWidth()
   379→                .padding(
   380→                    start = if (isOwnMessage) 48.dp else 0.dp,
   381→                    end = if (!isOwnMessage) 48.dp else 0.dp
   382→                ),
   383→            horizontalArrangement = if (isOwnMessage) Arrangement.End else Arrangement.Start
   384→        ) {
   385→            Box(
   386→                modifier = Modifier
   387→                    .clip(RoundedCornerShape(BubbleCornerSize))
   388→                    .background(IncomingBubbleColor)
   389→                    .padding(horizontal = 14.dp, vertical = 10.dp)
   390→            ) {
   391→                Text(
   392→                    text = "This message was deleted",
   393→                    color = TextSecondary,
   394→                    fontSize = 14.sp,
   395→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   396→                )
   397→            }
   398→        }
   399→        return
   400→    }
   401→
   402→    // Soft-deleted for me — invisible
   403→    if (message.isDeletedForMe) return
   404→
   405→    val bubbleColor = if (isOwnMessage) NeonMintAlpha15 else IncomingBubbleColor
   406→    val alignment = if (isOwnMessage) Arrangement.End else Arrangement.Start
   407→    val shape = RoundedCornerShape(
   408→        topStart = BubbleCornerSize,
   409→        topEnd = BubbleCornerSize,
   410→        bottomStart = if (isOwnMessage) BubbleCornerSize else 4.dp,
   411→        bottomEnd = if (isOwnMessage) 4.dp else BubbleCornerSize
   412→    )
   413→
   414→    Row(
   415→        modifier = Modifier
   416→            .fillMaxWidth()
   417→            .padding(
   418→                start = if (isOwnMessage) 48.dp else 0.dp,
   419→                end = if (!isOwnMessage) 48.dp else 0.dp
   420→            ),
   421→        horizontalArrangement = alignment
   422→    ) {
   423→        Column(
   424→            modifier = Modifier
   425→                .clip(shape)
   426→                .background(bubbleColor)
   427→                .combinedClickable(
   428→                    interactionSource = remember { MutableInteractionSource() },
   429→                    indication = null,
   430→                    onClick = {},
   431→                    onLongClick = onLongClick
   432→                )
   433→                .padding(horizontal = 14.dp, vertical = 10.dp)
   434→        ) {
   435→            // Reply reference preview
   436→            if (message.replyToPreview != null) {
   437→                Row(
   438→                    modifier = Modifier
   439→                        .fillMaxWidth()
   440→                        .clip(RoundedCornerShape(8.dp))
   441→                        .background(NeonMintAlpha15)
   442→                        .padding(horizontal = 8.dp, vertical = 4.dp)
   443→                ) {
   444→                    Text(
   445→                        text = message.replyToSenderName ?: "",
   446→                        color = NeonMint,
   447→                        fontSize = 12.sp,
   448→                        fontWeight = FontWeight.SemiBold,
   449→                        maxLines = 1
   450→                    )
   451→                    Spacer(modifier = Modifier.width(6.dp))
   452→                    Text(
   453→                        text = message.replyToPreview ?: "",
   454→                        color = TextSecondary,
   455→                        fontSize = 12.sp,
   456→                        maxLines = 1,
   457→                        overflow = TextOverflow.Ellipsis,
   458→                        modifier = Modifier.weight(1f)
   459→                    )
   460→                }
   461→                Spacer(modifier = Modifier.height(4.dp))
   462→            }
   463→
   464→            // Forwarded label
   465→            if (message.isForwarded) {
   466→                Text(
   467→                    text = "Forwarded",
   468→                    color = TextSecondary,
   469→                    fontSize = 11.sp,
   470→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   471→                )
   472→                Spacer(modifier = Modifier.height(2.dp))
   473→            }
   474→
   475→            // Media thumbnail
   476→            if (message.type == MessageContentType.IMAGE || message.type == MessageContentType.VIDEO) {
   477→                val imageUrl = message.mediaThumbnailUrl ?: message.mediaUrl
   478→                if (!imageUrl.isNullOrBlank()) {
   479→                    AsyncImage(
   480→                        model = imageUrl,
   481→                        contentDescription = "Media",
   482→                        modifier = Modifier
   483→                            .fillMaxWidth()
   484→                            .height(180.dp)
   485→                            .clip(RoundedCornerShape(10.dp)),
   486→                        contentScale = ContentScale.Crop
   487→                    )
   488→                    Spacer(modifier = Modifier.height(4.dp))
   489→                }
   490→                // Play icon overlay for video
   491→                if (message.type == MessageContentType.VIDEO) {
   492→                    Box(
   493→                        modifier = Modifier
   494→                            .size(36.dp)
   495→                            .clip(CircleShape)
   496→                            .background(DarkPetrolCharcoal.copy(alpha = 0.7f)),
   497→                        contentAlignment = Alignment.Center
   498→                    ) {
   499→                        Icon(
   500→                            imageVector = Icons.Filled.PlayArrow,
   501→                            contentDescription = "Play",
   502→                            tint = TextPrimary,
   503→                            modifier = Modifier.size(20.dp)
   504→                        )
   505→                    }
   506→                }
   507→            }
   508→
   509→            // Voice message indicator
   510→            if (message.type == MessageContentType.AUDIO_VOICE) {
   511→                Row(verticalAlignment = Alignment.CenterVertically) {
   512→                    Icon(
   513→                        imageVector = Icons.Filled.PlayArrow,
   514→                        contentDescription = "Voice",
   515→                        tint = NeonMint,
   516→                        modifier = Modifier.size(20.dp)
   517→                    )
   518→                    Spacer(modifier = Modifier.width(6.dp))
   519→                    Text(
   520→                        text = "Voice message",
   521→                        color = TextSecondary,
   522→                        fontSize = 14.sp
   523→                    )
   524→                }
   525→                Spacer(modifier = Modifier.height(4.dp))
   526→            }
   527→
   528→            // Text content
   529→            if (message.content.isNotBlank()) {
   530→                Text(
   531→                    text = message.content,
   532→                    color = TextPrimary,
   533→                    fontSize = 15.sp,
   534→                    lineHeight = 20.sp
   535→                )
   536→            }
   537→
   538→            // Media caption
   539→            if (!message.caption.isNullOrBlank() && message.type != MessageContentType.TEXT) {
   540→                Spacer(modifier = Modifier.height(4.dp))
   541→                Text(
   542→                    text = message.caption,
   543→                    color = TextPrimary,
   544→                    fontSize = 14.sp
   545→                )
   546→            }
   547→
   548→            // Timestamp and read receipt row
   549→            Row(
   550→                modifier = Modifier.fillMaxWidth(),
   551→                horizontalArrangement = Arrangement.End,
   552→                verticalAlignment = Alignment.CenterVertically
   553→            ) {
   554→                Text(
   555→                    text = formatTimestamp(message.timestamp),
   556→                    color = TextSecondary,
   557→                    fontSize = 11.sp
   558→                )
   559→                if (isOwnMessage) {
   560→                    Spacer(modifier = Modifier.width(4.dp))
   561→                    val tickColor = if (message.isRead) NeonMint else TextSecondary
   562→                    Text(text = if (message.isRead) "✓✓" else "✓", color = tickColor, fontSize = 11.sp)
   563→                }
   564→            }
   565→
   566→            // Reactions row — emoji badges below the bubble
   567→            if (message.reactions.isNotEmpty()) {
   568→                Spacer(modifier = Modifier.height(4.dp))
   569→                Row(horizontalArrangement = Arrangement.spacedBy(2.dp)) {
   570→                    message.reactions.distinctBy { it.emoji }.forEach { reaction ->
   571→                        Box(
   572→                            modifier = Modifier
   573→                                .clip(RoundedCornerShape(12.dp))
   574→                                .background(GlassBorderColor)
   575→                                .padding(horizontal = 6.dp, vertical = 2.dp)
   576→                        ) {
   577→                            Text(
   578→                                text = "${reaction.emoji} ${message.reactions.count { it.emoji == reaction.emoji }}",
   579→                                fontSize = 12.sp
   580→                            )
   581→                        }
   582→                    }
   583→                }
   584→            }
   585→        }
   586→    }
   587→}

   589→// ── Reply Preview Bar ─────────────────────────────────────────────────────────
   590→
   591→@Composable
   592→private fun ReplyPreviewBar(
   593→    senderName: String,
   594→    previewText: String,
   595→    onDismiss: () -> Unit
   596→) {

   952→/** Formats epoch milliseconds to a readable time string. */
   953→private fun formatTimestamp(epochMs: Long): String {
   954→    if (epochMs == 0L) return ""
   955→    val sdf = java.text.SimpleDateFormat("HH:mm", java.util.Locale.getDefault())
   956→    return sdf.format(java.util.Date(epochMs))
   957→}
