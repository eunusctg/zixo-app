   531→// ════════════════════════════════════════════════════════════════
   532→// QR Code Image Composable (brand emerald green on frosted glass)
   533→// ════════════════════════════════════════════════════════════════
   534→
   535→@Composable
   536→private fun QrCodeImage(
   537→    data: String,
   538→    size: androidx.compose.ui.unit.Dp,
   539→    qrColor: Color = NeonMint,
   540→    backgroundColor: Color = Color.White,
   541→    modifier: Modifier = Modifier
   542→) {
   543→    if (data.isBlank()) return
   544→
   545→    val density = LocalDensity.current
   546→    val sizePx = with(density) { size.roundToPx() }
   547→
   548→    val qrBitmap = remember(data, sizePx) {
   549→        generateQrBitmap(data, sizePx, qrColor, backgroundColor)
   550→    }
   551→
   552→    Box(
   553→        modifier = modifier
   554→            .size(size)
   555→            .clip(RoundedCornerShape(16.dp))
   556→            .background(Color(0x0D1A2A32)) // Very subtle dark glass behind QR
   557→            .border(
   558→                width = 1.dp,
   559→                color = Color(0x1AFFFFFF),
   560→                shape = RoundedCornerShape(16.dp)
   561→            )
   562→            .padding(12.dp),
   563→        contentAlignment = Alignment.Center
   564→    ) {
   565→        androidx.compose.foundation.Image(
   566→            bitmap = qrBitmap.asImageBitmap(),
   567→            contentDescription = "QR Code",
   568→            modifier = Modifier
   569→                .size(size - 24.dp)
   570→                .clip(RoundedCornerShape(8.dp)),
   571→            contentScale = ContentScale.Fit
   572→        )
   573→    }
   574→}
   575→
   576→/**
   577→ * Generates a QR code bitmap with custom foreground/background colors.
   578→ */
   579→private fun generateQrBitmap(
   580→    data: String,
   581→    sizePx: Int,
   582→    fgColor: Color,
   583→    bgColor: Color
   584→): Bitmap {
   585→    val writer = QRCodeWriter()
   586→    val bitMatrix = writer.encode(data, BarcodeFormat.QR_CODE, sizePx, sizePx)
   587→    val width = bitMatrix.width
   588→    val height = bitMatrix.height
   589→    val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
   590→    val fgArgb = fgColor.copy(alpha = 1f).let {
   591→        AndroidColor.argb(255, (it.red * 255).toInt(), (it.green * 255).toInt(), (it.blue * 255).toInt())
   592→    }
   593→    val bgArgb = bgColor.let {
   594→        AndroidColor.argb(
   595→            (it.alpha * 255).toInt(),
   596→            (it.red * 255).toInt(),
   597→            (it.green * 255).toInt(),
   598→            (it.blue * 255).toInt()
   599→        )
   600→    }
   601→    for (x in 0 until width) {
   602→        for (y in 0 until height) {
   603→            bitmap.setPixel(x, y, if (bitMatrix[x, y]) fgArgb else bgArgb)
   604→        }
   605→    }
   606→    return bitmap
   607→}
