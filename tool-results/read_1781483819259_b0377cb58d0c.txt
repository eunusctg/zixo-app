    17→/**
    18→ * Firebase Realtime DB Signaling Client for WebRTC — Persistent Mesh Architecture.
    19→ *
    20→ * Handles all signaling operations for WebRTC calls via Firebase Realtime Database:
    21→ * - Send SDP Offer/Answer to /calls/{callId}/signaling/
    22→ * - Send/receive ICE candidates via /calls/{callId}/candidates/
    23→ * - Persistent continuous listeners for incoming signaling data
    24→ * - Call state tracking (IDLE, DIALING, RINGING, CONNECTED, ENDED)
    25→ * - Online presence tracking via /presence/{uid}/
    26→ * - Mesh signaling support for group calls
    27→ *
    28→ * ## Persistent Listener Architecture:
    29→ * All listeners use [ValueEventListener] with continuous observation (NOT one-time reads).
    30→ * The Firebase Realtime Database maintains persistent WebSocket connections that
    31→ * instantly propagate signaling data across all connected devices.
    32→ *
    33→ * ## Mesh Signaling for Group Calls:
    34→ * In a group call, each participant establishes a direct PeerConnection with every
    35→ * other participant. The signaling path is organized as:
    36→ * ```
    37→ * /calls/{callId}/
    38→ *   ├── status/              → "dialing" | "ringing" | "connected" | "ended"
    39→ *   ├── participants/        → { uid: { joinedAt, isAudioEnabled, isVideoEnabled } }
    40→ *   ├── signaling/
    41→ *   │   ├── {senderUid}/
    42→ *   │   │   ├── offer/       → { sdp, targetUid, timestamp }
    43→ *   │   │   └── answer/      → { sdp, targetUid, timestamp }
    44→ *   ├── candidates/
    45→ *   │   ├── {senderUid}/
    46→ *   │   │   ├── {pushId}/    → { sdpMid, sdpMLineIndex, sdp, targetUid, timestamp }
    47→ * ```
    48→ * This structure allows N×(N-1) mesh connections where each participant sends
    49→ * individual offers to every other participant.
    50→ *
    51→ * All operations run on Dispatchers.IO and never block the Main Thread.
    52→ */
    53→@Singleton
    54→class FirebaseSignalingClient @Inject constructor(
    55→    private val realtimeDb: FirebaseDatabase
    56→) {

    62→    /**
    63→     * Creates a new call entry in Firebase Realtime DB.
    64→     * Sets the initial status to "dialing" and registers the caller as a participant.
    65→     *
    66→     * @param callId The unique call ID.
    67→     * @param callerUid The UID of the call initiator.
    68→     * @param targetUid The UID of the call recipient.
    69→     * @param isVideoCall Whether this is a video call.
    70→     */
    71→    suspend fun createCall(
    72→        callId: String,
    73→        callerUid: String,
    74→        targetUid: String,
    75→        isVideoCall: Boolean
    76→    ) = kotlinx.coroutines.withContext(Dispatchers.IO) {
    77→        try {
    78→            val callData = mapOf(
    79→                "status" to "dialing",
    80→                "callerUid" to callerUid,
    81→                "targetUid" to targetUid,
    82→                "isVideoCall" to isVideoCall,
    83→                "createdAt" to System.currentTimeMillis(),
    84→                "participants" to mapOf(
    85→                    callerUid to mapOf(
    86→                        "joinedAt" to System.currentTimeMillis(),
    87→                        "isAudioEnabled" to true,
    88→                        "isVideoEnabled" to isVideoCall
    89→                    )
    90→                )
    91→            )
    92→
    93→            realtimeDb.getReference("calls")
    94→                .child(callId)
    95→                .setValue(callData)
    96→                .await()
    97→
    98→            Timber.d("Call created: %s (video=%s)", callId, isVideoCall)
    99→        } catch (e: Exception) {
   100→            Timber.e(e, "Failed to create call: %s", callId)
   101→        }
   102→    }

   158→    /**
   159→     * Sends an SDP offer to the Firebase Realtime DB signaling path.
   160→     *
   161→     * For 1-to-1 calls: writes to /calls/{callId}/signaling/offer
   162→     * For mesh group calls: writes to /calls/{callId}/signaling/{senderUid}/offer
   163→     *
   164→     * @param callId The call ID.
   165→     * @param senderUid The UID of the caller sending the offer.
   166→     * @param targetUid The UID of the intended recipient (for mesh routing).
   167→     * @param sdpOffer The SDP offer string.
   168→     */
   169→    suspend fun sendOffer(
   170→        callId: String,
   171→        senderUid: String,
   172→        targetUid: String = "",
   173→        sdpOffer: String?
   174→    ) = kotlinx.coroutines.withContext(Dispatchers.IO) {
   175→        try {
   176→            if (sdpOffer == null) return@withContext
   177→
   178→            val offerData = mapOf(
   179→                "type" to "offer",
   180→                "sdp" to sdpOffer,
   181→                "senderUid" to senderUid,
   182→                "targetUid" to targetUid,
   183→                "timestamp" to System.currentTimeMillis()
   184→            )
   185→
   186→            // For 1-to-1 calls, write to the simple offer path
   187→            // For mesh calls, write to the sender-specific path
   188→            if (targetUid.isBlank()) {
   189→                realtimeDb.getReference("calls")
   190→                    .child(callId)
   191→                    .child("signaling")
   192→                    .child("offer")
   193→                    .setValue(offerData)
   194→                    .await()
   195→            } else {
   196→                realtimeDb.getReference("calls")
   197→                    .child(callId)
   198→                    .child("signaling")
   199→                    .child(senderUid)
   200→                    .child("offer")
   201→                    .setValue(offerData)
   202→                    .await()
   203→            }
   204→
   205→            Timber.d("SDP offer sent for call: %s", callId)
   206→        } catch (e: Exception) {
   207→            Timber.e(e, "Failed to send SDP offer for call: %s", callId)
   208→        }
   209→    }

   263→    /**

   267→     * @param senderUid The UID of the sender.

   273→    suspend fun sendIceCandidate(

   275→        senderUid: String,

   278→        sdpMLineIndex: Int,

   282→            val candidateData = mapOf(
   283→                "sdpMid" to sdpMid,

   298→                realtimeDb.getReference("calls")
   299→                    .child(callId)
   300→                    .child("candidates")
   301→                    .child(senderUid)

   306→

   317→    /**
   318→     * Observes the SDP offer for a call via persistent Firebase listener.
   319→     *
   320→     * @param callId The call ID.
   321→     * @param localUid The local user's UID (for mesh filtering).
   322→     * @return A flow emitting SDP offer data.
   323→     */
   324→    fun observeOffer(callId: String, localUid: String = ""): Flow<SdpData?> = callbackFlow {
   325→        val ref = realtimeDb.getReference("calls")
   326→            .child(callId)
   327→            .child("signaling")
   328→
   329→        val listener = ref.addValueEventListener(object : ValueEventListener {
   330→            override fun onDataChange(snapshot: DataSnapshot) {
   331→                try {
   332→                    // Check simple offer path first (1-to-1)
   333→                    val simpleOffer = snapshot.child("offer")
   334→                    if (simpleOffer.exists()) {
   335→                        val sdp = simpleOffer.child("sdp").getValue(String::class.java)
   336→                        val senderUid = simpleOffer.child("senderUid").getValue(String::class.java) ?: ""
   337→                        if (sdp != null && senderUid != localUid) {
   338→                            trySend(SdpData(sdp = sdp, senderUid = senderUid, type = "offer"))
   339→                            return
   340→                        }
   341→                    }
   342→
   343→                    // Check mesh paths (signaling/{uid}/offer)
   344→                    for (child in snapshot.children) {
   345→                        val offerChild = child.child("offer")
   346→                        if (offerChild.exists()) {
   347→                            val sdp = offerChild.child("sdp").getValue(String::class.java)
   348→                            val senderUid = offerChild.child("senderUid").getValue(String::class.java) ?: ""
   349→                            val targetUid = offerChild.child("targetUid").getValue(String::class.java) ?: ""
   350→                            if (sdp != null && senderUid != localUid &&
   351→                                (targetUid.isBlank() || targetUid == localUid)
   352→                            ) {
   353→                                trySend(SdpData(sdp = sdp, senderUid = senderUid, type = "offer"))
   354→                                return
   355→                            }
   356→                        }
   357→                    }
   358→
   359→                    trySend(null)
   360→                } catch (e: Exception) {
   361→                    Timber.e(e, "Error parsing offer for call: %s", callId)
   362→                }
   363→            }
   364→
   365→            override fun onCancelled(error: DatabaseError) {
   366→                Timber.e(error.toException(), "Offer listener cancelled for call: %s", callId)
   367→            }
   368→        })
   369→
   370→        awaitClose { ref.removeEventListener(listener) }
   371→    }.flowOn(Dispatchers.IO)
   372→

   429→    /**
   430→     * Observes incoming ICE candidates for a call via persistent Firebase listener.
   431→     *
   432→     * @param callId The call ID.
   433→     * @param localUid The local user's UID (to filter out self-sent candidates).
   434→     * @return A flow emitting lists of ICE candidate data.
   435→     */
   436→    fun observeIceCandidates(callId: String, localUid: String): Flow<List<IceCandidateData>> =
   437→        callbackFlow {
   438→            val ref = realtimeDb.getReference("calls")
   439→                .child(callId)
   440→                .child("candidates")
   441→
   442→            val listener = ref.addValueEventListener(object : ValueEventListener {
   443→                override fun onDataChange(snapshot: DataSnapshot) {
   444→                    try {
   445→                        val candidates = mutableListOf<IceCandidateData>()
   446→                        for (child in snapshot.children) {
   447→                            // Check if this is a sender-specific path (mesh)
   448→                            val hasSubChildren = child.hasChildren() && child.child("sdpMid").exists() == false
   449→                            if (hasSubChildren) {
   450→                                // Mesh path: candidates/{senderUid}/{pushId}
   451→                                for (subChild in child.children) {
   452→                                    val senderUid = subChild.child("senderUid").getValue(String::class.java)
   453→                                    val targetUid = subChild.child("targetUid").getValue(String::class.java) ?: ""
   454→                                    if (senderUid == localUid) continue
   455→                                    if (targetUid.isNotBlank() && targetUid != localUid) continue
   456→
   457→                                    val sdpMid = subChild.child("sdpMid").getValue(String::class.java) ?: ""
   458→                                    val sdpMLineIndex = subChild.child("sdpMLineIndex").getValue(Int::class.java) ?: 0
   459→                                    val sdp = subChild.child("sdp").getValue(String::class.java) ?: ""
   460→                                    candidates.add(IceCandidateData(sdpMid, sdpMLineIndex, sdp))
   461→                                }
   462→                            } else {
   463→                                // Simple path: candidates/{pushId}
   464→                                val senderUid = child.child("senderUid").getValue(String::class.java)
   465→                                val targetUid = child.child("targetUid").getValue(String::class.java) ?: ""
   466→                                if (senderUid == localUid) continue
   467→                                if (targetUid.isNotBlank() && targetUid != localUid) continue
   468→
   469→                                val sdpMid = child.child("sdpMid").getValue(String::class.java) ?: ""
   470→                                val sdpMLineIndex = child.child("sdpMLineIndex").getValue(Int::class.java) ?: 0
   471→                                val sdp = child.child("sdp").getValue(String::class.java) ?: ""
   472→                                candidates.add(IceCandidateData(sdpMid, sdpMLineIndex, sdp))
   473→                            }
   474→                        }
   475→                        trySend(candidates)
   476→                    } catch (e: Exception) {
   477→                        Timber.e(e, "Error parsing ICE candidates for call: %s", callId)
   478→                    }
   479→                }
   480→
   481→                override fun onCancelled(error: DatabaseError) {
   482→                    Timber.e(error.toException(), "ICE candidate listener cancelled for call: %s", callId)
   483→                }
   484→            })
   485→
   486→            awaitClose { ref.removeEventListener(listener) }
   487→        }.flowOn(Dispatchers.IO)
   488→

   514→    // ════════════════════════════════════════════════════════════════
   515→    // Incoming Call Detection
   516→    // ════════════════════════════════════════════════════════════════
   517→
   518→    /**
   519→     * Observes incoming calls for the local user via persistent Firebase listener.
   520→     *
   521→     * Listens to /calls/ where targetUid matches the local user's UID and
   522→     * the status is "dialing" or "ringing".
   523→     *
   524→     * @param localUid The local user's UID.
   525→     * @return A flow emitting lists of incoming call data.
   526→     */
   527→    fun observeIncomingCalls(localUid: String): Flow<List<IncomingCallData>> = callbackFlow {
   528→        val ref = realtimeDb.getReference("calls")
   529→            .orderByChild("targetUid")
   530→            .equalTo(localUid)
   531→
   532→        val listener = ref.addValueEventListener(object : ValueEventListener {
   533→            override fun onDataChange(snapshot: DataSnapshot) {
   534→                try {
   535→                    val calls = mutableListOf<IncomingCallData>()
   536→                    for (child in snapshot.children) {
   537→                        val status = child.child("status").getValue(String::class.java) ?: ""
   538→                        if (status == "dialing" || status == "ringing") {
   539→                            val callId = child.key ?: continue
   540→                            val callerUid = child.child("callerUid").getValue(String::class.java) ?: ""
   541→                            val isVideoCall = child.child("isVideoCall").getValue(Boolean::class.java) ?: false
   542→                            val createdAt = child.child("createdAt").getValue(Long::class.java) ?: 0L
   543→
   544→                            calls.add(IncomingCallData(
   545→                                callId = callId,
   546→                                callerUid = callerUid,
   547→                                targetUid = localUid,
   548→                                isVideoCall = isVideoCall,
   549→                                status = status,
   550→                                createdAt = createdAt
   551→                            ))
   552→                        }
   553→                    }
   554→                    trySend(calls)
   555→                } catch (e: Exception) {
   556→                    Timber.e(e, "Error parsing incoming calls")
   557→                }
   558→            }
   559→
   560→            override fun onCancelled(error: DatabaseError) {
   561→                Timber.e(error.toException(), "Incoming calls listener cancelled")
   562→            }
   563→        })
   564→
   565→        awaitClose { ref.removeEventListener(listener) }
   566→    }.flowOn(Dispatchers.IO)
   567→

   569→    // Online Presence Tracking
   570→    // ════════════════════════════════════════════════════════════════
   571→
   572→    /**
   573→     * Sets the user's online presence in Firebase Realtime DB.
   574→     *
   575→     * Uses the standard Firebase presence pattern:
   576→     * - Sets /presence/{uid}/online = true with lastSeen timestamp
   577→     * - Registers an onDisconnect handler that sets online = false when the
   578→     *   client disconnects (even if the app crashes)
   579→     *
   580→     * @param uid The user's UID.
   581→     */
   582→    suspend fun setOnlinePresence(uid: String) = kotlinx.coroutines.withContext(Dispatchers.IO) {
   583→        try {
   584→            val presenceRef = realtimeDb.getReference("presence").child(uid)
   585→            val now = System.currentTimeMillis()
   586→
   587→            presenceRef.child("online").setValue(true).await()
   588→            presenceRef.child("lastSeen").setValue(now).await()
   589→
   590→            // On disconnect, mark as offline
   591→            presenceRef.child("online").onDisconnect().setValue(false)
   592→            presenceRef.child("lastSeen").onDisconnect().setValue(now)
   593→
   594→            Timber.d("Online presence set for: %s", uid)
   595→        } catch (e: Exception) {
   596→            Timber.e(e, "Failed to set online presence for: %s", uid)
   597→        }
   598→    }
   599→
   600→    /**
   601→     * Observes the online status of a specific user via persistent Firebase listener.
   602→     *
   603→     * @param uid The user's UID to observe.
   604→     * @return A flow emitting [PresenceData] with online status and lastSeen.
   605→     */
   606→    fun observePresence(uid: String): Flow<PresenceData> = callbackFlow {
   607→        val ref = realtimeDb.getReference("presence").child(uid)
   608→
   609→        val listener = ref.addValueEventListener(object : ValueEventListener {
   610→            override fun onDataChange(snapshot: DataSnapshot) {
   611→                try {
   612→                    val isOnline = snapshot.child("online").getValue(Boolean::class.java) ?: false
   613→                    val lastSeen = snapshot.child("lastSeen").getValue(Long::class.java) ?: 0L
   614→                    trySend(PresenceData(isOnline = isOnline, lastSeen = lastSeen))
   615→                } catch (e: Exception) {
   616→                    Timber.e(e, "Error parsing presence for: %s", uid)
   617→                    trySend(PresenceData(isOnline = false, lastSeen = 0L))
   618→                }
   619→            }
   620→
   621→            override fun onCancelled(error: DatabaseError) {
   622→                Timber.e(error.toException(), "Presence listener cancelled for: %s", uid)
   623→                trySend(PresenceData(isOnline = false, lastSeen = 0L))
   624→            }
   625→        })
   626→
   627→        awaitClose { ref.removeEventListener(listener) }
   628→    }.flowOn(Dispatchers.IO)
