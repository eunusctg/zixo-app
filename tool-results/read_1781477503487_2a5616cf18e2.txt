   443→   443→
   444→   444→@Composable
   445→   445→private fun ZixoQrCode(
   446→   446→    data: String,
   447→   447→    size: androidx.compose.ui.unit.Dp,
   448→   448→    modifier: Modifier = Modifier
   449→   449→) {
   450→   450→    if (data.isBlank()) return
   451→   451→
   452→   452→    val density = LocalDensity.current
   453→   453→    val sizePx = with(density) { size.roundToPx() }
   454→   454→
   455→   455→    val qrBitmap = remember(data, sizePx) {
   456→   456→        generateQrBitmap(data, sizePx)
   457→   457→    }
   458→   458→
   459→   459→    Card(
   460→   460→        modifier = modifier.padding(2.dp),
   461→   461→        shape = RoundedCornerShape(8.dp),
   462→   462→        colors = CardDefaults.cardColors(containerColor = Color.White),
   463→   463→        elevation = CardDefaults.cardElevation(defaultElevation = 0.dp),
   464→   464→    ) {
   465→   465→        androidx.compose.foundation.Image(
   466→   466→            bitmap = qrBitmap.asImageBitmap(),
   467→   467→            contentDescription = "QR Code for Zixo number",
   468→   468→            modifier = Modifier
   469→   469→                .size(size)
   470→   470→                .padding(8.dp),
   471→   471→            contentScale = ContentScale.Fit
   472→   472→        )
   473→   473→    }
   474→   474→}
   475→   475→
   476→   476→/**
   477→   477→ * Generates a QR code bitmap using ZXing core library.
   478→   478→ */
   479→   479→private fun generateQrBitmap(data: String, sizePx: Int): Bitmap {
   480→   480→    val writer = QRCodeWriter()
   481→   481→    val bitMatrix = writer.encode(
   482→   482→        data,
   483→   483→        BarcodeFormat.QR_CODE,
   484→   484→        sizePx,
   485→   485→        sizePx
   486→   486→    )
   487→   487→    val width = bitMatrix.width
   488→   488→    val height = bitMatrix.height
   489→   489→    val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.RGB_565)
   490→   490→    for (x in 0 until width) {
   491→   491→        for (y in 0 until height) {
   492→   492→            bitmap.setPixel(
   493→   493→                x, y,
   494→   494→                if (bitMatrix[x, y]) AndroidColor.BLACK else AndroidColor.WHITE
   495→   495→            )
   496→   496→        }
   497→   497→    }
   498→   498→    return bitmap
   499→   499→}

   771→   771→/**
   772→   772→ * Formats an 8-digit number as two 4-digit blocks separated by a space.
   773→   773→ * e.g., "12345678" -> "1234 5678"
   774→   774→ */
   775→   775→private fun formatZixoNumber(number: String): String {
   776→   776→    val digits = number.filter { it.isDigit() }
   777→   777→    return if (digits.length >= 8) {
   778→   778→        "${digits.substring(0, 4)} ${digits.substring(4, 8)}"
   779→   779→    } else {
   780→   780→        digits
   781→   781→    }
   782→   782→}
