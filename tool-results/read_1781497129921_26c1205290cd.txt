   107→/**
   108→ * Group chat screen — extends ChatMessageScreen with group-specific features.
   109→ *
   110→ * Features:
   111→ * - Group-specific header: group name, avatar, participant count
   112→ * - Tap header to show group info panel (slide-up Liquid Glass sheet)
   113→ * - Group info panel with avatar, name, description, participant list
   114→ * - Admin badges on participant rows
   115→ * - "Add participant" button (admin only, mutual contacts)
   116→ * - "Leave group" option
   117→ * - Participant avatars in the top bar
   118→ * - Group calling: audio/video call buttons start pure WebRTC peer connections
   119→ * - WebRTC group call button in top bar
   120→ * - Group info button in top bar (navigates to group details)
   121→ * - Ephemeral timer indicator
   122→ * - Communication gate check — blocked overlay if group contains non-mutual contacts
   123→ * - Copy option in long-press action menu
   124→ * - Same keyboard avoidance (74dp input tray, reverseLayout, imePadding)
   125→ * - NavController for navigation
   126→ *
   127→ * @param threadId The ID of the group chat thread to display.
   128→ * @param navController Navigation controller for screen transitions.
   129→ * @param onBack Callback for the back navigation button.
   130→ */
   131→@OptIn(ExperimentalFoundationApi::class)
   132→@Composable
   133→fun GroupChatScreen(
   134→    threadId: String,
   135→    navController: NavController,
   136→    onBack: () -> Unit = {},
   137→    viewModel: ChatViewModel = hiltViewModel()
   138→) {
   139→    val messages by viewModel.messages.collectAsState()
   140→    val thread by viewModel.thread.collectAsState()
   141→    val inputText by viewModel.inputText.collectAsState()
   142→    val isSending by viewModel.isSending.collectAsState()
   143→    val callState by viewModel.callState.collectAsState()
   144→    val replyToMessage by viewModel.replyToMessage.collectAsState()
   145→    val showActionMenu by viewModel.showActionMenu.collectAsState()
   146→    val currentUserId by viewModel.currentUserId.collectAsState()
   147→    val errorMessage by viewModel.errorMessage.collectAsState()
   148→    val communicationGate by viewModel.communicationGate.collectAsState()
   149→
   150→    val snackbarHostState = remember { SnackbarHostState() }
   151→    val listState = rememberLazyListState()
   152→    var showGroupInfoPanel by remember { mutableStateOf(false) }
   153→
   154→    // Auto-scroll to bottom on new messages
   155→    LaunchedEffect(messages.size) {
   156→        if (messages.isNotEmpty()) {
   157→            listState.animateScrollToItem(0)
   158→        }
   159→    }
   160→
   161→    // Display error messages as snackbars
   162→    LaunchedEffect(errorMessage) {
   163→        errorMessage?.let {
   164→            snackbarHostState.showSnackbar(it)
   165→            viewModel.clearError()
   166→        }
   167→    }
   168→
   169→    // Group-specific data
   170→    val groupName = thread?.groupName ?: "Group"
   171→    val groupAvatar = thread?.groupAvatarUrl ?: ""
   172→    val participants = thread?.participantProfiles?.values?.toList() ?: emptyList()
   173→    val participantCount = participants.size
   174→    val isAdmin = thread?.groupAdminUids?.contains(currentUserId) == true
   175→    val ephemeralSeconds = thread?.ephemeralTimerSeconds ?: 0
   176→
   177→    // Check if communication is blocked (group must only contain mutual contacts)
   178→    val isBlocked = communicationGate is CommunicationGate.Blocked
   179→
   180→    Scaffold(
   181→        snackbarHost = { SnackbarHost(snackbarHostState) },
   182→        containerColor = Color.Transparent,
   183→        modifier = Modifier.fillMaxSize()
   184→    ) { innerPadding ->
   185→        Box(modifier = Modifier.fillMaxSize()) {
   186→            ZixoGlassBackground()
   187→
   188→            Column(
   189→                modifier = Modifier
   190→                    .fillMaxSize()
   191→                    .imePadding()
   192→                    .padding(innerPadding)
   193→            ) {
   194→                // ── Group Top Bar ──────────────────────────────────────────
   195→                GroupChatTopBar(
   196→                    groupName = groupName,
   197→                    groupAvatar = groupAvatar,
   198→                    participantCount = participantCount,
   199→                    participants = participants,
   200→                    onBackClick = onBack,
   201→                    onHeaderClick = { showGroupInfoPanel = true },
   202→                    onGroupInfoClick = { showGroupInfoPanel = true },
   203→                    onAudioCallClick = { viewModel.startGroupAudioCall() },
   204→                    onVideoCallClick = { viewModel.startGroupVideoCall() }
   205→                )
   206→
   207→                // ── Ephemeral Timer Indicator ─────────────────────────────
   208→                if (ephemeralSeconds > 0) {
   209→                    Row(
   210→                        modifier = Modifier
   211→                            .fillMaxWidth()
   212→                            .background(NeonMintAlpha15)
   213→                            .padding(horizontal = 16.dp, vertical = 4.dp),
   214→                        verticalAlignment = Alignment.CenterVertically,
   215→                        horizontalArrangement = Arrangement.Center
   216→                    ) {
   217→                        Text(
   218→                            text = "⏱ Messages disappear in ${formatEphemeralTimer(ephemeralSeconds)}",
   219→                            color = NeonMint,
   220→                            fontSize = 12.sp,
   221→                            fontWeight = FontWeight.Medium
   222→                        )
   223→                    }
   224→                }
   225→
   226→                // ── Message List ───────────────────────────────────────────
   227→                LazyColumn(
   228→                    state = listState,
   229→                    reverseLayout = true,
   230→                    modifier = Modifier
   231→                        .weight(1f)
   232→                        .fillMaxWidth()
   233→                        .padding(horizontal = 12.dp),
   234→                    verticalArrangement = Arrangement.spacedBy(6.dp)
   235→                ) {
   236→                    items(
   237→                        count = messages.size,
   238→                        key = { index -> messages[index].id }
   239→                    ) { index ->
   240→                        val message = messages[index]
   241→                        val isOwnMessage = message.senderUid == currentUserId
   242→
   243→                        GroupMessageBubble(
   244→                            message = message,
   245→                            isOwnMessage = isOwnMessage,
   246→                            participants = participants,
   247→                            onLongClick = { viewModel.showActionMenu.value = message }
   248→                        )
   249→                    }
   250→                }
   251→
   252→                // ── Reply Preview Bar ──────────────────────────────────────
   253→                AnimatedVisibility(
   254→                    visible = replyToMessage != null,
   255→                    enter = slideInVertically() + fadeIn(),
   256→                    exit = slideOutVertically() + fadeOut()
   257→                ) {
   258→                    replyToMessage?.let { replied ->
   259→                        GroupReplyPreviewBar(
   260→                            senderName = replied.replyToSenderName ?: replied.senderDisplayName,
   261→                            previewText = replied.replyToPreview ?: replied.content,
   262→                            onDismiss = { viewModel.clearReplyTo() }
   263→                        )
   264→                    }
   265→                }
   266→
   267→                // ── Input Tray (74dp Liquid Glass Pill) ────────────────────
   268→                GroupInputTray(
   269→                    value = inputText,
   270→                    onValueChange = { viewModel.onInputTextChanged(it) },
   271→                    onSendClick = { viewModel.sendMessage() },
   272→                    isSending = isSending,
   273→                    enabled = !isBlocked,
   274→                    modifier = Modifier.fillMaxWidth()
   275→                )
   276→            }
   277→
   278→            // ── Communication Gate Blocked Overlay ────────────────────────
   279→            if (isBlocked) {
   280→                GroupCommunicationGateOverlay(
   281→                    reason = (communicationGate as? CommunicationGate.Blocked)?.reason
   282→                        ?: "Group contains non-mutual contacts"
   283→                )
   284→            }
   285→
   286→            // ── Group Info Panel (Slide-up Sheet) ─────────────────────────
   287→            if (showGroupInfoPanel) {
   288→                GroupInfoPanel(
   289→                    groupName = groupName,
   290→                    groupAvatar = groupAvatar,
   291→                    groupDescription = thread?.groupDescription ?: "",
   292→                    participants = participants,
   293→                    currentUserId = currentUserId,
   294→                    isAdmin = isAdmin,
   295→                    onClose = { showGroupInfoPanel = false }
   296→                )
   297→            }
   298→
   299→            // ── Action Menu Overlay ────────────────────────────────────────
   300→            showActionMenu?.let { actionMessage ->
   301→                GroupActionMenuOverlay(
   302→                    message = actionMessage,
   303→                    isOwnMessage = actionMessage.senderUid == currentUserId,
   304→                    onReact = { emoji, isThreeD ->
   305→                        viewModel.addReaction(actionMessage.id, emoji, isThreeD)
   306→                    },
   307→                    onReply = {
   308→                        viewModel.setReplyTo(actionMessage.id)
   309→                    },
   310→                    onForward = {
   311→                        viewModel.forwardMessage(actionMessage.id, emptyList())
   312→                    },
   313→                    onDeleteForMe = {
   314→                        viewModel.deleteForMe(actionMessage.id)
   315→                    },
   316→                    onDeleteForEveryone = {
   317→                        viewModel.deleteForEveryone(actionMessage.id)
   318→                    },
   319→                    onDismiss = { viewModel.dismissActionMenu() }
   320→                )
   321→            }
   322→
   323→            // ── Call State Overlay ─────────────────────────────────────────
   324→            if (callState !is CallState.IDLE) {
   325→                GroupCallOverlay(
   326→                    callState = callState,
   327→                    groupName = groupName,
   328→                    groupAvatar = groupAvatar
   329→                )
   330→            }
   331→        }
   332→    }
   333→}

   449→@OptIn(ExperimentalFoundationApi::class)
   450→@Composable
   451→private fun GroupMessageBubble(
   452→    message: MessageModel,
   453→    isOwnMessage: Boolean,
   454→    participants: List<ThreadParticipant>,
   455→    onLongClick: () -> Unit
   456→) {
   457→    if (message.isDeletedForEveryone) {
   458→        Row(
   459→            modifier = Modifier
   460→                .fillMaxWidth()
   461→                .padding(start = 48.dp, end = 48.dp),
   462→            horizontalArrangement = if (isOwnMessage) Arrangement.End else Arrangement.Start
   463→        ) {
   464→            Box(
   465→                modifier = Modifier
   466→                    .clip(RoundedCornerShape(BubbleCornerSize))
   467→                    .background(IncomingBubbleColor)
   468→                    .padding(horizontal = 14.dp, vertical = 10.dp)
   469→            ) {
   470→                Text(
   471→                    text = "This message was deleted",
   472→                    color = TextSecondary,
   473→                    fontSize = 14.sp,
   474→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   475→                )
   476→            }
   477→        }
   478→        return
   479→    }
   480→
   481→    if (message.isDeletedForMe) return
   482→
   483→    val bubbleColor = if (isOwnMessage) NeonMintAlpha15 else IncomingBubbleColor
   484→    val alignment = if (isOwnMessage) Arrangement.End else Arrangement.Start
   485→    val shape = RoundedCornerShape(
   486→        topStart = BubbleCornerSize,
   487→        topEnd = BubbleCornerSize,
   488→        bottomStart = if (isOwnMessage) BubbleCornerSize else 4.dp,
   489→        bottomEnd = if (isOwnMessage) 4.dp else BubbleCornerSize
   490→    )
   491→
   492→    Row(
   493→        modifier = Modifier
   494→            .fillMaxWidth()
   495→            .padding(start = if (isOwnMessage) 48.dp else 0.dp, end = if (!isOwnMessage) 48.dp else 0.dp),
   496→        horizontalArrangement = alignment
   497→    ) {
   498→        Column(
   499→            modifier = Modifier
   500→                .clip(shape)
   501→                .background(bubbleColor)
   502→                .combinedClickable(
   503→                    interactionSource = remember { MutableInteractionSource() },
   504→                    indication = null,
   505→                    onClick = {},
   506→                    onLongClick = onLongClick
   507→                )
   508→                .padding(horizontal = 14.dp, vertical = 10.dp)
   509→        ) {
   510→            // Sender name label (group-specific: show for incoming messages)
   511→            if (!isOwnMessage) {
   512→                val sender = participants.firstOrNull { it.uid == message.senderUid }
   513→                val senderColor = senderNameColor(message.senderUid)
   514→                Text(
   515→                    text = sender?.displayName ?: message.senderDisplayName,
   516→                    color = senderColor,
   517→                    fontSize = 13.sp,
   518→                    fontWeight = FontWeight.SemiBold,
   519→                    maxLines = 1
   520→                )
   521→                Spacer(modifier = Modifier.height(2.dp))
   522→            }
   523→
   524→            // Reply reference preview
   525→            if (message.replyToPreview != null) {
   526→                Row(
   527→                    modifier = Modifier
   528→                        .fillMaxWidth()
   529→                        .clip(RoundedCornerShape(8.dp))
   530→                        .background(NeonMintAlpha15)
   531→                        .padding(horizontal = 8.dp, vertical = 4.dp)
   532→                ) {
   533→                    Text(
   534→                        text = message.replyToSenderName ?: "",
   535→                        color = NeonMint,
   536→                        fontSize = 12.sp,
   537→                        fontWeight = FontWeight.SemiBold,
   538→                        maxLines = 1
   539→                    )
   540→                    Spacer(modifier = Modifier.width(6.dp))
   541→                    Text(
   542→                        text = message.replyToPreview ?: "",
   543→                        color = TextSecondary,
   544→                        fontSize = 12.sp,
   545→                        maxLines = 1,
   546→                        overflow = TextOverflow.Ellipsis,
   547→                        modifier = Modifier.weight(1f)
   548→                    )
   549→                }
   550→                Spacer(modifier = Modifier.height(4.dp))
   551→            }
   552→
   553→            // Forwarded label
   554→            if (message.isForwarded) {
   555→                Text(
   556→                    text = "Forwarded",
   557→                    color = TextSecondary,
   558→                    fontSize = 11.sp,
   559→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   560→                )
   561→                Spacer(modifier = Modifier.height(2.dp))
   562→            }
   563→
   564→            // Media thumbnail
   565→            if (message.type == MessageContentType.IMAGE || message.type == MessageContentType.VIDEO) {
   566→                val imageUrl = message.mediaThumbnailUrl ?: message.mediaUrl
   567→                if (!imageUrl.isNullOrBlank()) {
   568→                    AsyncImage(
   569→                        model = imageUrl,
   570→                        contentDescription = "Media",
   571→                        modifier = Modifier
   572→                            .fillMaxWidth()
   573→                            .height(180.dp)
   574→                            .clip(RoundedCornerShape(10.dp)),
   575→                        contentScale = ContentScale.Crop
   576→                    )
   577→                    Spacer(modifier = Modifier.height(4.dp))
   578→                }
   579→                if (message.type == MessageContentType.VIDEO) {

  1258→/** Formats epoch milliseconds to a readable time string. */
  1259→private fun formatTimestamp(epochMs: Long): String {
  1260→    if (epochMs == 0L) return ""
  1261→    val sdf = java.text.SimpleDateFormat("HH:mm", java.util.Locale.getDefault())
  1262→    return sdf.format(java.util.Date(epochMs))
  1263→}
  1264→
  1265→/** Formats ephemeral timer seconds into a human-readable string. */
  1266→private fun formatEphemeralTimer(seconds: Int): String {
  1267→    return when {
  1268→        seconds >= 86400 -> "${seconds / 86400} day${if (seconds / 86400 > 1) "s" else ""}"
  1269→        seconds >= 3600 -> "${seconds / 3600} hour${if (seconds / 3600 > 1) "s" else ""}"
  1270→        seconds >= 60 -> "${seconds / 60} minute${if (seconds / 60 > 1) "s" else ""}"
  1271→        else -> "$seconds second${if (seconds > 1) "s" else ""}"
  1272→    }
  1273→}
