   568→@OptIn(ExperimentalMaterial3Api::class)
   569→@Composable
   570→private fun TwoStepVerificationSheet(
   571→    isCurrentlyEnabled: Boolean,
   572→    onDismiss: () -> Unit,
   573→    onEnable: (pin: String, email: String) -> Unit,
   574→    onDisable: () -> Unit
   575→) {
   576→    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
   577→    val keyboardController = LocalSoftwareKeyboardController.current
   578→
   579→    var currentStep by rememberSaveable { mutableStateOf(if (isCurrentlyEnabled) 0 else 1) }
   580→    // Step 0 = manage (toggle off), Step 1 = enter PIN, Step 2 = confirm PIN, Step 3 = email backup
   581→
   582→    var pin by rememberSaveable { mutableStateOf("") }
   583→    var confirmPin by rememberSaveable { mutableStateOf("") }
   584→    var backupEmail by rememberSaveable { mutableStateOf("") }
   585→    var pinVisible by rememberSaveable { mutableStateOf(false) }
   586→    var confirmPinVisible by rememberSaveable { mutableStateOf(false) }
   587→    var pinError by rememberSaveable { mutableStateOf<String?>(null) }
   588→
   589→    ModalBottomSheet(
   590→        onDismissRequest = onDismiss,

   792→    enabled: Boolean,

   799→        modifier = Modifier
   800→            .fillMaxWidth()
   801→            .clip(RoundedCornerShape(12.dp))
   802→            .background(bgColor)
   803→            .then(
   804→                if (enabled) Modifier.clickable(onClick = onClick) else Modifier
   805→            )
   806→            .padding(vertical = 14.dp),
   807→        contentAlignment = Alignment.Center
   808→    ) {
   809→        Text(
   810→            text = label,
   811→            color = textColor,
   812→            fontSize = 14.sp,
   813→            fontWeight = FontWeight.Bold,
   814→            letterSpacing = 1.sp
   815→        )
   816→    }
