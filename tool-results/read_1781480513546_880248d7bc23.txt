    37→/**
    38→ * Concrete implementation of [CallRepository].
    39→ *
    40→ * Manages WebRTC calls using the LiveKit Android SDK with Firebase
    41→ * Firestore for call signaling and history. All LiveKit operations
    42→ * (token fetch, room connect, track publish) run on [Dispatchers.IO]
    43→ * to prevent Main Thread blocking.
    44→ *
    45→ * Zero-trust enforcement: All call initiation methods verify mutual
    46→ * contact status through [ContactRepository.checkCommunicationGate]
    47→ * before executing. Non-contact calls are blocked at this boundary.
    48→ *
    49→ * Call state transitions:
    50→ * - Outgoing: IDLE → DIALING → CONNECTED → ENDED
    51→ * - Incoming: IDLE → RINGING → CONNECTED → ENDED
    52→ */
    53→@Singleton
    54→class CallRepositoryImpl @Inject constructor(
    55→    private val firestore: FirebaseFirestore,
    56→    private val firebaseAuth: FirebaseAuth,
    57→    private val liveKitService: LiveKitService,
    58→    private val cloudflareApiService: CloudflareApiService,
    59→    private val contactRepository: ContactRepository,
    60→    @ApplicationContext private val applicationContext: Context
    61→) : CallRepository {

    63→    private val currentUid: String?
    64→        get() = firebaseAuth.currentUser?.uid

    67→    private val usersCollection get() = firestore.collection("users")

    76→    // ── Initiate Audio Call ───────────────────────────────────────────────────
    77→
    78→    override fun initiateAudioCall(targetUid: String): Flow<CallState> = flow {
    79→        val myUid = currentUid
    80→        if (myUid == null) {
    81→            emit(CallState.IDLE)
    82→            return@flow
    83→        }
    84→
    85→        try {
    86→            _callState.value = CallState.IDLE
    87→
    88→            val gate = contactRepository.checkCommunicationGate(targetUid)
    89→            if (gate !is CommunicationGate.Allowed) {
    90→                Timber.w("Audio call blocked: not a mutual contact with %s", targetUid)
    91→                _callState.value = CallState.IDLE
    92→                emit(CallState.IDLE)
    93→                return@flow
    94→            }
    95→
    96→            val contact = gate.contact
    97→            val callId = UUID.randomUUID().toString()
    98→            currentCallId = callId
    99→
   100→            val dialingState = CallState.DIALING(
   101→                callId = callId,
   102→                targetUid = targetUid,
   103→                targetDisplayName = contact.contactDisplayName,
   104→                targetAvatarUrl = contact.contactAvatarUrl,
   105→                isVideoCall = false
   106→            )
   107→            _callState.value = dialingState
   108→            emit(dialingState)
   109→
   110→            val myProfile = usersCollection.document(myUid).get().await()
   111→            val now = System.currentTimeMillis()
   112→
   113→            val callData = hashMapOf(
   114→                "id" to callId,
   115→                "callerUid" to myUid,
   116→                "callerDisplayName" to (myProfile.getString("displayName") ?: ""),
   117→                "callerAvatarUrl" to (myProfile.getString("photoUrl") ?: ""),
   118→                "calleeUid" to targetUid,
   119→                "calleeDisplayName" to contact.contactDisplayName,
   120→                "calleeAvatarUrl" to contact.contactAvatarUrl,
   121→                "isVideoCall" to false,
   122→                "status" to "DIALING",
   123→                "createdAt" to now,
   124→                "threadId" to ""
   125→            )
   126→            callsCollection.document(callId).set(callData).await()
   127→
   128→            val roomName = "call_$callId"
   129→            val tokenResponse = cloudflareApiService.generateLiveKitToken(
   130→                identity = myUid,
   131→                roomName = roomName
   132→            ).firstOrNull()
   133→
   134→            if (tokenResponse == null) {
   135→                Timber.e("Failed to fetch LiveKit token for audio call")
   136→                endCallInternal(CallEndReason.NETWORK_ERROR)
   137→                emit(_callState.value)
   138→                return@flow
   139→            }
   140→
   141→            liveKitService.connect(tokenResponse.wsUrl, tokenResponse.token)
   142→                .flowOn(Dispatchers.IO)
   143→                .collect {}
   144→
   145→            liveKitService.startAudioCall(roomName)
   146→                .flowOn(Dispatchers.IO)
   147→                .collect {}
   148→
   149→            callsCollection.document(callId).update("status", "CONNECTED").await()
   150→
   151→            callStartTime = System.currentTimeMillis()
   152→            startDurationTracker()
   153→
   154→            CallForegroundService.start(
   155→                applicationContext,
   156→                CallForegroundService.CALL_TYPE_AUDIO,
   157→                contact.contactDisplayName,
   158→                roomName
   159→            )
   160→
   161→            val connectedState = CallState.CONNECTED(
   162→                callId = callId,
   163→                targetUid = targetUid,
   164→                targetDisplayName = contact.contactDisplayName,
   165→                targetAvatarUrl = contact.contactAvatarUrl,
   166→                isVideoCall = false,
   167→                connectedAt = System.currentTimeMillis()
   168→            )
   169→            _callState.value = connectedState
   170→            emit(connectedState)
   171→        } catch (e: Exception) {
   172→            Timber.e(e, "Failed to initiate audio call")
   173→            endCallInternal(CallEndReason.NETWORK_ERROR)
   174→            emit(_callState.value)
   175→        }
   176→    }.flowOn(Dispatchers.IO)

   500→    // ── Accept Call ───────────────────────────────────────────────────────────
   501→
   502→    override suspend fun acceptCall(callId: String) {
   503→        try {
   504→            val myUid = currentUid ?: return
   505→            withContext(Dispatchers.IO) {
   506→                val callDoc = callsCollection.document(callId).get().await()
   507→                if (!callDoc.exists()) {
   508→                    Timber.w("Call document not found: %s", callId)
   509→                    return@withContext
   510→                }
   511→
   512→                val status = callDoc.getString("status")
   513→                if (status != "DIALING" && status != "RINGING") {
   514→                    Timber.w("Cannot accept call in state: %s", status)
   515→                    return@withContext
   516→                }
   517→
   518→                val roomName = "call_$callId"
   519→                val tokenResponse = cloudflareApiService.generateLiveKitToken(
   520→                    identity = myUid,
   521→                    roomName = roomName
   522→                ).firstOrNull()
   523→
   524→                if (tokenResponse == null) {
   525→                    Timber.e("Failed to fetch LiveKit token for accepting call")
   526→                    return@withContext
   527→                }
   528→
   529→                liveKitService.connect(tokenResponse.wsUrl, tokenResponse.token)
   530→                    .flowOn(Dispatchers.IO)
   531→                    .collect {}
   532→
   533→                val isVideoCall = callDoc.getBoolean("isVideoCall") ?: false
   534→                if (isVideoCall) {
   535→                    liveKitService.startVideoCall(roomName)
   536→                        .flowOn(Dispatchers.IO)
   537→                        .collect {}
   538→                } else {
   539→                    liveKitService.startAudioCall(roomName)
   540→                        .flowOn(Dispatchers.IO)
   541→                        .collect {}
   542→                }
   543→
   544→                callsCollection.document(callId).update("status", "CONNECTED").await()
   545→
   546→                callStartTime = System.currentTimeMillis()
   547→                startDurationTracker()
   548→
   549→                CallForegroundService.start(
   550→                    applicationContext,
   551→                    if (isVideoCall) CallForegroundService.CALL_TYPE_VIDEO
   552→                    else CallForegroundService.CALL_TYPE_AUDIO,
   553→                    callDoc.getString("callerDisplayName") ?: "",
   554→                    roomName
   555→                )
   556→
   557→                val connectedState = CallState.CONNECTED(
   558→                    callId = callId,
   559→                    targetUid = callDoc.getString("callerUid") ?: "",
   560→                    targetDisplayName = callDoc.getString("callerDisplayName") ?: "",
   561→                    targetAvatarUrl = callDoc.getString("callerAvatarUrl") ?: "",
   562→                    isVideoCall = isVideoCall,
   563→                    connectedAt = System.currentTimeMillis()
   564→                )
   565→                _callState.value = connectedState
   566→            }
   567→        } catch (e: Exception) {
   568→            Timber.e(e, "Failed to accept call: %s", callId)
   569→        }
   570→    }

   572→    // ── Decline Call ──────────────────────────────────────────────────────────
   573→
   574→    override suspend fun declineCall(callId: String) {
   575→        try {
   576→            withContext(Dispatchers.IO) {
   577→                callsCollection.document(callId).update(
   578→                    mapOf(
   579→                        "status" to "DECLINED",
   580→                        "endedAt" to System.currentTimeMillis(),
   581→                        "endReason" to CallEndReason.DECLINED.name
   582→                    )
   583→                ).await()
   584→
   585→                _callState.value = CallState.ENDED(
   586→                    callId = callId,
   587→                    endReason = CallEndReason.DECLINED
   588→                )
   589→                currentCallId = null
   590→            }
   591→        } catch (e: Exception) {
   592→            Timber.e(e, "Failed to decline call: %s", callId)
   593→        }
   594→    }

   596→    // ── End Call ──────────────────────────────────────────────────────────────
   597→
   598→    override suspend fun endCall() {
   599→        try {
   600→            endCallInternal(CallEndReason.COMPLETED)
   601→        } catch (e: Exception) {
   602→            Timber.e(e, "Failed to end call")
   603→        }
   604→    }

   606→    private suspend fun endCallInternal(reason: CallEndReason) {
   607→        val callId = currentCallId
   608→        val duration = if (callStartTime > 0) {
   609→            (System.currentTimeMillis() - callStartTime) / 1000
   610→        } else 0L
   611→
   612→        try {
   613→            liveKitService.disconnect()
   614→        } catch (e: Exception) {
   615→            Timber.e(e, "Error disconnecting from LiveKit room")
   616→        }
   617→
   618→        CallForegroundService.stop(applicationContext)
   619→
   620→        callDurationJob?.cancel()
   621→        callDurationJob = null
   622→
   623→        if (callId != null) {
   624→            try {
   625→                callsCollection.document(callId).update(
   626→                    mapOf(
   627→                        "status" to "ENDED",
   628→                        "endedAt" to System.currentTimeMillis(),
   629→                        "endReason" to reason.name,
   630→                        "durationSeconds" to duration
   631→                    )
   632→                ).await()
   633→
   634→                val callDoc = callsCollection.document(callId).get().await()
   635→                if (callDoc.exists()) {
   636→                    saveCallHistory(callDoc, duration, reason)
   637→                }
   638→            } catch (e: Exception) {
   639→                Timber.e(e, "Failed to update call document on end")
   640→            }
   641→        }
   642→
   643→        _callState.value = CallState.ENDED(
   644→            callId = callId ?: "",
   645→            durationSeconds = duration,
   646→            endReason = reason
   647→        )
   648→        currentCallId = null
   649→        callStartTime = 0L
   650→    }

   697→    override fun observeCallDuration(): Flow<Long> = callbackFlow {
   698→        while (isActive) {
   699→            if (callStartTime > 0L && _callState.value is CallState.CONNECTED) {
   700→                val duration = (System.currentTimeMillis() - callStartTime) / 1000
   701→                trySend(duration)
   702→            }
   703→            delay(1000L)
   704→        }
   705→        awaitClose {}
   706→    }.flowOn(Dispatchers.IO)

   708→    // ── Call History ──────────────────────────────────────────────────────────
   709→
   710→    override fun observeCallHistory(): Flow<List<CallHistoryEntry>> = callbackFlow {
   711→        val myUid = currentUid
   712→        if (myUid == null) {
   713→            trySend(emptyList())
   714→            close()
   715→            return@callbackFlow
   716→        }
   717→
   718→        val subscription = firestore.collection("users").document(myUid)
   719→            .collection("call_history")
   720→            .orderBy("timestamp", Query.Direction.DESCENDING)
   721→            .addSnapshotListener { snapshot, error ->
   722→                if (error != null) {
   723→                    Timber.e(error, "Error observing call history")
   724→                    trySend(emptyList())
   725→                    return@addSnapshotListener
   726→                }
   727→
   728→                val entries = snapshot?.documents?.mapNotNull { doc ->
   729→                    try {
   730→                        mapToCallHistoryEntry(doc)
   731→                    } catch (e: Exception) {
   732→                        Timber.e(e, "Failed to map call history entry: %s", doc.id)
   733→                        null
   734→                    }
   735→                } ?: emptyList()
   736→
   737→                trySend(entries)
   738→            }
   739→
   740→        awaitClose { subscription.remove() }
   741→    }.flowOn(Dispatchers.IO)

   749→    private suspend fun saveCallHistory(
   750→        callDoc: com.google.firebase.firestore.DocumentSnapshot,
   751→        durationSeconds: Long,
   752→        endReason: CallEndReason
   753→    ) {
   754→        val myUid = currentUid ?: return
   755→
   756→        val entry = CallHistoryEntry(
   757→            id = UUID.randomUUID().toString(),
   758→            callId = callDoc.id,
   759→            callerUid = callDoc.getString("callerUid") ?: "",
   760→            callerDisplayName = callDoc.getString("callerDisplayName") ?: "",
   761→            callerAvatarUrl = callDoc.getString("callerAvatarUrl") ?: "",
   762→            calleeUid = callDoc.getString("calleeUid") ?: "",
   763→            calleeDisplayName = callDoc.getString("calleeDisplayName") ?: "",
   764→            calleeAvatarUrl = callDoc.getString("calleeAvatarUrl") ?: "",
   765→            isVideoCall = callDoc.getBoolean("isVideoCall") ?: false,
   766→            durationSeconds = durationSeconds,
   767→            timestamp = callDoc.getLong("createdAt") ?: System.currentTimeMillis(),
   768→            endReason = endReason,
   769→            threadId = callDoc.getString("threadId") ?: ""
   770→        )
   771→
   772→        firestore.collection("users").document(myUid)
   773→            .collection("call_history").document(entry.id)
   774→            .set(entry.toFirestoreMap()).await()
   775→
   776→        Timber.d("Call history saved: %s", entry.callId)
   777→    }

   779→    private fun mapToCallHistoryEntry(
   780→        doc: com.google.firebase.firestore.DocumentSnapshot
   781→    ): CallHistoryEntry {
   782→        val endReasonStr = doc.getString("endReason") ?: "COMPLETED"
   783→        return CallHistoryEntry(
   784→            id = doc.id,
   785→            callId = doc.getString("callId") ?: "",
   786→            callerUid = doc.getString("callerUid") ?: "",
   787→            callerDisplayName = doc.getString("callerDisplayName") ?: "",
   788→            callerAvatarUrl = doc.getString("callerAvatarUrl") ?: "",
   789→            calleeUid = doc.getString("calleeUid") ?: "",
   790→            calleeDisplayName = doc.getString("calleeDisplayName") ?: "",
   791→            calleeAvatarUrl = doc.getString("calleeAvatarUrl") ?: "",
   792→            isVideoCall = doc.getBoolean("isVideoCall") ?: false,
   793→            durationSeconds = doc.getLong("durationSeconds") ?: 0L,
   794→            timestamp = doc.getLong("timestamp") ?: 0L,
   795→            endReason = try {
   796→                CallEndReason.valueOf(endReasonStr)
   797→            } catch (_: Exception) {
   798→                CallEndReason.COMPLETED
   799→            },
   800→            threadId = doc.getString("threadId") ?: ""
   801→        )
   802→    }
