    41→/**
    42→ * Concrete implementation of [CallRepository].
    43→ *
    44→ * Manages WebRTC calls using pure WebRTC (io.github.webrtc-sdk:android)
    45→ * with Firebase Realtime Database for signaling. NO LiveKit dependency.
    46→ *
    47→ * All WebRTC operations (PeerConnectionFactory, SDP, ICE) run on
    48→ * [Dispatchers.IO]. The Android Main Thread is NEVER blocked by call operations.
    49→ *
    50→ * Zero-trust enforcement: All call initiation methods verify mutual
    51→ * contact status through [ContactRepository.verifyMutualContact]
    52→ * before executing. Non-contact calls are blocked at this boundary.
    53→ */
    54→@Singleton
    55→class CallRepositoryImpl @Inject constructor(
    56→    private val firestore: FirebaseFirestore,
    57→    private val realtimeDb: FirebaseDatabase,
    58→    private val firebaseAuth: FirebaseAuth,
    59→    private val contactRepository: ContactRepository,
    60→    private val webRtcClient: WebRtcClient,
    61→    private val signalingClient: FirebaseSignalingClient,
    62→    @ApplicationContext private val context: Context
    63→) : CallRepository {
    64→
    65→    private val currentUid: String?
    66→        get() = firebaseAuth.currentUser?.uid

    82→    /**
    83→     * Observes the real-time state of a call session from Firebase Realtime DB.
    84→     *
    85→     * Provides a continuous [com.google.firebase.database.ValueEventListener] on
    86→     * `/calls/{callId}/`, streaming every state change as a Map. All listener
    87→     * operations are safely decoupled from the UI thread via [callbackFlow] on
    88→     * [Dispatchers.IO].
    89→     *
    90→     * This is the primary mechanism by which the calling UI reacts to remote
    91→     * state transitions without polling.
    92→     */
    93→    override fun observeCallSession(callId: String): Flow<Map<String, Any>> = callbackFlow {
    94→        val callRef = realtimeDb.getReference("calls/$callId")
    95→        val listener = object : com.google.firebase.database.ValueEventListener {
    96→            override fun onDataChange(snapshot: com.google.firebase.database.DataSnapshot) {
    97→                try {
    98→                    val data = snapshot.value as? Map<String, Any> ?: emptyMap()
    99→                    trySend(data)
   100→                } catch (e: Exception) {
   101→                    Timber.e(e, "Error parsing call session data for callId: %s", callId)
   102→                }
   103→            }
   104→
   105→            override fun onCancelled(error: com.google.firebase.database.DatabaseError) {
   106→                Timber.e(error.toException(), "Call session listener cancelled for callId: %s", callId)
   107→                close(error.toException())
   108→            }
   109→        }
   110→        callRef.addValueEventListener(listener)
   111→        awaitClose { callRef.removeEventListener(listener) }
   112→    }.flowOn(Dispatchers.IO)
   113→
   114→    /**
   115→     * Observes ICE candidates for a specific call from Firebase Realtime DB.
   116→     *
   117→     * Listens to `/calls/{callId}/iceCandidates/` with a [com.google.firebase.database.ChildEventListener],
   118→     * streaming each newly added ICE candidate as it arrives from the remote peer.
   119→     * Candidates are parsed into [IceCandidate] objects and forwarded to the
   120→     * WebRTC engine for addition to the PeerConnection.
   121→     */
   122→    override fun observeIceCandidates(callId: String): Flow<IceCandidate> = callbackFlow {
   123→        val candidatesRef = realtimeDb.getReference("calls/$callId/iceCandidates")
   124→        val childListener = object : com.google.firebase.database.ChildEventListener {
   125→            override fun onChildAdded(
   126→                snapshot: com.google.firebase.database.DataSnapshot,
   127→                previousChildName: String?
   128→            ) {
   129→                try {
   130→                    val sdpMid = snapshot.child("sdpMid").getValue(String::class.java) ?: ""
   131→                    val sdpMLineIndex = snapshot.child("sdpMLineIndex").getValue(Int::class.java) ?: 0
   132→                    val sdp = snapshot.child("sdp").getValue(String::class.java) ?: ""
   133→                    if (sdp.isNotBlank()) {
   134→                        trySend(IceCandidate(sdpMid, sdpMLineIndex, sdp))
   135→                    }
   136→                } catch (e: Exception) {
   137→                    Timber.e(e, "Error parsing ICE candidate for callId: %s", callId)
   138→                }
   139→            }
   140→
   141→            override fun onChildChanged(
   142→                snapshot: com.google.firebase.database.DataSnapshot,
   143→                previousChildName: String?
   144→            ) { /* ICE candidates are append-only */ }
   145→
   146→            override fun onChildRemoved(snapshot: com.google.firebase.database.DataSnapshot) {}
   147→
   148→            override fun onChildMoved(
   149→                snapshot: com.google.firebase.database.DataSnapshot,
   150→                previousChildName: String?
   151→            ) {}
   152→
   153→            override fun onCancelled(error: com.google.firebase.database.DatabaseError) {
   154→                Timber.e(error.toException(), "ICE candidate listener cancelled for callId: %s", callId)
   155→                close(error.toException())
   156→            }
   157→        }
   158→        candidatesRef.addChildEventListener(childListener)
   159→        awaitClose { candidatesRef.removeEventListener(childListener) }
   160→    }.flowOn(Dispatchers.IO)
   161→
   162→    /**
   163→     * Emits a call state update to Firebase Realtime DB.
   164→     *
   165→     * Writes the state string to `/calls/{callId}/callState`.
   166→     * This operation runs on [Dispatchers.IO] and never blocks the Main Thread.
   167→     */
   168→    override suspend fun emitCallState(callId: String, state: String) = withContext(Dispatchers.IO) {
   169→        try {
   170→            realtimeDb.getReference("calls/$callId/callState").setValue(state)
   171→            Timber.d("Call state emitted: %s for callId: %s", state, callId)
   172→        } catch (e: Exception) {
   173→            Timber.e(e, "Failed to emit call state: %s for callId: %s", state, callId)
   174→        }
   175→    }

   177→    // ── Initiate Call ─────────────────────────────────────────────────────────
   178→
   179→    override fun initiateCall(targetUid: String, isVideoCall: Boolean): Flow<CallState> = flow {
   180→        val myUid = currentUid
   181→        if (myUid == null) {
   182→            emit(CallState.IDLE)
   183→            return@flow
   184→        }
   185→
   186→        try {
   187→            // Zero-trust: verify mutual contact before initiating
   188→            var gateResult: CommunicationGate? = null
   189→            contactRepository.verifyMutualContact(targetUid).collect { gateResult = it }
   190→
   191→            if (gateResult !is CommunicationGate.Allowed) {
   192→                Timber.w("Call blocked — not a mutual contact: %s", targetUid)
   193→                emit(CallState.IDLE)
   194→                return@flow
   195→            }
   196→
   197→            val contact = (gateResult as CommunicationGate.Allowed).contact
   198→
   199→            // Create the call document in Firebase Realtime DB
   200→            val callId = UUID.randomUUID().toString()
   201→
   202→            _callState.value = CallState.DIALING(
   203→                callId = callId,
   204→                targetUid = targetUid,
   205→                targetDisplayName = contact.contactDisplayName,
   206→                targetAvatarUrl = contact.contactAvatarUrl,
   207→                isVideoCall = isVideoCall
   208→            )
   209→            emit(_callState.value)
   210→
   211→            // Start foreground service BEFORE WebRTC initialization
   212→            // This guarantees Android does not terminate media streams
   213→            CallForegroundService.start(
   214→                context = context,
   215→                callType = if (isVideoCall) CallForegroundService.CALL_TYPE_VIDEO else CallForegroundService.CALL_TYPE_AUDIO,
   216→                callerName = contact.contactDisplayName,
   217→                callId = callId
   218→            )
   219→
   220→            // Create the call entry in Firebase Realtime DB with structured data
   221→            signalingClient.createCall(callId, myUid, targetUid, isVideoCall)
   222→
   223→            // Initialize WebRTC peer connection on Dispatchers.IO
   224→            // AudioManager is calibrated inside initializePeerConnection
   225→            webRtcClient.initializePeerConnection(isVideoCall)
   226→
   227→            // Set up ICE candidate forwarding: local ICE → Firebase signaling
   228→            webRtcClient.onIceCandidateGenerated = { candidate ->
   229→                GlobalScope.launch(Dispatchers.IO) {
   230→                    try {
   231→                        signalingClient.sendIceCandidate(
   232→                            callId = callId,
   233→                            senderUid = myUid,
   234→                            targetUid = targetUid,
   235→                            sdpMid = candidate.sdpMid,
   236→                            sdpMLineIndex = candidate.sdpMLineIndex,
   237→                            sdp = candidate.sdp
   238→                        )
   239→                    } catch (e: Exception) {
   240→                        Timber.e(e, "Failed to forward ICE candidate to signaling")
   241→                    }
   242→                }
   243→            }
   244→
   245→            // Create SDP offer
   246→            val sdpOffer = webRtcClient.createOffer()
   247→
   248→            // Send the offer via Firebase signaling
   249→            signalingClient.sendOffer(callId, myUid, targetUid, sdpOffer)
   250→
   251→            // Update call status to ringing
   252→            signalingClient.updateCallStatus(callId, "ringing")
   253→
   254→            // Observe answer and ICE candidates concurrently
   255→            var isAnswerReceived = false
   256→
   257→            // Listen for answer
   258→            signalingClient.observeAnswer(callId, myUid).collect { answerData ->
   259→                if (answerData != null && !isAnswerReceived) {
   260→                    isAnswerReceived = true
   261→                    webRtcClient.setRemoteAnswer(answerData.sdp)
   262→
   263→                    // Update call status to connected
   264→                    signalingClient.updateCallStatus(callId, "connected")
   265→
   266→                    _callState.value = CallState.CONNECTED(
   267→                        callId = callId,
   268→                        targetUid = targetUid,
   269→                        targetDisplayName = contact.contactDisplayName,
   270→                        targetAvatarUrl = contact.contactAvatarUrl,
   271→                        isVideoCall = isVideoCall,
   272→                        connectedAt = System.currentTimeMillis()
   273→                    )
   274→                    connectedAt = System.currentTimeMillis()
   275→                    emit(_callState.value)
   276→
   277→                    // Start duration tracking
   278→                    startDurationTracking(callId)
   279→
   280→                    // Start observing ICE candidates for this call
   281→                    observeIceCandidates(callId, myUid)
   282→                }
   283→            }
   284→        } catch (e: Exception) {
   285→            Timber.e(e, "Failed to initiate call")
   286→            _callState.value = CallState.IDLE
   287→            emit(CallState.IDLE)
   288→        }
   289→    }.flowOn(Dispatchers.IO)

   291→    // ── Observe Incoming Calls ────────────────────────────────────────────────
   292→
   293→    override fun observeIncomingCalls(): Flow<CallState> = callbackFlow {
   294→        val myUid = currentUid
   295→        if (myUid == null) {
   296→            trySend(CallState.IDLE)
   297→            close()
   298→            return@callbackFlow
   299→        }
   300→
   301→        val callsRef = realtimeDb.getReference("calls")
   302→        val listener = callsRef.orderByChild("calleeUid").equalTo(myUid)
   303→            .addChildEventListener(object : com.google.firebase.database.ChildEventListener {
   304→                override fun onChildAdded(
   305→                    snapshot: com.google.firebase.database.DataSnapshot,
   306→                    previousChildName: String?
   307→                ) {
   308→                    try {
   309→                        val status = snapshot.child("status").getValue(String::class.java)
   310→                        if (status == "ringing") {
   311→                            val callId = snapshot.key ?: return
   312→                            val callerUid = snapshot.child("callerUid").getValue(String::class.java) ?: return
   313→                            val callerDisplayName = snapshot.child("callerDisplayName").getValue(String::class.java) ?: ""
   314→                            val isVideoCall = snapshot.child("isVideoCall").getValue(Boolean::class.java) ?: false
   315→
   316→                            val incomingState = CallState.RINGING(
   317→                                callId = callId,
   318→                                callerUid = callerUid,
   319→                                callerDisplayName = callerDisplayName,
   320→                                callerAvatarUrl = "",
   321→                                isVideoCall = isVideoCall
   322→                            )
   323→                            _incomingCalls.value = incomingState
   324→                            trySend(incomingState)
   325→                        }
   326→                    } catch (e: Exception) {
   327→                        Timber.e(e, "Error parsing incoming call")
   328→                    }
   329→                }
   330→
   331→                override fun onChildChanged(
   332→                    snapshot: com.google.firebase.database.DataSnapshot,
   333→                    previousChildName: String?
   334→                ) { /* handled in observeCallState */ }
   335→
   336→                override fun onChildRemoved(snapshot: com.google.firebase.database.DataSnapshot) {}
   337→                override fun onChildMoved(snapshot: com.google.firebase.database.DataSnapshot, previousChildName: String?) {}
   338→                override fun onCancelled(error: com.google.firebase.database.DatabaseError) {
   339→                    Timber.e(error.toException(), "Incoming calls listener cancelled")
   340→                }
   341→            })
   342→
   343→        awaitClose { callsRef.removeEventListener(listener) }
   344→    }.flowOn(Dispatchers.IO)

   346→    // ── Answer Call ───────────────────────────────────────────────────────────
   347→
   348→    override fun answerCall(callId: String): Flow<CallState> = flow {
   349→        try {
   350→            val myUid = currentUid ?: throw IllegalStateException("Not authenticated")
   351→
   352→            // Get call data
   353→            val callSnapshot = realtimeDb.getReference("calls").child(callId).get().await()
   354→            val callerUid = callSnapshot.child("callerUid").getValue(String::class.java) ?: ""
   355→            val isVideoCall = callSnapshot.child("isVideoCall").getValue(Boolean::class.java) ?: false
   356→
   357→            // Verify mutual contact
   358→            var gateResult: CommunicationGate? = null
   359→            contactRepository.verifyMutualContact(callerUid).collect { gateResult = it }
   360→
   361→            if (gateResult !is CommunicationGate.Allowed) {
   362→                Timber.w("Cannot answer call — not a mutual contact")
   363→                emit(CallState.IDLE)
   364→                return@flow
   365→            }
   366→
   367→            val contact = (gateResult as CommunicationGate.Allowed).contact
   368→
   369→            // Update call status
   370→            realtimeDb.getReference("calls").child(callId).child("status")
   371→                .setValue("connected").await()
   372→
   373→            // Start foreground service
   374→            CallForegroundService.start(
   375→                context = context,
   376→                callType = if (isVideoCall) CallForegroundService.CALL_TYPE_VIDEO else CallForegroundService.CALL_TYPE_AUDIO,
   377→                callerName = contact.contactDisplayName,
   378→                callId = callId
   379→            )
   380→
   381→            // Initialize WebRTC and observe offer
   382→            webRtcClient.initializePeerConnection(isVideoCall)
   383→
   384→            signalingClient.observeOffer(callId).collect { offer ->
   385→                if (offer != null) {
   386→                    webRtcClient.setRemoteOffer(offer)
   387→                    val sdpAnswer = webRtcClient.createAnswer()
   388→                    signalingClient.sendAnswer(callId, myUid, sdpAnswer)
   389→
   390→                    _callState.value = CallState.CONNECTED(
   391→                        callId = callId,
   392→                        targetUid = callerUid,
   393→                        targetDisplayName = contact.contactDisplayName,
   394→                        targetAvatarUrl = contact.contactAvatarUrl,
   395→                        isVideoCall = isVideoCall,
   396→                        connectedAt = System.currentTimeMillis()
   397→                    )
   398→                    connectedAt = System.currentTimeMillis()
   399→                    emit(_callState.value)
   400→                    startDurationTracking(callId)
   401→                }
   402→            }
   403→        } catch (e: Exception) {
   404→            Timber.e(e, "Failed to answer call")
   405→            _callState.value = CallState.IDLE
   406→            emit(CallState.IDLE)
   407→        }
   408→    }.flowOn(Dispatchers.IO)

   427→    // ── End Call ──────────────────────────────────────────────────────────────
   428→
   429→    override fun endCall(callId: String): Flow<Result<Unit>> = flow {
   430→        try {
   431→            val currentState = _callState.value
   432→            val durationSeconds = if (currentState is CallState.CONNECTED) {
   433→                (System.currentTimeMillis() - connectedAt) / 1000
   434→            } else 0L
   435→
   436→            // Update call status in Firebase via signaling client
   437→            signalingClient.updateCallStatus(callId, "ended")
   438→
   439→            // Clean up WebRTC resources (releases audio focus too)
   440→            webRtcClient.disconnect()
   441→
   442→            // Clean up signaling data from Firebase Realtime DB
   443→            signalingClient.cleanupSignalingData(callId)
   444→
   445→            // Stop foreground service
   446→            CallForegroundService.stop(context)
   447→
   448→            // Cancel duration tracking
   449→            durationJob?.cancel()
   450→            durationJob = null
   451→
   452→            // Log the call
   453→            logCall(callId, currentState, durationSeconds)
   454→
   455→            _callState.value = CallState.ENDED(
   456→                callId = callId,
   457→                durationSeconds = durationSeconds,
   458→                endReason = CallEndReason.COMPLETED
   459→            )
   460→            emit(Result.success(Unit))
   461→        } catch (e: Exception) {
   462→            Timber.e(e, "Failed to end call — forcing cleanup")
   463→            // Force cleanup even if something fails
   464→            try { webRtcClient.disconnect() } catch (_: Exception) {}
   465→            try { CallForegroundService.stop(context) } catch (_: Exception) {}
   466→            try { signalingClient.cleanupSignalingData(callId) } catch (_: Exception) {}
   467→            _callState.value = CallState.IDLE
   468→            emit(Result.failure(e))
   469→        }
   470→    }.flowOn(Dispatchers.IO)

   524→    // ── Observe Call State ────────────────────────────────────────────────────
   525→
   526→    override fun observeCallState(callId: String): Flow<CallState> = flow {
   527→        callState.collect { state ->
   528→            if (state is CallState.IDLE ||
   529→                (state is CallState.DIALING && state.callId == callId) ||
   530→                (state is CallState.RINGING && state.callId == callId) ||
   531→                (state is CallState.CONNECTED && state.callId == callId) ||
   532→                (state is CallState.ENDED && state.callId == callId)
   533→            ) {
   534→                emit(state)
   535→            }
   536→        }
   537→    }.flowOn(Dispatchers.IO)

   539→    // ── Call History ──────────────────────────────────────────────────────────
   540→
   541→    override fun getCallHistory(): Flow<List<CallLogEntry>> = callbackFlow {
   542→        val myUid = currentUid
   543→        if (myUid == null) {
   544→            trySend(emptyList())
   545→            close()
   546→            return@callbackFlow
   547→        }
   548→
   549→        val subscription = callLogCollection
   550→            .whereArrayContains("participantUids", myUid)
   551→            .orderBy("timestamp", Query.Direction.DESCENDING)
   552→            .addSnapshotListener { snapshot, error ->
   553→                if (error != null) {

   574→    // ── ICE Candidate Observer ──────────────────────────────────────────────

   576→    /**
   577→     * Observes remote ICE candidates from the signaling server and adds them
   578→     * to the local PeerConnection. This runs continuously during a call to
   579→     * ensure that new ICE candidates are processed as they arrive.
   580→     *
   581→     * @param callId The call ID.
   582→     * @param localUid The local user's UID (to filter out self-sent candidates).
   583→     */
   584→    private fun observeIceCandidates(callId: String, localUid: String) {
   585→        kotlinx.coroutines.GlobalScope.launch(Dispatchers.IO) {
   586→            try {
   587→                signalingClient.observeIceCandidates(callId, localUid).collect { candidates ->
   588→                    for (candidate in candidates) {
   589→                        try {
   590→                            webRtcClient.addIceCandidate(
   591→                                candidate.sdpMid,
   592→                                candidate.sdpMLineIndex,
   593→                                candidate.sdp
   594→                            )
   595→                        } catch (e: Exception) {
   596→                            Timber.e(e, "Failed to add remote ICE candidate")
   597→                        }
   598→                    }
   599→                }
   600→            } catch (e: Exception) {
   601→                Timber.e(e, "ICE candidate observer failed for call: %s", callId)
   602→            }
   603→        }
   604→    }

   608→    private fun startDurationTracking(callId: String) {
   609→        durationJob?.cancel()
   610→        durationJob = kotlinx.coroutines.GlobalScope.launch(Dispatchers.IO) {
   611→            while (isActive) {
   612→                delay(1000)
   613→                val currentState = _callState.value
   614→                if (currentState is CallState.CONNECTED) {
   615→                    val elapsed = (System.currentTimeMillis() - connectedAt) / 1000
   616→                    _callState.value = currentState.copy(durationSeconds = elapsed)
   617→                }
   618→            }
   619→        }
   620→    }

   622→    private suspend fun logCall(callId: String, state: CallState, durationSeconds: Long) {
   623→        try {
   624→            val myUid = currentUid ?: return
   625→            val entry = when (state) {
   626→                is CallState.CONNECTED -> CallLogEntry(
   627→                    id = UUID.randomUUID().toString(),
   628→                    callId = callId,
   629→                    callerUid = myUid,
   630→                    calleeUid = state.targetUid,
   631→                    callerName = firebaseAuth.currentUser?.displayName ?: "",
   632→                    calleeName = state.targetDisplayName,
   633→                    type = CallDirection.OUTGOING,
   634→                    isVideoCall = state.isVideoCall,
   635→                    callType = if (state.isVideoCall) com.zixo.app.domain.model.CallTechnology.WEBRTC_VIDEO else com.zixo.app.domain.model.CallTechnology.WEBRTC_AUDIO,
   636→                    duration = durationSeconds,
   637→                    timestamp = System.currentTimeMillis(),
   638→                    endReason = CallEndReason.COMPLETED
   639→                )
   640→                is CallState.DIALING -> CallLogEntry(
   641→                    id = UUID.randomUUID().toString(),
   642→                    callId = callId,
   643→                    callerUid = myUid,
   644→                    calleeUid = state.targetUid,
   645→                    callerName = firebaseAuth.currentUser?.displayName ?: "",
   646→                    calleeName = state.targetDisplayName,
   647→                    type = CallDirection.OUTGOING,
   648→                    isVideoCall = state.isVideoCall,
   649→                    callType = if (state.isVideoCall) com.zixo.app.domain.model.CallTechnology.WEBRTC_VIDEO else com.zixo.app.domain.model.CallTechnology.WEBRTC_AUDIO,
   650→                    duration = 0L,
   651→                    timestamp = System.currentTimeMillis(),
   652→                    endReason = CallEndReason.CANCELLED
   653→                )
   654→                else -> return
   655→            }
   656→
   657→            callLogCollection.document(entry.id).set(entryToFirestoreMap(entry)).await()
   658→        } catch (e: Exception) {
   659→            Timber.e(e, "Failed to log call")
   660→        }
   661→    }

   663→    private fun mapToCallLogEntry(
   664→        doc: com.google.firebase.firestore.DocumentSnapshot,
   665→        myUid: String
   666→    ): CallLogEntry? {
   667→        return try {
   668→            val callerUid = doc.getString("callerUid") ?: return null
   669→            val calleeUid = doc.getString("calleeUid") ?: return null
   670→
   671→            val direction = when {
   672→                callerUid == myUid -> CallDirection.OUTGOING
   673→                else -> CallDirection.INCOMING
   674→            }
   675→
   676→            val endReasonStr = doc.getString("endReason") ?: "COMPLETED"
   677→            val endReason = try { CallEndReason.valueOf(endReasonStr) }
   678→                catch (_: Exception) { CallEndReason.COMPLETED }
   679→
   680→            val callTechnologyStr = doc.getString("callType") ?: "WEBRTC_AUDIO"
   681→
   682→            CallLogEntry(
   683→                id = doc.id,
   684→                callId = doc.getString("callId") ?: "",
   685→                callerUid = callerUid,
   686→                calleeUid = calleeUid,
   687→                callerName = doc.getString("callerName") ?: "",
   688→                calleeName = doc.getString("calleeName") ?: "",
   689→                callerAvatar = doc.getString("callerAvatar"),
   690→                calleeAvatar = doc.getString("calleeAvatar"),
   691→                type = if (endReason == CallEndReason.MISSED) CallDirection.MISSED else direction,
   692→                isVideoCall = doc.getBoolean("isVideoCall") ?: false,
   693→                callType = try { com.zixo.app.domain.model.CallTechnology.valueOf(callTechnologyStr) } catch (_: Exception) { com.zixo.app.domain.model.CallTechnology.WEBRTC_AUDIO },
   694→                isGroupCall = doc.getBoolean("isGroupCall") ?: false,
   695→                duration = doc.getLong("duration") ?: 0L,
   696→                timestamp = doc.getLong("timestamp") ?: 0L,
   697→                endReason = endReason,
   698→                threadId = doc.getString("threadId") ?: "",
   699→                isRead = doc.getBoolean("isRead") ?: false
   700→            )
   701→        } catch (e: Exception) {
   702→            Timber.e(e, "Failed to map call log entry")
   703→            null
   704→        }
   705→    }

   764→    override fun clearCallHistory(): Flow<Result<Unit>> = flow {
   765→        try {
   766→            val myUid = currentUid ?: throw IllegalStateException("Not authenticated")
   767→
   768→            val snapshot = callLogCollection
   769→                .whereArrayContains("participantUids", myUid)
   770→                .get()
   771→                .await()
   772→
   773→            val batch = firestore.batch()
   774→            for (doc in snapshot.documents) {
   775→                batch.delete(doc.reference)
   776→            }
   777→            batch.commit().await()
   778→
   779→            Timber.d("Call history cleared: %d entries removed", snapshot.size())
   780→            emit(Result.success(Unit))
   781→        } catch (e: Exception) {
   782→            Timber.e(e, "Failed to clear call history")
   783→            emit(Result.failure(e))
   784→        }
   785→    }.flowOn(Dispatchers.IO)
