   456→// ════════════════════════════════════════════════════════════════
   457→// QR Code Thumbnail (small version for the Zixo number card)
   458→// ════════════════════════════════════════════════════════════════
   459→
   460→@Composable
   461→private fun QrCodeThumbnail(
   462→    data: String,
   463→    size: androidx.compose.ui.unit.Dp,
   464→    modifier: Modifier = Modifier
   465→) {
   466→    if (data.isBlank()) return
   467→
   468→    val density = LocalDensity.current
   469→    val sizePx = with(density) { size.roundToPx() }
   470→
   471→    val qrBitmap = remember(data, sizePx) {
   472→        generateQrBitmapThumbnail(data, sizePx)
   473→    }
   474→
   475→    Box(
   476→        modifier = modifier
   477→            .size(size)
   478→            .clip(RoundedCornerShape(8.dp))
   479→            .background(Color.White)
   480→            .padding(4.dp),
   481→        contentAlignment = Alignment.Center
   482→    ) {
   483→        androidx.compose.foundation.Image(
   484→            bitmap = qrBitmap.asImageBitmap(),
   485→            contentDescription = "QR Code",
   486→            modifier = Modifier
   487→                .size(size - 8.dp)
   488→                .clip(RoundedCornerShape(4.dp)),
   489→            contentScale = ContentScale.Fit
   490→        )
   491→    }
   492→}
   493→
   494→private fun generateQrBitmapThumbnail(data: String, sizePx: Int): Bitmap {
   495→    val writer = QRCodeWriter()
   496→    val bitMatrix = writer.encode(data, BarcodeFormat.QR_CODE, sizePx, sizePx)
   497→    val width = bitMatrix.width
   498→    val height = bitMatrix.height
   499→    val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.RGB_565)
   500→    for (x in 0 until width) {
   501→        for (y in 0 until height) {
   502→            bitmap.setPixel(
   503→                x, y,
   504→                if (bitMatrix[x, y]) AndroidColor.BLACK else AndroidColor.WHITE
   505→            )
   506→        }
   507→    }
   508→    return bitmap
   509→}

   728→private fun formatZixoNumber(number: String): String {
   729→    val digits = number.filter { it.isDigit() }
   730→    return if (digits.length >= 8) {
   731→        "${digits.substring(0, 4)} ${digits.substring(4, 8)}"
   732→    } else {
   733→        digits
   734→    }
