   413→private fun ProfileHeaderBlock(
   414→    displayName: String,
   415→    bio: String,
   416→    avatarUrl: String,
   417→    onQrClick: () -> Unit,
   418→    onProfileClick: () -> Unit
   419→) {
   420→    Box(
   421→        modifier = Modifier
   422→            .fillMaxWidth()
   423→            .diagonalMeshGradient()
   424→            .clickable(
   425→                interactionSource = remember { MutableInteractionSource() },
   426→                indication = null,
   427→                onClick = onProfileClick
   428→            )
   429→            .padding(horizontal = 20.dp, vertical = 24.dp)
   430→    ) {
   431→        Row(
   432→            verticalAlignment = Alignment.CenterVertically
   433→        ) {
   434→            // Avatar
   435→            Box(
   436→                modifier = Modifier
   437→                    .size(80.dp)
   438→                    .clip(CircleShape)
   439→                    .background(Color(0xFF1A2A32))
   440→                    .border(2.dp, Color.White.copy(alpha = 0.3f), CircleShape),
   441→                contentAlignment = Alignment.Center
   442→            ) {
   443→                if (avatarUrl.isNotBlank()) {
   444→                    AsyncImage(
   445→                        model = ImageRequest.Builder(LocalContext.current)
   446→                            .data(avatarUrl)

   567→ * - Real bitmap rendered from ZXing (not a placeholder pattern)

   586→            .fillMaxSize()
   587→            .background(Color.Black.copy(alpha = 0.6f))
   588→            .clickable(
   589→                interactionSource = remember { MutableInteractionSource() },
