    89→// ── Screen Entry Point ────────────────────────────────────────────────────────

    91→/**
    92→ * Full chat message screen with keyboard avoidance and Liquid Glass design.
    93→ *
    94→ * Features:
    95→ * - Wrapped in [Scaffold] with `Modifier.imePadding()` for keyboard avoidance
    96→ * - [LazyColumn] with `reverseLayout = true` for messages — newest at bottom
    97→ * - Auto-scroll on new messages via [LaunchedEffect]
    98→ * - 74dp liquid glass pill-shaped input tray
    99→ * - Outgoing messages: glowing neon mint transparency background
   100→ * - Incoming messages: translucent dark petrol grey curves
   101→ * - Long-press action menu with 3D reactions, reply, forward, delete
   102→ * - Reply preview bar above input tray
   103→ * - Call state overlay when active
   104→ * - Top bar with contact info and call icons
   105→ * - Media messages with Coil [AsyncImage] thumbnails
   106→ *
   107→ * @param threadId The ID of the chat thread to display.
   108→ * @param onBack Callback for the back navigation button.
   109→ */
   110→@OptIn(ExperimentalFoundationApi::class)
   111→@Composable
   112→fun ChatMessageScreen(
   113→    threadId: String,
   114→    onBack: () -> Unit = {},
   115→    viewModel: ChatViewModel = hiltViewModel()
   116→) {
   117→    val messages by viewModel.messages.collectAsState()
   118→    val thread by viewModel.thread.collectAsState()
   119→    val inputText by viewModel.inputText.collectAsState()
   120→    val isSending by viewModel.isSending.collectAsState()
   121→    val callState by viewModel.callState.collectAsState()
   122→    val replyToMessage by viewModel.replyToMessage.collectAsState()
   123→    val showActionMenu by viewModel.showActionMenu.collectAsState()
   124→    val currentUserId by viewModel.currentUserId.collectAsState()
   125→    val errorMessage by viewModel.errorMessage.collectAsState()
   126→
   127→    val snackbarHostState = remember { SnackbarHostState() }
   128→    val listState = rememberLazyListState()
   129→
   130→    // Load thread data on first composition
   131→    LaunchedEffect(threadId) {
   132→        viewModel.loadThread(threadId)
   133→    }
   134→
   135→    // Auto-scroll to bottom on new messages
   136→    LaunchedEffect(messages.size) {
   137→        if (messages.isNotEmpty()) {
   138→            listState.animateScrollToItem(0)
   139→        }
   140→    }
   141→
   142→    // Display error messages as snackbars
   143→    LaunchedEffect(errorMessage) {
   144→        errorMessage?.let {
   145→            snackbarHostState.showSnackbar(it)
   146→            viewModel.clearError()
   147→        }
   148→    }
   149→
   150→    // Determine contact info from thread
   151→    val otherParticipant = thread?.participantProfiles?.values?.firstOrNull {
   152→        it.uid != currentUserId
   153→    }
   154→    val contactName = otherParticipant?.displayName ?: "Chat"
   155→    val contactAvatar = otherParticipant?.avatarUrl ?: ""
   156→    val isOnline = otherParticipant?.isOnline ?: false
   157→
   158→    Scaffold(
   159→        snackbarHost = { SnackbarHost(snackbarHostState) },
   160→        containerColor = Color.Transparent,
   161→        modifier = Modifier.fillMaxSize()
   162→    ) { innerPadding ->
   163→        Box(modifier = Modifier.fillMaxSize()) {
   164→            // Animated glass background
   165→            ZixoGlassBackground()
   166→
   167→            Column(
   168→                modifier = Modifier
   169→                    .fillMaxSize()
   170→                    .imePadding()
   171→                    .padding(innerPadding)
   172→            ) {
   173→                // ── Top Bar ────────────────────────────────────────────────
   174→                ChatTopBar(
   175→                    contactName = contactName,
   176→                    isOnline = isOnline,
   177→                    onBackClick = onBack,
   178→                    onAudioCallClick = { viewModel.startAudioCall() },
   179→                    onVideoCallClick = { viewModel.startVideoCall() }
   180→                )
   181→
   182→                // ── Message List ───────────────────────────────────────────
   183→                LazyColumn(
   184→                    state = listState,
   185→                    reverseLayout = true,
   186→                    modifier = Modifier
   187→                        .weight(1f)
   188→                        .fillMaxWidth()
   189→                        .padding(horizontal = 12.dp),
   190→                    verticalArrangement = Arrangement.spacedBy(6.dp)
   191→                ) {
   192→                    items(
   193→                        count = messages.size,
   194→                        key = { index -> messages[index].id }
   195→                    ) { index ->
   196→                        val message = messages[index]
   197→                        val isOwnMessage = message.senderUid == currentUserId
   198→
   199→                        MessageBubble(
   200→                            message = message,
   201→                            isOwnMessage = isOwnMessage,
   202→                            onLongClick = { viewModel.showActionMenu.value = message }
   203→                        )
   204→                    }
   205→                }
   206→
   207→                // ── Reply Preview Bar ──────────────────────────────────────
   208→                AnimatedVisibility(
   209→                    visible = replyToMessage != null,
   210→                    enter = slideInVertically() + fadeIn(),
   211→                    exit = slideOutVertically() + fadeOut()
   212→                ) {
   213→                    replyToMessage?.let { replied ->
   214→                        ReplyPreviewBar(
   215→                            senderName = replied.replyToSenderName ?: replied.senderDisplayName,
   216→                            previewText = replied.replyToPreview ?: replied.content,
   217→                            onDismiss = { viewModel.clearReply() }
   218→                        )
   219→                    }
   220→                }
   221→
   222→                // ── Input Tray (74dp Liquid Glass Pill) ────────────────────
   223→                InputTray(
   224→                    value = inputText,
   225→                    onValueChange = { viewModel.inputText.value = it },
   226→                    onSendClick = { viewModel.sendMessage() },
   227→                    isSending = isSending,
   228→                    modifier = Modifier.fillMaxWidth()
   229→                )
   230→            }
   231→
   232→            // ── Action Menu Overlay ────────────────────────────────────────
   233→            showActionMenu?.let { actionMessage ->
   234→                ActionMenuOverlay(
   235→                    message = actionMessage,
   236→                    isOwnMessage = actionMessage.senderUid == currentUserId,
   237→                    onReact = { emoji, isThreeD ->
   238→                        viewModel.addReaction(actionMessage.id, emoji, isThreeD)
   239→                    },
   240→                    onReply = {
   241→                        viewModel.replyToMessage.value = actionMessage
   242→                        viewModel.dismissActionMenu()
   243→                    },
   244→                    onForward = {
   245→                        viewModel.forwardMessage(actionMessage.id, emptyList())
   246→                    },
   247→                    onDeleteForMe = {
   248→                        viewModel.deleteForMe(actionMessage.id)
   249→                    },
   250→                    onDeleteForEveryone = {
   251→                        viewModel.deleteForEveryone(actionMessage.id)
   252→                    },
   253→                    onDismiss = { viewModel.dismissActionMenu() }
   254→                )
   255→            }
   256→
   257→            // ── Call State Overlay ─────────────────────────────────────────
   258→            if (callState !is CallState.IDLE) {
   259→                CallStateOverlay(
   260→                    callState = callState,
   261→                    contactName = contactName,
   262→                    contactAvatar = contactAvatar
   263→                )
   264→            }
   265→        }
   266→    }
   267→}

   331→@OptIn(ExperimentalFoundationApi::class)
   332→@Composable
   333→private fun MessageBubble(
   334→    message: MessageModel,
   335→    isOwnMessage: Boolean,
   336→    onLongClick: () -> Unit
   337→) {
   338→    // Deleted placeholder
   339→    if (message.isDeletedForEveryone) {
   340→        Row(
   341→            modifier = Modifier
   342→                .fillMaxWidth()
   343→                .padding(
   344→                    start = if (isOwnMessage) 48.dp else 0.dp,
   345→                    end = if (!isOwnMessage) 48.dp else 0.dp
   346→                ),
   347→            horizontalArrangement = if (isOwnMessage) Arrangement.End else Arrangement.Start
   348→        ) {
   349→            Box(
   350→                modifier = Modifier
   351→                    .clip(RoundedCornerShape(BubbleCornerSize))
   352→                    .background(IncomingBubbleColor)
   353→                    .padding(horizontal = 14.dp, vertical = 10.dp)
   354→            ) {
   355→                Text(
   356→                    text = "This message was deleted",
   357→                    color = TextSecondary,
   358→                    fontSize = 14.sp,
   359→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   360→                )
   361→            }
   362→        }
   363→        return
   364→    }
   365→
   366→    // Soft-deleted for me — invisible
   367→    if (message.isDeletedForMe) return
   368→
   369→    val bubbleColor = if (isOwnMessage) NeonMintAlpha15 else IncomingBubbleColor
   370→    val alignment = if (isOwnMessage) Arrangement.End else Arrangement.Start
   371→    val shape = RoundedCornerShape(
   372→        topStart = BubbleCornerSize,
   373→        topEnd = BubbleCornerSize,
   374→        bottomStart = if (isOwnMessage) BubbleCornerSize else 4.dp,
   375→        bottomEnd = if (isOwnMessage) 4.dp else BubbleCornerSize
   376→    )
   377→
   378→    Row(
   379→        modifier = Modifier
   380→            .fillMaxWidth()
   381→            .padding(
   382→                start = if (isOwnMessage) 48.dp else 0.dp,
   383→                end = if (!isOwnMessage) 48.dp else 0.dp
   384→            ),
   385→        horizontalArrangement = alignment
   386→    ) {
   387→        Column(
   388→            modifier = Modifier
   389→                .clip(shape)
   390→                .background(bubbleColor)
   391→                .combinedClickable(
   392→                    interactionSource = remember { MutableInteractionSource() },
   393→                    indication = null,
   394→                    onClick = {},
   395→                    onLongClick = onLongClick
   396→                )
   397→                .padding(horizontal = 14.dp, vertical = 10.dp)
   398→        ) {
   399→            // Reply reference preview
   400→            if (message.replyToPreview != null) {
   401→                Row(
   402→                    modifier = Modifier
   403→                        .fillMaxWidth()
   404→                        .clip(RoundedCornerShape(8.dp))
   405→                        .background(NeonMintAlpha15)
   406→                        .padding(horizontal = 8.dp, vertical = 4.dp)
   407→                ) {
   408→                    Text(
   409→                        text = message.replyToSenderName ?: "",
   410→                        color = NeonMint,
   411→                        fontSize = 12.sp,
   412→                        fontWeight = FontWeight.SemiBold,
   413→                        maxLines = 1
   414→                    )
   415→                    Spacer(modifier = Modifier.width(6.dp))
   416→                    Text(
   417→                        text = message.replyToPreview ?: "",
   418→                        color = TextSecondary,
   419→                        fontSize = 12.sp,
   420→                        maxLines = 1,
   421→                        overflow = TextOverflow.Ellipsis,
   422→                        modifier = Modifier.weight(1f)
   423→                    )
   424→                }
   425→                Spacer(modifier = Modifier.height(4.dp))
   426→            }
   427→
   428→            // Forwarded label
   429→            if (message.isForwarded) {
   430→                Text(
   431→                    text = "Forwarded",
   432→                    color = TextSecondary,
   433→                    fontSize = 11.sp,
   434→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   435→                )
   436→                Spacer(modifier = Modifier.height(2.dp))
   437→            }
   438→
   439→            // Media thumbnail
   440→            if (message.type == MessageContentType.IMAGE || message.type == MessageContentType.VIDEO) {
   441→                val imageUrl = message.mediaThumbnailUrl ?: message.mediaUrl
   442→                if (!imageUrl.isNullOrBlank()) {
   443→                    AsyncImage(
   444→                        model = imageUrl,
   445→                        contentDescription = "Media",
   446→                        modifier = Modifier
   447→                            .fillMaxWidth()
   448→                            .height(180.dp)
   449→                            .clip(RoundedCornerShape(10.dp)),
   450→                        contentScale = ContentScale.Crop
   451→                    )
   452→                    Spacer(modifier = Modifier.height(4.dp))
   453→                }
   454→                // Play icon overlay for video
   455→                if (message.type == MessageContentType.VIDEO) {
   456→                    Box(
   457→                        modifier = Modifier
   458→                            .size(36.dp)
   459→                            .clip(CircleShape)
   460→                            .background(DarkPetrolCharcoal.copy(alpha = 0.7f)),
   461→                        contentAlignment = Alignment.Center
   462→                    ) {
   463→                        Icon(
   464→                            imageVector = Icons.Filled.PlayArrow,
   465→                            contentDescription = "Play",
   466→                            tint = TextPrimary,
   467→                            modifier = Modifier.size(20.dp)
   468→                        )
   469→                    }
   470→                }
   471→            }
   472→
   473→            // Voice message indicator
   474→            if (message.type == MessageContentType.AUDIO_VOICE) {
   475→                Row(verticalAlignment = Alignment.CenterVertically) {
   476→                    Icon(
   477→                        imageVector = Icons.Filled.PlayArrow,
   478→                        contentDescription = "Voice",
   479→                        tint = NeonMint,
   480→                        modifier = Modifier.size(20.dp)
   481→                    )
   482→                    Spacer(modifier = Modifier.width(6.dp))
   483→                    Text(
   484→                        text = "Voice message",
   485→                        color = TextSecondary,
   486→                        fontSize = 14.sp
   487→                    )
   488→                }
   489→                Spacer(modifier = Modifier.height(4.dp))
   490→            }
   491→
   492→            // Text content
   493→            if (message.content.isNotBlank()) {
   494→                Text(
   495→                    text = message.content,
   496→                    color = TextPrimary,
   497→                    fontSize = 15.sp,
   498→                    lineHeight = 20.sp
   499→                )
   500→            }
   501→
   502→            // Media caption
   503→            if (!message.caption.isNullOrBlank() && message.type != MessageContentType.TEXT) {
   504→                Spacer(modifier = Modifier.height(4.dp))
   505→                Text(
   506→                    text = message.caption,
   507→                    color = TextPrimary,
   508→                    fontSize = 14.sp
   509→                )
   510→            }
   511→
   512→            // Timestamp and read receipt row
   513→            Row(
   514→                modifier = Modifier.fillMaxWidth(),
   515→                horizontalArrangement = Arrangement.End,
   516→                verticalAlignment = Alignment.CenterVertically
   517→            ) {
   518→                Text(
   519→                    text = formatTimestamp(message.timestamp),
   520→                    color = TextSecondary,
   521→                    fontSize = 11.sp
   522→                )
   523→                if (isOwnMessage) {
   524→                    Spacer(modifier = Modifier.width(4.dp))
   525→                    val tickColor = if (message.isRead) NeonMint else TextSecondary
   526→                    Text(text = if (message.isRead) "✓✓" else "✓", color = tickColor, fontSize = 11.sp)
   527→                }
   528→            }
   529→
   530→            // Reactions row
   531→            if (message.reactions.isNotEmpty()) {
   532→                Spacer(modifier = Modifier.height(4.dp))
   533→                Row(horizontalArrangement = Arrangement.spacedBy(2.dp)) {
   534→                    message.reactions.distinctBy { it.emoji }.forEach { reaction ->
   535→                        Box(
   536→                            modifier = Modifier
   537→                                .clip(RoundedCornerShape(12.dp))
   538→                                .background(GlassBorderColor)
   539→                                .padding(horizontal = 6.dp, vertical = 2.dp)
   540→                        ) {
   541→                            Text(
   542→                                text = "${reaction.emoji} ${message.reactions.count { it.emoji == reaction.emoji }}",
   543→                                fontSize = 12.sp
   544→                            )
   545→                        }
   546→                    }
   547→                }
   548→            }
   549→        }
   550→    }
   551→}

   603→// ── Input Tray ────────────────────────────────────────────────────────────────

   852→/** Formats epoch milliseconds to a readable time string. */
   853→private fun formatTimestamp(epochMs: Long): String {
   854→    if (epochMs == 0L) return ""
   855→    val sdf = java.text.SimpleDateFormat("HH:mm", java.util.Locale.getDefault())
   856→    return sdf.format(java.util.Date(epochMs))
   857→}
