   146→@OptIn(ExperimentalMaterial3Api::class)
   147→@Composable
   148→fun StatusTabScreen(
   149→    navController: NavController,
   150→    viewModel: StatusViewModel = hiltViewModel()
   151→) {
   152→    val contactStatuses by viewModel.contactStatuses.collectAsStateWithLifecycle()
   153→    val myStatuses by viewModel.myStatuses.collectAsStateWithLifecycle()
   154→    val mutualContacts by viewModel.mutualContacts.collectAsStateWithLifecycle()
   155→    val isUploading by viewModel.isUploading.collectAsStateWithLifecycle()
   156→    val uploadProgress by viewModel.uploadProgress.collectAsStateWithLifecycle()
   157→    val errorMessage by viewModel.errorMessage.collectAsStateWithLifecycle()
   158→    val viewingGroup by viewModel.viewingStatusGroup.collectAsStateWithLifecycle()
   159→    val viewingIndex by viewModel.currentViewingIndex.collectAsStateWithLifecycle()
   160→
   161→    var showAddSheet by remember { mutableStateOf(false) }
   162→    var showTextComposer by remember { mutableStateOf(false) }
   163→    var showMediaComposer by remember { mutableStateOf(false) }
   164→    var isVideoMode by remember { mutableStateOf(false) }
   165→    var replyText by remember { mutableStateOf("") }
   166→
   167→    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
   168→
   169→    // Clear error on consumption
   170→    LaunchedEffect(errorMessage) {
   171→        if (errorMessage != null) {
   172→            delay(3_000L)
   173→            viewModel.clearError()
   174→        }
   175→    }
   176→
   177→    Box(modifier = Modifier.fillMaxSize()) {
   178→        ZixoGlassBackground()
   179→
   180→        Column(modifier = Modifier.fillMaxSize()) {
   181→            // ── Header ─────────────────────────────
   182→            Text(
   183→                text = "Status",
   184→                color = TextPrimary,
   185→                fontSize = 24.sp,
   186→                fontWeight = FontWeight.Bold,
   187→                modifier = Modifier.padding(start = 20.dp, top = 16.dp, end = 20.dp, bottom = 8.dp)
   188→            )
   189→
   190→            // ── My Status Card ─────────────────────
   191→            MyStatusCard(
   192→                myStatuses = myStatuses.myStatuses,
   193→                isUploading = isUploading,
   194→                uploadProgress = uploadProgress,
   195→                onAddClick = { showAddSheet = true }
   196→            )
   197→
   198→            // ── My Status Thumbnails (horizontal scroll) ──
   199→            if (myStatuses.myStatuses.isNotEmpty()) {
   200→                MyStatusThumbnailsRow(
   201→                    statuses = myStatuses.myStatuses,
   202→                    onDelete = { statusId -> viewModel.deleteStatus(statusId) }
   203→                )
   204→            }
   205→
   206→            Spacer(modifier = Modifier.height(12.dp))
   207→
   208→            // ── Recent Updates Header ──────────────
   209→            Text(
   210→                text = "Recent Updates",
   211→                color = TextSecondary,
   212→                fontSize = 14.sp,
   213→                fontWeight = FontWeight.SemiBold,
   214→                modifier = Modifier.padding(horizontal = 20.dp, vertical = 8.dp)
   215→            )
   216→
   217→            // ── Status Feed ────────────────────────
   218→            // ONLY statuses from mutual contacts are shown —
   219→            // contact-gated at the repository boundary
   220→            if (contactStatuses.isEmpty()) {
   221→                EmptyStatusFeed()
   222→            } else {
   223→                LazyColumn(
   224→                    modifier = Modifier.fillMaxSize(),
   225→                    verticalArrangement = Arrangement.spacedBy(4.dp),
   226→                    contentPadding = PaddingValues(bottom = 80.dp)
   227→                ) {
   228→                    items(
   229→                        items = contactStatuses,
   230→                        key = { it.senderUid }
   231→                    ) { group ->
   232→                        StatusGroupItem(
   233→                            group = group,
   234→                            onClick = { viewModel.startViewing(group) }
   235→                        )
   236→                    }
   237→                }
   238→            }
   239→        }
   240→
   241→        // ── Error Snack ──────────────────────────
   242→        AnimatedVisibility(
   243→            visible = errorMessage != null,
   244→            enter = fadeIn() + slideInVertically(initialOffsetY = { -it }),
   245→            exit = fadeOut() + slideOutVertically(targetOffsetY = { -it }),
   246→            modifier = Modifier
   247→                .align(Alignment.TopCenter)
   248→                .padding(16.dp)
   249→        ) {
   250→            errorMessage?.let { msg ->
   251→                Box(
   252→                    modifier = Modifier
   253→                        .fillMaxWidth()
   254→                        .liquidGlassCard()
   255→                        .padding(12.dp)
   256→                ) {
   257→                    Text(text = msg, color = DestructiveText, fontSize = 14.sp)
   258→                }
   259→            }
   260→        }
   261→    }
   262→
   263→    // ── Add Status Bottom Sheet ─────────────────
   264→    if (showAddSheet) {
   265→        AddStatusBottomSheet(
   266→            sheetState = sheetState,
   267→            onDismiss = { showAddSheet = false },
   268→            onTextClick = {
   269→                showAddSheet = false
   270→                showTextComposer = true
   271→            },
   272→            onPhotoClick = {
   273→                showAddSheet = false
   274→                isVideoMode = false
   275→                showMediaComposer = true
   276→            },
   277→            onVideoClick = {
   278→                showAddSheet = false
   279→                isVideoMode = true
   280→                showMediaComposer = true
   281→            }
   282→        )
   283→    }
   284→
   285→    // ── Text Status Composer ────────────────────
   286→    if (showTextComposer) {
   287→        TextStatusComposer(
   288→            isUploading = isUploading,
   289→            onDismiss = { showTextComposer = false },
   290→            onPost = { text, bgColor ->
   291→                viewModel.postTextStatus(text, bgColor)
   292→                showTextComposer = false
   293→            }
   294→        )
   295→    }
   296→
   297→    // ── Media Status Composer ───────────────────
   298→    if (showMediaComposer) {
   299→        MediaStatusComposer(
   300→            isVideoMode = isVideoMode,
   301→            isUploading = isUploading,
   302→            uploadProgress = uploadProgress,
   303→            onDismiss = { showMediaComposer = false },
   304→            onPost = { filePath, caption ->
   305→                val type = if (isVideoMode) StatusContentType.VIDEO else StatusContentType.IMAGE
   306→                viewModel.postMediaStatus(filePath, caption, type)
   307→                showMediaComposer = false
   308→            }
   309→        )
   310→    }
   311→
   312→    // ── Full-Screen Status Viewer ───────────────
   313→    if (viewingGroup != null) {
   314→        StatusViewerOverlay(
   315→            group = viewingGroup!!,
   316→            currentIndex = viewingIndex,
   317→            replyText = replyText,
   318→            onReplyTextChange = { replyText = it },
   319→            onPrevious = viewModel::previousStatus,
   320→            onNext = viewModel::nextStatus,
   321→            onIndexChange = viewModel::setViewingIndex,
   322→            onClose = viewModel::stopViewing,
   323→            onReaction = { statusId, emoji -> viewModel.reactToStatus(statusId, emoji) }
   324→        )
   325→    }
   326→}

   436→@Composable
   437→private fun MyStatusThumbnailsRow(
   438→    statuses: List<StatusModel>,
   439→    onDelete: (String) -> Unit
   440→) {
   441→    LazyRow(
   442→        modifier = Modifier.fillMaxWidth(),
   443→        contentPadding = PaddingValues(horizontal = 16.dp),
   444→        horizontalArrangement = Arrangement.spacedBy(10.dp)
   445→    ) {
   446→        items(items = statuses, key = { it.id }) { status ->
   447→            Box(
   448→                modifier = Modifier
   449→                    .size(60.dp)
   450→                    .clip(RoundedCornerShape(12.dp))
   451→                    .liquidGlassCard()
   452→                    .clickable { onDelete(status.id) },
   453→                contentAlignment = Alignment.Center
   454→            ) {
   455→                when (status.type) {
   456→                    StatusContentType.TEXT -> {
   457→                        val bgColor = status.backgroundColor?.let { parseHexColor(it) }
   458→                            ?: EmeraldGreen
   459→                        Box(
   460→                            modifier = Modifier
   461→                                .fillMaxSize()
   462→                                .background(bgColor),
   463→                            contentAlignment = Alignment.Center
   464→                        ) {
   465→                            Text(
   466→                                text = status.textContent?.take(2) ?: "",
   467→                                color = Color.White,
   468→                                fontSize = 14.sp,
   469→                                fontWeight = FontWeight.Bold
   470→                            )
   471→                        }
   472→                    }
   473→                    StatusContentType.IMAGE -> {
   474→                        if (!status.mediaThumbnailUrl.isNullOrBlank()) {
   475→                            AsyncImage(
   476→                                model = ImageRequest.Builder(LocalContext.current)
   477→                                    .data(status.mediaThumbnailUrl)
   478→                                    .crossfade(true)
   479→                                    .build(),
   480→                                contentDescription = "Status thumbnail",
   481→                                modifier = Modifier.fillMaxSize(),
   482→                                contentScale = ContentScale.Crop
   483→                            )
   484→                        } else {
   485→                            Box(
   486→                                modifier = Modifier
   487→                                    .fillMaxSize()
   488→                                    .background(DarkPetrolCharcoal),
   489→                                contentAlignment = Alignment.Center
   490→                            ) {
   491→                                Icon(
   492→                                    imageVector = Icons.Filled.Image,
   493→                                    contentDescription = null,
   494→                                    tint = TextSecondary,
   495→                                    modifier = Modifier.size(20.dp)
   496→                                )
   497→                            }
   498→                        }
   499→                    }
   500→                    StatusContentType.VIDEO -> {
   501→                        Box(
   502→                            modifier = Modifier
   503→                                .fillMaxSize()
   504→                                .background(DarkPetrolCharcoal),
   505→                            contentAlignment = Alignment.Center
   506→                        ) {
   507→                            Icon(
   508→                                imageVector = Icons.Filled.Videocam,
   509→                                contentDescription = null,
   510→                                tint = TextSecondary,
   511→                                modifier = Modifier.size(20.dp)
   512→                            )
   513→                        }
   514→                    }
   515→                    StatusContentType.SHAPE -> {
   516→                        val bgColor = status.backgroundColor?.let { parseHexColor(it) }
   517→                            ?: EmeraldGreen
   518→                        Box(
   519→                            modifier = Modifier
   520→                                .fillMaxSize()
   521→                                .background(bgColor),
   522→                            contentAlignment = Alignment.Center
   523→                        ) {
   524→                            ShapeIcon(shapeType = status.shapeType, tint = Color.White, size = 20.dp)
   525→                        }
   526→                    }
   527→                    StatusContentType.EMOJI_3D -> {
   528→                        Box(
   529→                            modifier = Modifier
   530→                                .fillMaxSize()
   531→                                .background(DarkPetrolCharcoal),
   532→                            contentAlignment = Alignment.Center
   533→                        ) {
   534→                            Text(
   535→                                text = status.emoji3dCode ?: "✨",
   536→                                fontSize = 24.sp
   537→                            )
   538→                        }
   539→                    }
   540→                }
   541→            }
   542→        }
   543→    }
   544→}

   595→// Status Ring Avatar with Gradient Ring
   596→// ──────────────────────────────────────────────
   597→
   598→/**
   599→ * Avatar with colored ring segments around it.
   600→ *
   601→ * - **Unviewed**: gradient ring (EmeraldGreen → NeonMint)
   602→ * - **Viewed**: solid grey ring (TextTertiary)
   603→ */
   604→@Composable
   605→private fun StatusRingAvatar(
   606→    avatarUrl: String,
   607→    name: String,
   608→    size: Dp,
   609→    segmentCount: Int,
   610→    isUnviewed: Boolean,
   611→    useGradient: Boolean = false
   612→) {
   613→    val strokeWidth = 3.dp
   614→    val gapAngle = 6f
   615→    val totalGap = gapAngle * segmentCount
   616→    val segmentSweep = if (segmentCount > 0) (360f - totalGap) / segmentCount else 0f
   617→
   618→    Box(contentAlignment = Alignment.Center) {
   619→        // Colored ring segments
   620→        if (segmentCount > 0) {
   621→            Canvas(modifier = Modifier.size(size + strokeWidth * 2)) {
   622→                var startAngle = -90f
   623→                for (i in 0 until segmentCount) {
   624→                    val brush = if (isUnviewed && useGradient) {
   625→                        // Each segment gets its own gradient slice
   626→                        UnviewedRingBrush
   627→                    } else if (isUnviewed) {
   628→                        null
   629→                    } else {
   630→                        null
   631→                    }
   632→
   633→                    if (isUnviewed && useGradient) {
   634→                        drawArc(
   635→                            brush = UnviewedRingBrush,
   636→                            startAngle = startAngle,
   637→                            sweepAngle = segmentSweep,
   638→                            useCenter = false,
   639→                            style = Stroke(
   640→                                width = strokeWidth.toPx(),
   641→                                cap = StrokeCap.Round
   642→                            )
   643→                        )
   644→                    } else {
   645→                        val color = if (isUnviewed) NeonMint else ViewedRingColor
   646→                        drawArc(
   647→                            color = color,
   648→                            startAngle = startAngle,
   649→                            sweepAngle = segmentSweep,
   650→                            useCenter = false,
   651→                            style = Stroke(
   652→                                width = strokeWidth.toPx(),
   653→                                cap = StrokeCap.Round
   654→                            )
   655→                        )
   656→                    }
   657→                    startAngle += segmentSweep + gapAngle
   658→                }
   659→            }
   660→        }
   661→
   662→        AvatarComponent(
   663→            imageUrl = avatarUrl,
   664→            name = name,
   665→            size = size
   666→        )
   667→    }
   668→}

  1379→    )

  1690→private fun formatRelativeTime(timestampMs: Long): String {
  1691→    val now = System.currentTimeMillis()
  1692→    val diff = now - timestampMs
  1693→    val seconds = diff / 1000
  1694→    val minutes = seconds / 60
  1695→    val hours = minutes / 60
  1696→    return when {
  1697→        seconds < 60 -> "just now"
  1698→        minutes < 60 -> "${minutes}m ago"
  1699→        hours < 24 -> "${hours}h ago"
  1700→        else -> "${hours / 24}d ago"
  1701→    }
  1702→}
  1703→
  1704→private fun colorToHex(color: Color): String {
  1705→    val alpha = (color.alpha * 255).toInt()
  1706→    val red = (color.red * 255).toInt()
  1707→    val green = (color.green * 255).toInt()
  1708→    val blue = (color.blue * 255).toInt()
  1709→    return "#${String.format("%02X", alpha)}${String.format("%02X", red)}${String.format("%02X", green)}${String.format("%02X", blue)}"
  1710→}
  1711→
  1712→private fun parseHexColor(hex: String): Color {
  1713→    return try {
  1714→        val cleanHex = hex.removePrefix("#")
  1715→        val argb = cleanHex.toLong(16)
  1716→        Color(argb)
  1717→    } catch (_: Exception) {
  1718→        EmeraldGreen
  1719→    }
  1720→}
