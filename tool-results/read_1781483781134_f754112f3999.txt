   656→   656→// ─────────────────────────────────────────────────────────────────────────────
   657→   657→// Simple QR Canvas (Placeholder — generates a visual QR-like pattern)
   658→   658→// ─────────────────────────────────────────────────────────────────────────────
   659→   659→
   660→   660→/**
   661→   661→ * Draws a placeholder QR-code-like pattern for the account identification URI.
   662→   662→ * In production, this would use a real QR code library (e.g., zxing).
   663→   663→ */
   664→   664→@Composable
   665→   665→private fun SimpleQrCanvas(
   666→   666→    data: String,
   667→   667→    color: Color,
   668→   668→    modifier: Modifier = Modifier
   669→   669→) {
   670→   670→    Canvas(modifier = modifier) {
   671→   671→        val gridSize = 21 // Standard QR Version 1
   672→   672→        val cellSize = size.minDimension / gridSize
   673→   673→        val hash = data.hashCode()
   674→   674→
   675→   675→        for (row in 0 until gridSize) {
   676→   676→            for (col in 0 until gridSize) {
   677→   677→                val isFinderPattern = (
   678→   678→                    (row < 7 && col < 7) ||
   679→   679→                    (row < 7 && col > gridSize - 8) ||
   680→   680→                    (row > gridSize - 8 && col < 7)
   681→   681→                )
   682→   682→                val isFinderBorder = isFinderPattern && (
   683→   683→                    row == 0 || row == 6 || col == 0 || col == 6 ||
   684→   684→                    (row < 7 && col < 7 && (row == 0 || row == 6 || col == 0 || col == 6)) ||
   685→   685→                    (row < 7 && col > gridSize - 8 && (row == 0 || row == 6 || col == gridSize - 7 || col == gridSize - 1)) ||
   686→   686→                    (row > gridSize - 8 && col < 7 && (row == gridSize - 7 || row == gridSize - 1 || col == 0 || col == 6))
   687→   687→                )
   688→   688→                val isFinderInner = isFinderPattern && (
   689→   689→                    (row in 2..4 && col in 2..4) ||
   690→   690→                    (row in 2..4 && col in gridSize - 5..gridSize - 3) ||
   691→   691→                    (row in gridSize - 5..gridSize - 3 && col in 2..4)
   692→   692→                )
   693→   693→
   694→   694→                val cellHash = (hash * 31 + row * 17 + col * 7) and 0xFF
   695→   695→                val shouldFill = isFinderBorder || isFinderInner || (!isFinderPattern && cellHash % 3 != 0)
   696→   696→
   697→   697→                if (shouldFill) {
   698→   698→                    drawRect(
   699→   699→                        color = color,
   700→   700→                        topLeft = androidx.compose.ui.geometry.Offset(
   701→   701→                            x = col * cellSize,
   702→   702→                            y = row * cellSize
   703→   703→                        ),
   704→   704→                        size = Size(cellSize, cellSize)
   705→   705→                    )
   706→   706→                }
   707→   707→            }
   708→   708→        }
   709→   709→    }
   710→   710→}
