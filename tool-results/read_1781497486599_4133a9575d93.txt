    43→    43→ *

    84→    84→     *
    85→    85→     * Provides a continuous [com.google.firebase.database.ValueEventListener] on
    86→    86→     * `/calls/{callId}/`, streaming every state change as a Map. All listener
    87→    87→     * operations are safely decoupled from the UI thread via [callbackFlow] on
    88→    88→     * [Dispatchers.IO].
    89→    89→     *
    90→    90→     * This is the primary mechanism by which the calling UI reacts to remote
    91→    91→     * state transitions without polling.
    92→    92→     */
    93→    93→    override fun observeCallSession(callId: String): Flow<Map<String, Any>> = callbackFlow {
    94→    94→        val callRef = realtimeDb.getReference("calls/$callId")
    95→    95→        val listener = object : com.google.firebase.database.ValueEventListener {
    96→    96→            override fun onDataChange(snapshot: com.google.firebase.database.DataSnapshot) {
    97→    97→                try {
    98→    98→                    val data = snapshot.value as? Map<String, Any> ?: emptyMap()
    99→    99→                    trySend(data)
   100→   100→                } catch (e: Exception) {
   101→   101→                    Timber.e(e, "Error parsing call session data for callId: %s", callId)
   102→   102→                }
   103→   103→            }
   104→   104→
   105→   105→            override fun onCancelled(error: com.google.firebase.database.DatabaseError) {
   106→   106→                Timber.e(error.toException(), "Call session listener cancelled for callId: %s", callId)
   107→   107→                close(error.toException())
   108→   108→            }
   109→   109→        }
   110→   110→        callRef.addValueEventListener(listener)
   111→   111→        awaitClose { callRef.removeEventListener(listener) }
   112→   112→    }.flowOn(Dispatchers.IO)

   115→   115→     * Observes ICE candidates for a specific call from Firebase Realtime DB.
   116→   116→     *
   117→   117→     * Listens to `/calls/{callId}/iceCandidates/` with a [com.google.firebase.database.ChildEventListener],
   118→   118→     * streaming each newly added ICE candidate as it arrives from the remote peer.

   123→   123→        val candidatesRef = realtimeDb.getReference("calls/$callId/iceCandidates")
   124→   124→        val childListener = object : com.google.firebase.database.ChildEventListener {
   125→   125→            override fun onChildAdded(
   126→   126→                snapshot: com.google.firebase.database.DataSnapshot,
   127→   127→                previousChildName: String?
   128→   128→            ) {
   129→   129→                try {
   130→   130→                    val sdpMid = snapshot.child("sdpMid").getValue(String::class.java) ?: ""
   131→   131→                    val sdpMLineIndex = snapshot.child("sdpMLineIndex").getValue(Int::class.java) ?: 0
   132→   132→                    val sdp = snapshot.child("sdp").getValue(String::class.java) ?: ""
   133→   133→                    if (sdp.isNotBlank()) {
   134→   134→                        trySend(IceCandidate(sdpMid, sdpMLineIndex, sdp))
   135→   135→                    }
   136→   136→                } catch (e: Exception) {
   137→   137→                    Timber.e(e, "Error parsing ICE candidate for callId: %s", callId)
   138→   138→                }
   139→   139→            }
   140→   140→
   141→   141→            override fun onChildChanged(
   142→   142→                snapshot: com.google.firebase.database.DataSnapshot,
   143→   143→                previousChildName: String?
   144→   144→            ) { /* ICE candidates are append-only */ }
   145→   145→
   146→   146→            override fun onChildRemoved(snapshot: com.google.firebase.database.DataSnapshot) {}
   147→   147→
   148→   148→            override fun onChildMoved(
   149→   149→                snapshot: com.google.firebase.database.DataSnapshot,
   150→   150→                previousChildName: String?
   151→   151→            ) {}
   152→   152→
   153→   153→            override fun onCancelled(error: com.google.firebase.database.DatabaseError) {

   155→   155→                close(error.toException())
   156→   156→            }
   157→   157→        }
   158→   158→        candidatesRef.addChildEventListener(childListener)
   159→   159→        awaitClose { candidatesRef.removeEventListener(childListener) }
   160→   160→    }.flowOn(Dispatchers.IO)

   277→   277→    override fun initiateCall(targetUid: String, isVideoCall: Boolean): Flow<CallState> = flow {
   278→   278→        val myUid = currentUid
   279→   279→        if (myUid == null) {
   280→   280→            emit(CallState.IDLE)
   281→   281→            return@flow
   282→   282→        }
   283→   283→
   284→   284→        try {
   285→   285→            // Zero-trust: verify mutual contact before initiating
   286→   286→            var gateResult: CommunicationGate? = null
   287→   287→            contactRepository.verifyMutualContact(targetUid).collect { gateResult = it }
   288→   288→
   289→   289→            if (gateResult !is CommunicationGate.Allowed) {
   290→   290→                Timber.w("Call blocked — not a mutual contact: %s", targetUid)
   291→   291→                emit(CallState.IDLE)
   292→   292→                return@flow
   293→   293→            }
   294→   294→
   295→   295→            val contact = (gateResult as CommunicationGate.Allowed).contact
   296→   296→
   297→   297→            // Create the call document in Firebase Realtime DB
   298→   298→            val callId = UUID.randomUUID().toString()
   299→   299→
   300→   300→            _callState.value = CallState.DIALING(
   301→   301→                callId = callId,
   302→   302→                targetUid = targetUid,
   303→   303→                targetDisplayName = contact.contactDisplayName,
   304→   304→                targetAvatarUrl = contact.contactAvatarUrl,
   305→   305→                isVideoCall = isVideoCall
   306→   306→            )
   307→   307→            emit(_callState.value)
   308→   308→
   309→   309→            // Start foreground service BEFORE WebRTC initialization
   310→   310→            // This guarantees Android does not terminate media streams
   311→   311→            CallForegroundService.start(
   312→   312→                context = context,
   313→   313→                callType = if (isVideoCall) CallForegroundService.CALL_TYPE_VIDEO else CallForegroundService.CALL_TYPE_AUDIO,
   314→   314→                callerName = contact.contactDisplayName,
   315→   315→                callId = callId
   316→   316→            )
   317→   317→
   318→   318→            // Create the call entry in Firebase Realtime DB with structured data
   319→   319→            signalingClient.createCall(callId, myUid, targetUid, isVideoCall)
   320→   320→
   321→   321→            // Initialize WebRTC peer connection on Dispatchers.IO
   322→   322→            // AudioManager is calibrated inside initializePeerConnection
   323→   323→            webRtcClient.initializePeerConnection(isVideoCall)
   324→   324→
   325→   325→            // Set up ICE candidate forwarding: local ICE → Firebase signaling
   326→   326→            webRtcClient.onIceCandidateGenerated = { candidate ->
   327→   327→                GlobalScope.launch(Dispatchers.IO) {
   328→   328→                    try {
   329→   329→                        signalingClient.sendIceCandidate(
   330→   330→                            callId = callId,
   331→   331→                            senderUid = myUid,
   332→   332→                            targetUid = targetUid,
   333→   333→                            sdpMid = candidate.sdpMid,
   334→   334→                            sdpMLineIndex = candidate.sdpMLineIndex,
   335→   335→                            sdp = candidate.sdp
   336→   336→                        )
   337→   337→                    } catch (e: Exception) {
   338→   338→                        Timber.e(e, "Failed to forward ICE candidate to signaling")
   339→   339→                    }
   340→   340→                }
   341→   341→            }
   342→   342→
   343→   343→            // Create SDP offer
   344→   344→            val sdpOffer = webRtcClient.createOffer()
   345→   345→
   346→   346→            // Send the offer via Firebase signaling
   347→   347→            signalingClient.sendOffer(callId, myUid, targetUid, sdpOffer)
   348→   348→
   349→   349→            // Update call status to ringing
   350→   350→            signalingClient.updateCallStatus(callId, "ringing")
   351→   351→
   352→   352→            // Observe answer and ICE candidates concurrently
   353→   353→            var isAnswerReceived = false
   354→   354→
   355→   355→            // Listen for answer
   356→   356→            signalingClient.observeAnswer(callId, myUid).collect { answerData ->
   357→   357→                if (answerData != null && !isAnswerReceived) {
   358→   358→                    isAnswerReceived = true
   359→   359→                    webRtcClient.setRemoteAnswer(answerData.sdp)
   360→   360→
   361→   361→                    // Update call status to connected
   362→   362→                    signalingClient.updateCallStatus(callId, "connected")
   363→   363→
   364→   364→                    _callState.value = CallState.CONNECTED(
   365→   365→                        callId = callId,
   366→   366→                        targetUid = targetUid,
   367→   367→                        targetDisplayName = contact.contactDisplayName,
   368→   368→                        targetAvatarUrl = contact.contactAvatarUrl,
   369→   369→                        isVideoCall = isVideoCall,
   370→   370→                        connectedAt = System.currentTimeMillis()
   371→   371→                    )
   372→   372→                    connectedAt = System.currentTimeMillis()
   373→   373→                    emit(_callState.value)
   374→   374→
   375→   375→                    // Start duration tracking
   376→   376→                    startDurationTracking(callId)
   377→   377→
   378→   378→                    // Start observing ICE candidates for this call
   379→   379→                    observeIceCandidates(callId, myUid)
   380→   380→                }
   381→   381→            }
   382→   382→        } catch (e: Exception) {
   383→   383→            Timber.e(e, "Failed to initiate call")
   384→   384→            _callState.value = CallState.IDLE
   385→   385→            emit(CallState.IDLE)
   386→   386→        }
   387→   387→    }.flowOn(Dispatchers.IO)

   389→   389→    // ── Observe Incoming Calls ────────────────────────────────────────────────
   390→   390→
   391→   391→    override fun observeIncomingCalls(): Flow<CallState> = callbackFlow {
   392→   392→        val myUid = currentUid
   393→   393→        if (myUid == null) {
   394→   394→            trySend(CallState.IDLE)
   395→   395→            close()
   396→   396→            return@callbackFlow
   397→   397→        }
   398→   398→
   399→   399→        val callsRef = realtimeDb.getReference("calls")
   400→   400→        val listener = callsRef.orderByChild("calleeUid").equalTo(myUid)
   401→   401→            .addChildEventListener(object : com.google.firebase.database.ChildEventListener {
   402→   402→                override fun onChildAdded(
   403→   403→                    snapshot: com.google.firebase.database.DataSnapshot,
   404→   404→                    previousChildName: String?
   405→   405→                ) {
   406→   406→                    try {
   407→   407→                        val status = snapshot.child("status").getValue(String::class.java)
   408→   408→                        if (status == "ringing") {
   409→   409→                            val callId = snapshot.key ?: return
   410→   410→                            val callerUid = snapshot.child("callerUid").getValue(String::class.java) ?: return
   411→   411→                            val callerDisplayName = snapshot.child("callerDisplayName").getValue(String::class.java) ?: ""
   412→   412→                            val isVideoCall = snapshot.child("isVideoCall").getValue(Boolean::class.java) ?: false
   413→   413→
   414→   414→                            val incomingState = CallState.RINGING(
   415→   415→                                callId = callId,
   416→   416→                                callerUid = callerUid,
   417→   417→                                callerDisplayName = callerDisplayName,
   418→   418→                                callerAvatarUrl = "",
   419→   419→                                isVideoCall = isVideoCall
   420→   420→                            )
   421→   421→                            _incomingCalls.value = incomingState
   422→   422→                            trySend(incomingState)
   423→   423→                        }
   424→   424→                    } catch (e: Exception) {
   425→   425→                        Timber.e(e, "Error parsing incoming call")
   426→   426→                    }
   427→   427→                }
   428→   428→
   429→   429→                override fun onChildChanged(
   430→   430→                    snapshot: com.google.firebase.database.DataSnapshot,
   431→   431→                    previousChildName: String?
   432→   432→                ) { /* handled in observeCallState */ }
   433→   433→
   434→   434→                override fun onChildRemoved(snapshot: com.google.firebase.database.DataSnapshot) {}
   435→   435→                override fun onChildMoved(snapshot: com.google.firebase.database.DataSnapshot, previousChildName: String?) {}
   436→   436→                override fun onCancelled(error: com.google.firebase.database.DatabaseError) {
   437→   437→                    Timber.e(error.toException(), "Incoming calls listener cancelled")
   438→   438→                }
   439→   439→            })
   440→   440→
   441→   441→        awaitClose { callsRef.removeEventListener(listener) }

   444→   444→    // ── Answer Call ───────────────────────────────────────────────────────────
   445→   445→
   446→   446→    override fun answerCall(callId: String): Flow<CallState> = flow {
   447→   447→        try {
   448→   448→            val myUid = currentUid ?: throw IllegalStateException("Not authenticated")
   449→   449→
   450→   450→            // Get call data
   451→   451→            val callSnapshot = realtimeDb.getReference("calls").child(callId).get().await()
   452→   452→            val callerUid = callSnapshot.child("callerUid").getValue(String::class.java) ?: ""
   453→   453→            val isVideoCall = callSnapshot.child("isVideoCall").getValue(Boolean::class.java) ?: false
   454→   454→
   455→   455→            // Verify mutual contact
   456→   456→            var gateResult: CommunicationGate? = null
   457→   457→            contactRepository.verifyMutualContact(callerUid).collect { gateResult = it }
   458→   458→
   459→   459→            if (gateResult !is CommunicationGate.Allowed) {
   460→   460→                Timber.w("Cannot answer call — not a mutual contact")
   461→   461→                emit(CallState.IDLE)
   462→   462→                return@flow
   463→   463→            }
   464→   464→
   465→   465→            val contact = (gateResult as CommunicationGate.Allowed).contact
   466→   466→
   467→   467→            // Update call status
   468→   468→            realtimeDb.getReference("calls").child(callId).child("status")
   469→   469→                .setValue("connected").await()
   470→   470→
   471→   471→            // Start foreground service
   472→   472→            CallForegroundService.start(
   473→   473→                context = context,
   474→   474→                callType = if (isVideoCall) CallForegroundService.CALL_TYPE_VIDEO else CallForegroundService.CALL_TYPE_AUDIO,
   475→   475→                callerName = contact.contactDisplayName,
   476→   476→                callId = callId
   477→   477→            )
   478→   478→
   479→   479→            // Initialize WebRTC and observe offer
   480→   480→            webRtcClient.initializePeerConnection(isVideoCall)
   481→   481→
   482→   482→            signalingClient.observeOffer(callId).collect { offer ->
   483→   483→                if (offer != null) {
   484→   484→                    webRtcClient.setRemoteOffer(offer)
   485→   485→                    val sdpAnswer = webRtcClient.createAnswer()
   486→   486→                    signalingClient.sendAnswer(callId, myUid, sdpAnswer)
   487→   487→
   488→   488→                    _callState.value = CallState.CONNECTED(
   489→   489→                        callId = callId,
   490→   490→                        targetUid = callerUid,
   491→   491→                        targetDisplayName = contact.contactDisplayName,
   492→   492→                        targetAvatarUrl = contact.contactAvatarUrl,
   493→   493→                        isVideoCall = isVideoCall,
   494→   494→                        connectedAt = System.currentTimeMillis()
   495→   495→                    )
   496→   496→                    connectedAt = System.currentTimeMillis()
   497→   497→                    emit(_callState.value)
   498→   498→                    startDurationTracking(callId)
   499→   499→                }
   500→   500→            }
   501→   501→        } catch (e: Exception) {
   502→   502→            Timber.e(e, "Failed to answer call")
   503→   503→            _callState.value = CallState.IDLE
   504→   504→            emit(CallState.IDLE)
   505→   505→        }
   506→   506→    }.flowOn(Dispatchers.IO)

   527→   527→    override fun endCall(callId: String): Flow<Result<Unit>> = flow {
   528→   528→        try {
   529→   529→            val currentState = _callState.value
   530→   530→            val durationSeconds = if (currentState is CallState.CONNECTED) {
   531→   531→                (System.currentTimeMillis() - connectedAt) / 1000
   532→   532→            } else 0L
   533→   533→
   534→   534→            // Update call status in Firebase via signaling client
   535→   535→            signalingClient.updateCallStatus(callId, "ended")
   536→   536→
   537→   537→            // Clean up WebRTC resources (releases audio focus too)
   538→   538→            webRtcClient.disconnect()

   540→   540→            // Clean up signaling data from Firebase Realtime DB
   541→   541→            signalingClient.cleanupSignalingData(callId)
   542→   542→
   543→   543→            // Stop foreground service
   544→   544→            CallForegroundService.stop(context)
   545→   545→
   546→   546→            // Cancel duration tracking
   547→   547→            durationJob?.cancel()
   548→   548→            durationJob = null
   549→   549→
   550→   550→            // Log the call
   551→   551→            logCall(callId, currentState, durationSeconds)
   552→   552→
   553→   553→            _callState.value = CallState.ENDED(
   554→   554→                callId = callId,
   555→   555→                durationSeconds = durationSeconds,
   556→   556→                endReason = CallEndReason.COMPLETED
   557→   557→            )

   559→   559→        } catch (e: Exception) {

   562→   562→            try { webRtcClient.disconnect() } catch (_: Exception) {}
   563→   563→            try { CallForegroundService.stop(context) } catch (_: Exception) {}

   565→   565→            _callState.value = CallState.IDLE
   566→   566→            emit(Result.failure(e))
   567→   567→        }
   568→   568→    }.flowOn(Dispatchers.IO)
