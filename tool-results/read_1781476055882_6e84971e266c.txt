   455→// ──────────────────────────────────────────────────────────────────────────────
   456→// QR Code composable using ZXing
   457→// ──────────────────────────────────────────────────────────────────────────────
   458→
   459→@Composable
   460→private fun ZixoQrCode(
   461→    data: String,
   462→    size: androidx.compose.ui.unit.Dp,
   463→    modifier: Modifier = Modifier
   464→) {
   465→    if (data.isBlank()) return
   466→
   467→    val qrBitmap = remember(data) {
   468→        generateQrBitmap(data, size.value.toInt())
   469→    }
   470→
   471→    Card(
   472→        modifier = modifier.padding(2.dp),
   473→        shape = RoundedCornerShape(8.dp),
   474→        colors = CardDefaults.cardColors(containerColor = Color.White),
   475→        elevation = CardDefaults.cardElevation(defaultElevation = 0.dp),
   476→    ) {
   477→        androidx.compose.foundation.Image(
   478→            bitmap = qrBitmap.asImageBitmap(),
   479→            contentDescription = "QR Code for Zixo number",
   480→            modifier = Modifier
   481→                .size(size)
   482→                .padding(8.dp),
   483→            contentScale = ContentScale.Fit
   484→        )
   485→    }
   486→}
   487→
   488→/**
   489→ * Generates a QR code bitmap using ZXing core library.
   490→ */
   491→private fun generateQrBitmap(data: String, sizePx: Int): Bitmap {
   492→    val writer = QRCodeWriter()
   493→    val bitMatrix = writer.encode(
   494→        data,
   495→        BarcodeFormat.QR_CODE,
   496→        sizePx,
   497→        sizePx
   498→    )
   499→    val width = bitMatrix.width
   500→    val height = bitMatrix.height
   501→    val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.RGB_565)
   502→    for (x in 0 until width) {
   503→        for (y in 0 until height) {
   504→            bitmap.setPixel(
   505→                x, y,
   506→                if (bitMatrix[x, y]) AndroidColor.BLACK else AndroidColor.WHITE
   507→            )
   508→        }
   509→    }
   510→    return bitmap
   511→}

   783→/**
   784→ * Formats an 8-digit number as two 4-digit blocks separated by a space.
   785→ * e.g., "12345678" -> "1234 5678"
   786→ */
   787→private fun formatZixoNumber(number: String): String {
   788→    val digits = number.filter { it.isDigit() }
   789→    return if (digits.length >= 8) {
   790→        "${digits.substring(0, 4)} ${digits.substring(4, 8)}"
   791→    } else {
   792→        digits
   793→    }
   794→}
