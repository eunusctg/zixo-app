    37→    37→/**
    38→    38→ * Concrete implementation of [CallRepository].
    39→    39→ *
    40→    40→ * Manages WebRTC calls using the LiveKit Android SDK with Firebase
    41→    41→ * Firestore for call signaling and history. All LiveKit operations
    42→    42→ * (token fetch, room connect, track publish) run on [Dispatchers.IO]
    43→    43→ * to prevent Main Thread blocking.
    44→    44→ *
    45→    45→ * Zero-trust enforcement: All call initiation methods verify mutual
    46→    46→ * contact status through [ContactRepository.checkCommunicationGate]
    47→    47→ * before executing. Non-contact calls are blocked at this boundary.
    48→    48→ *
    49→    49→ * Call state transitions:
    50→    50→ * - Outgoing: IDLE → DIALING → CONNECTED → ENDED
    51→    51→ * - Incoming: IDLE → RINGING → CONNECTED → ENDED
    52→    52→ */
    53→    53→@Singleton
    54→    54→class CallRepositoryImpl @Inject constructor(
    55→    55→    private val firestore: FirebaseFirestore,
    56→    56→    private val firebaseAuth: FirebaseAuth,
    57→    57→    private val liveKitService: LiveKitService,
    58→    58→    private val cloudflareApiService: CloudflareApiService,
    59→    59→    private val contactRepository: ContactRepository,
    60→    60→    @ApplicationContext private val applicationContext: Context
    61→    61→) : CallRepository {

    64→    64→        get() = firebaseAuth.currentUser?.uid

    77→    77→
    78→    78→    override fun initiateAudioCall(targetUid: String): Flow<CallState> = flow {
    79→    79→        val myUid = currentUid
    80→    80→        if (myUid == null) {
    81→    81→            emit(CallState.IDLE)
    82→    82→            return@flow
    83→    83→        }
    84→    84→
    85→    85→        try {
    86→    86→            _callState.value = CallState.IDLE
    87→    87→
    88→    88→            val gate = contactRepository.checkCommunicationGate(targetUid)
    89→    89→            if (gate !is CommunicationGate.Allowed) {
    90→    90→                Timber.w("Audio call blocked: not a mutual contact with %s", targetUid)
    91→    91→                _callState.value = CallState.IDLE
    92→    92→                emit(CallState.IDLE)
    93→    93→                return@flow
    94→    94→            }
    95→    95→
    96→    96→            val contact = gate.contact
    97→    97→            val callId = UUID.randomUUID().toString()
    98→    98→            currentCallId = callId
    99→    99→
   100→   100→            val dialingState = CallState.DIALING(
   101→   101→                callId = callId,
   102→   102→                targetUid = targetUid,
   103→   103→                targetDisplayName = contact.contactDisplayName,
   104→   104→                targetAvatarUrl = contact.contactAvatarUrl,
   105→   105→                isVideoCall = false
   106→   106→            )
   107→   107→            _callState.value = dialingState
   108→   108→            emit(dialingState)
   109→   109→
   110→   110→            val myProfile = usersCollection.document(myUid).get().await()
   111→   111→            val now = System.currentTimeMillis()
   112→   112→
   113→   113→            val callData = hashMapOf(
   114→   114→                "id" to callId,
   115→   115→                "callerUid" to myUid,
   116→   116→                "callerDisplayName" to (myProfile.getString("displayName") ?: ""),
   117→   117→                "callerAvatarUrl" to (myProfile.getString("photoUrl") ?: ""),
   118→   118→                "calleeUid" to targetUid,
   119→   119→                "calleeDisplayName" to contact.contactDisplayName,
   120→   120→                "calleeAvatarUrl" to contact.contactAvatarUrl,
   121→   121→                "isVideoCall" to false,
   122→   122→                "status" to "DIALING",
   123→   123→                "createdAt" to now,
   124→   124→                "threadId" to ""
   125→   125→            )
   126→   126→            callsCollection.document(callId).set(callData).await()
   127→   127→
   128→   128→            val roomName = "call_$callId"
   129→   129→            val tokenResponse = cloudflareApiService.generateLiveKitToken(
   130→   130→                identity = myUid,
   131→   131→                roomName = roomName
   132→   132→            ).firstOrNull()
   133→   133→
   134→   134→            if (tokenResponse == null) {
   135→   135→                Timber.e("Failed to fetch LiveKit token for audio call")
   136→   136→                endCallInternal(CallEndReason.NETWORK_ERROR)
   137→   137→                emit(_callState.value)
   138→   138→                return@flow
   139→   139→            }
   140→   140→
   141→   141→            liveKitService.connect(tokenResponse.wsUrl, tokenResponse.token)
   142→   142→                .flowOn(Dispatchers.IO)
   143→   143→                .collect {}
   144→   144→
   145→   145→            liveKitService.startAudioCall(roomName)
   146→   146→                .flowOn(Dispatchers.IO)
   147→   147→                .collect {}
   148→   148→
   149→   149→            callsCollection.document(callId).update("status", "CONNECTED").await()
   150→   150→
   151→   151→            callStartTime = System.currentTimeMillis()
   152→   152→            startDurationTracker()
   153→   153→
   154→   154→            CallForegroundService.start(
   155→   155→                applicationContext,
   156→   156→                CallForegroundService.CALL_TYPE_AUDIO,
   157→   157→                contact.contactDisplayName,
   158→   158→                roomName
   159→   159→            )
   160→   160→
   161→   161→            val connectedState = CallState.CONNECTED(
   162→   162→                callId = callId,
   163→   163→                targetUid = targetUid,
   164→   164→                targetDisplayName = contact.contactDisplayName,
   165→   165→                targetAvatarUrl = contact.contactAvatarUrl,
   166→   166→                isVideoCall = false,
   167→   167→                connectedAt = System.currentTimeMillis()
   168→   168→            )
   169→   169→            _callState.value = connectedState
   170→   170→            emit(connectedState)
   171→   171→        } catch (e: Exception) {
   172→   172→            Timber.e(e, "Failed to initiate audio call")
   173→   173→            endCallInternal(CallEndReason.NETWORK_ERROR)
   174→   174→            emit(_callState.value)
   175→   175→        }
   176→   176→    }.flowOn(Dispatchers.IO)

   489→   489→                participantCount = participantUids.size
   490→   490→            )
   491→   491→            _callState.value = connectedState
   492→   492→            emit(connectedState)
   493→   493→        } catch (e: Exception) {
   494→   494→            Timber.e(e, "Failed to initiate group video call")
   495→   495→            endCallInternal(CallEndReason.NETWORK_ERROR)
   496→   496→            emit(_callState.value)

   500→   500→    // ── Accept Call ───────────────────────────────────────────────────────────
   501→   501→
   502→   502→    override suspend fun acceptCall(callId: String) {
   503→   503→        try {
   504→   504→            val myUid = currentUid ?: return
   505→   505→            withContext(Dispatchers.IO) {
   506→   506→                val callDoc = callsCollection.document(callId).get().await()
   507→   507→                if (!callDoc.exists()) {
   508→   508→                    Timber.w("Call document not found: %s", callId)
   509→   509→                    return@withContext
   510→   510→                }
   511→   511→
   512→   512→                val status = callDoc.getString("status")
   513→   513→                if (status != "DIALING" && status != "RINGING") {
   514→   514→                    Timber.w("Cannot accept call in state: %s", status)
   515→   515→                    return@withContext
   516→   516→                }
   517→   517→
   518→   518→                val roomName = "call_$callId"
   519→   519→                val tokenResponse = cloudflareApiService.generateLiveKitToken(
   520→   520→                    identity = myUid,
   521→   521→                    roomName = roomName
   522→   522→                ).firstOrNull()
   523→   523→
   524→   524→                if (tokenResponse == null) {
   525→   525→                    Timber.e("Failed to fetch LiveKit token for accepting call")
   526→   526→                    return@withContext
   527→   527→                }
   528→   528→
   529→   529→                liveKitService.connect(tokenResponse.wsUrl, tokenResponse.token)
   530→   530→                    .flowOn(Dispatchers.IO)
   531→   531→                    .collect {}
   532→   532→
   533→   533→                val isVideoCall = callDoc.getBoolean("isVideoCall") ?: false
   534→   534→                if (isVideoCall) {
   535→   535→                    liveKitService.startVideoCall(roomName)
   536→   536→                        .flowOn(Dispatchers.IO)
   537→   537→                        .collect {}
   538→   538→                } else {
   539→   539→                    liveKitService.startAudioCall(roomName)
   540→   540→                        .flowOn(Dispatchers.IO)
   541→   541→                        .collect {}
   542→   542→                }
   543→   543→
   544→   544→                callsCollection.document(callId).update("status", "CONNECTED").await()
   545→   545→
   546→   546→                callStartTime = System.currentTimeMillis()
   547→   547→                startDurationTracker()
   548→   548→
   549→   549→                CallForegroundService.start(
   550→   550→                    applicationContext,
   551→   551→                    if (isVideoCall) CallForegroundService.CALL_TYPE_VIDEO
   552→   552→                    else CallForegroundService.CALL_TYPE_AUDIO,
   553→   553→                    callDoc.getString("callerDisplayName") ?: "",
   554→   554→                    roomName
   555→   555→                )
   556→   556→
   557→   557→                val connectedState = CallState.CONNECTED(
   558→   558→                    callId = callId,
   559→   559→                    targetUid = callDoc.getString("callerUid") ?: "",
   560→   560→                    targetDisplayName = callDoc.getString("callerDisplayName") ?: "",
   561→   561→                    targetAvatarUrl = callDoc.getString("callerAvatarUrl") ?: "",
   562→   562→                    isVideoCall = isVideoCall,
   563→   563→                    connectedAt = System.currentTimeMillis()
   564→   564→                )
   565→   565→                _callState.value = connectedState
   566→   566→            }
   567→   567→        } catch (e: Exception) {
   568→   568→            Timber.e(e, "Failed to accept call: %s", callId)
   569→   569→        }
   570→   570→    }

   606→   606→    private suspend fun endCallInternal(reason: CallEndReason) {
   607→   607→        val callId = currentCallId
   608→   608→        val duration = if (callStartTime > 0) {
   609→   609→            (System.currentTimeMillis() - callStartTime) / 1000
   610→   610→        } else 0L
   611→   611→
   612→   612→        try {
   613→   613→            liveKitService.disconnect()
   614→   614→        } catch (e: Exception) {
   615→   615→            Timber.e(e, "Error disconnecting from LiveKit room")
   616→   616→        }
   617→   617→
   618→   618→        CallForegroundService.stop(applicationContext)
   619→   619→
   620→   620→        callDurationJob?.cancel()
   621→   621→        callDurationJob = null
   622→   622→
   623→   623→        if (callId != null) {
   624→   624→            try {
   625→   625→                callsCollection.document(callId).update(
   626→   626→                    mapOf(
   627→   627→                        "status" to "ENDED",
   628→   628→                        "endedAt" to System.currentTimeMillis(),
   629→   629→                        "endReason" to reason.name,
   630→   630→                        "durationSeconds" to duration
   631→   631→                    )
   632→   632→                ).await()
   633→   633→
   634→   634→                val callDoc = callsCollection.document(callId).get().await()
   635→   635→                if (callDoc.exists()) {
   636→   636→                    saveCallHistory(callDoc, duration, reason)
   637→   637→                }
   638→   638→            } catch (e: Exception) {
   639→   639→                Timber.e(e, "Failed to update call document on end")
   640→   640→            }
   641→   641→        }
   642→   642→
   643→   643→        _callState.value = CallState.ENDED(
   644→   644→            callId = callId ?: "",
   645→   645→            durationSeconds = duration,
   646→   646→            endReason = reason
   647→   647→        )
   648→   648→        currentCallId = null
   649→   649→        callStartTime = 0L
   650→   650→    }

   708→   708→    // ── Call History ──────────────────────────────────────────────────────────
   709→   709→
   710→   710→    override fun observeCallHistory(): Flow<List<CallHistoryEntry>> = callbackFlow {
   711→   711→        val myUid = currentUid
   712→   712→        if (myUid == null) {
   713→   713→            trySend(emptyList())
   714→   714→            close()
   715→   715→            return@callbackFlow
   716→   716→        }
   717→   717→
   718→   718→        val subscription = firestore.collection("users").document(myUid)
   719→   719→            .collection("call_history")
   720→   720→            .orderBy("timestamp", Query.Direction.DESCENDING)
   721→   721→            .addSnapshotListener { snapshot, error ->
   722→   722→                if (error != null) {
   723→   723→                    Timber.e(error, "Error observing call history")
   724→   724→                    trySend(emptyList())
   725→   725→                    return@addSnapshotListener
   726→   726→                }
   727→   727→
   728→   728→                val entries = snapshot?.documents?.mapNotNull { doc ->
   729→   729→                    try {
   730→   730→                        mapToCallHistoryEntry(doc)
   731→   731→                    } catch (e: Exception) {
   732→   732→                        Timber.e(e, "Failed to map call history entry: %s", doc.id)
   733→   733→                        null
   734→   734→                    }
   735→   735→                } ?: emptyList()
   736→   736→
   737→   737→                trySend(entries)
   738→   738→            }
   739→   739→
   740→   740→        awaitClose { subscription.remove() }
   741→   741→    }.flowOn(Dispatchers.IO)
