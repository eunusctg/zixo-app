    99→/**
   100→ * Group chat screen — extends ChatMessageScreen with group-specific features.
   101→ *
   102→ * Features:
   103→ * - Group-specific header: group name, avatar, participant count
   104→ * - Tap header to show group info panel (slide-up Liquid Glass sheet)
   105→ * - Group info panel with avatar, name, description, participant list
   106→ * - Admin badges on participant rows
   107→ * - "Add participant" button (admin only, mutual contacts)
   108→ * - "Leave group" option
   109→ * - Participant avatars in the top bar
   110→ * - Group calling: audio/video call buttons start LiveKit Room sessions
   111→ * - Ephemeral timer indicator
   112→ * - Same keyboard avoidance (74dp input tray, reverseLayout, imePadding)
   113→ *
   114→ * @param threadId The ID of the group chat thread to display.
   115→ * @param onBack Callback for the back navigation button.
   116→ */
   117→@OptIn(ExperimentalFoundationApi::class)
   118→@Composable
   119→fun GroupChatScreen(
   120→    threadId: String,
   121→    onBack: () -> Unit = {},
   122→    viewModel: ChatViewModel = hiltViewModel()
   123→) {
   124→    val messages by viewModel.messages.collectAsState()
   125→    val thread by viewModel.thread.collectAsState()
   126→    val inputText by viewModel.inputText.collectAsState()
   127→    val isSending by viewModel.isSending.collectAsState()
   128→    val callState by viewModel.callState.collectAsState()
   129→    val replyToMessage by viewModel.replyToMessage.collectAsState()
   130→    val showActionMenu by viewModel.showActionMenu.collectAsState()
   131→    val currentUserId by viewModel.currentUserId.collectAsState()
   132→    val errorMessage by viewModel.errorMessage.collectAsState()
   133→
   134→    val snackbarHostState = remember { SnackbarHostState() }
   135→    val listState = rememberLazyListState()
   136→    var showGroupInfoPanel by remember { mutableStateOf(false) }
   137→
   138→    // Load thread data on first composition
   139→    LaunchedEffect(threadId) {
   140→        viewModel.loadThread(threadId)
   141→    }
   142→
   143→    // Auto-scroll to bottom on new messages
   144→    LaunchedEffect(messages.size) {
   145→        if (messages.isNotEmpty()) {
   146→            listState.animateScrollToItem(0)
   147→        }
   148→    }
   149→
   150→    // Display error messages as snackbars
   151→    LaunchedEffect(errorMessage) {
   152→        errorMessage?.let {
   153→            snackbarHostState.showSnackbar(it)
   154→            viewModel.clearError()
   155→        }
   156→    }
   157→
   158→    // Group-specific data
   159→    val groupName = thread?.groupName ?: "Group"
   160→    val groupAvatar = thread?.groupAvatarUrl ?: ""
   161→    val participants = thread?.participantProfiles?.values?.toList() ?: emptyList()
   162→    val participantCount = participants.size
   163→    val isAdmin = thread?.groupAdminUids?.contains(currentUserId) == true
   164→    val ephemeralSeconds = thread?.ephemeralTimerSeconds ?: 0
   165→
   166→    Scaffold(
   167→        snackbarHost = { SnackbarHost(snackbarHostState) },
   168→        containerColor = Color.Transparent,
   169→        modifier = Modifier.fillMaxSize()
   170→    ) { innerPadding ->
   171→        Box(modifier = Modifier.fillMaxSize()) {
   172→            ZixoGlassBackground()
   173→
   174→            Column(
   175→                modifier = Modifier
   176→                    .fillMaxSize()
   177→                    .imePadding()
   178→                    .padding(innerPadding)
   179→            ) {
   180→                // ── Group Top Bar ──────────────────────────────────────────
   181→                GroupChatTopBar(
   182→                    groupName = groupName,
   183→                    groupAvatar = groupAvatar,
   184→                    participantCount = participantCount,
   185→                    participants = participants,
   186→                    onBackClick = onBack,
   187→                    onHeaderClick = { showGroupInfoPanel = true },
   188→                    onAudioCallClick = { viewModel.startGroupAudioCall() },
   189→                    onVideoCallClick = { viewModel.startGroupVideoCall() }
   190→                )
   191→
   192→                // ── Ephemeral Timer Indicator ─────────────────────────────
   193→                if (ephemeralSeconds > 0) {
   194→                    Row(
   195→                        modifier = Modifier
   196→                            .fillMaxWidth()
   197→                            .background(NeonMintAlpha15)
   198→                            .padding(horizontal = 16.dp, vertical = 4.dp),
   199→                        verticalAlignment = Alignment.CenterVertically,
   200→                        horizontalArrangement = Arrangement.Center
   201→                    ) {
   202→                        Text(
   203→                            text = "⏱ Messages disappear in ${formatEphemeralTimer(ephemeralSeconds)}",
   204→                            color = NeonMint,
   205→                            fontSize = 12.sp,
   206→                            fontWeight = FontWeight.Medium
   207→                        )
   208→                    }
   209→                }
   210→
   211→                // ── Message List ───────────────────────────────────────────
   212→                LazyColumn(
   213→                    state = listState,
   214→                    reverseLayout = true,
   215→                    modifier = Modifier
   216→                        .weight(1f)
   217→                        .fillMaxWidth()
   218→                        .padding(horizontal = 12.dp),
   219→                    verticalArrangement = Arrangement.spacedBy(6.dp)
   220→                ) {
   221→                    items(
   222→                        count = messages.size,
   223→                        key = { index -> messages[index].id }
   224→                    ) { index ->
   225→                        val message = messages[index]
   226→                        val isOwnMessage = message.senderUid == currentUserId
   227→
   228→                        GroupMessageBubble(
   229→                            message = message,
   230→                            isOwnMessage = isOwnMessage,
   231→                            participants = participants,
   232→                            onLongClick = { viewModel.showActionMenu.value = message }
   233→                        )
   234→                    }
   235→                }
   236→
   237→                // ── Reply Preview Bar ──────────────────────────────────────
   238→                AnimatedVisibility(
   239→                    visible = replyToMessage != null,
   240→                    enter = slideInVertically() + fadeIn(),
   241→                    exit = slideOutVertically() + fadeOut()
   242→                ) {
   243→                    replyToMessage?.let { replied ->
   244→                        ReplyPreviewBar(
   245→                            senderName = replied.replyToSenderName ?: replied.senderDisplayName,
   246→                            previewText = replied.replyToPreview ?: replied.content,
   247→                            onDismiss = { viewModel.clearReply() }
   248→                        )
   249→                    }
   250→                }
   251→
   252→                // ── Input Tray (74dp Liquid Glass Pill) ────────────────────
   253→                GroupInputTray(
   254→                    value = inputText,
   255→                    onValueChange = { viewModel.inputText.value = it },
   256→                    onSendClick = { viewModel.sendMessage() },
   257→                    isSending = isSending,
   258→                    modifier = Modifier.fillMaxWidth()
   259→                )
   260→            }
   261→
   262→            // ── Group Info Panel (Slide-up Sheet) ─────────────────────────
   263→            if (showGroupInfoPanel) {
   264→                GroupInfoPanel(
   265→                    groupName = groupName,
   266→                    groupAvatar = groupAvatar,
   267→                    groupDescription = thread?.groupDescription ?: "",
   268→                    participants = participants,
   269→                    currentUserId = currentUserId,
   270→                    isAdmin = isAdmin,
   271→                    onClose = { showGroupInfoPanel = false }
   272→                )
   273→            }
   274→
   275→            // ── Action Menu Overlay ────────────────────────────────────────
   276→            showActionMenu?.let { actionMessage ->
   277→                GroupActionMenuOverlay(
   278→                    message = actionMessage,
   279→                    isOwnMessage = actionMessage.senderUid == currentUserId,
   280→                    onReact = { emoji, isThreeD ->
   281→                        viewModel.addReaction(actionMessage.id, emoji, isThreeD)
   282→                    },
   283→                    onReply = {
   284→                        viewModel.replyToMessage.value = actionMessage
   285→                        viewModel.dismissActionMenu()
   286→                    },
   287→                    onForward = {
   288→                        viewModel.forwardMessage(actionMessage.id, emptyList())
   289→                    },
   290→                    onDeleteForMe = {
   291→                        viewModel.deleteForMe(actionMessage.id)
   292→                    },
   293→                    onDeleteForEveryone = {
   294→                        viewModel.deleteForEveryone(actionMessage.id)
   295→                    },
   296→                    onDismiss = { viewModel.dismissActionMenu() }
   297→                )
   298→            }
   299→
   300→            // ── Call State Overlay ─────────────────────────────────────────
   301→            if (callState !is CallState.IDLE) {
   302→                GroupCallOverlay(
   303→                    callState = callState,
   304→                    groupName = groupName,
   305→                    groupAvatar = groupAvatar
   306→                )
   307→            }
   308→        }
   309→    }
   310→}

   412→@OptIn(ExperimentalFoundationApi::class)
   413→@Composable
   414→private fun GroupMessageBubble(
   415→    message: MessageModel,
   416→    isOwnMessage: Boolean,
   417→    participants: List<ThreadParticipant>,
   418→    onLongClick: () -> Unit
   419→) {
   420→    if (message.isDeletedForEveryone) {
   421→        Row(
   422→            modifier = Modifier
   423→                .fillMaxWidth()
   424→                .padding(start = 48.dp, end = 48.dp),
   425→            horizontalArrangement = if (isOwnMessage) Arrangement.End else Arrangement.Start
   426→        ) {
   427→            Box(
   428→                modifier = Modifier
   429→                    .clip(RoundedCornerShape(BubbleCornerSize))
   430→                    .background(IncomingBubbleColor)
   431→                    .padding(horizontal = 14.dp, vertical = 10.dp)
   432→            ) {
   433→                Text(
   434→                    text = "This message was deleted",
   435→                    color = TextSecondary,
   436→                    fontSize = 14.sp,
   437→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   438→                )
   439→            }
   440→        }
   441→        return
   442→    }
   443→
   444→    if (message.isDeletedForMe) return
   445→
   446→    val bubbleColor = if (isOwnMessage) NeonMintAlpha15 else IncomingBubbleColor
   447→    val alignment = if (isOwnMessage) Arrangement.End else Arrangement.Start
   448→    val shape = RoundedCornerShape(
   449→        topStart = BubbleCornerSize,
   450→        topEnd = BubbleCornerSize,
   451→        bottomStart = if (isOwnMessage) BubbleCornerSize else 4.dp,
   452→        bottomEnd = if (isOwnMessage) 4.dp else BubbleCornerSize
   453→    )
   454→
   455→    Row(
   456→        modifier = Modifier
   457→            .fillMaxWidth()
   458→            .padding(start = if (isOwnMessage) 48.dp else 0.dp, end = if (!isOwnMessage) 48.dp else 0.dp),
   459→        horizontalArrangement = alignment
   460→    ) {
   461→        Column(
   462→            modifier = Modifier
   463→                .clip(shape)
   464→                .background(bubbleColor)
   465→                .combinedClickable(
   466→                    interactionSource = remember { MutableInteractionSource() },
   467→                    indication = null,
   468→                    onClick = {},
   469→                    onLongClick = onLongClick
   470→                )
   471→                .padding(horizontal = 14.dp, vertical = 10.dp)
   472→        ) {
   473→            // Sender name label (group-specific: show for incoming messages)
   474→            if (!isOwnMessage) {
   475→                val sender = participants.firstOrNull { it.uid == message.senderUid }
   476→                val senderColor = senderNameColor(message.senderUid)
   477→                Text(
   478→                    text = sender?.displayName ?: message.senderDisplayName,
   479→                    color = senderColor,
   480→                    fontSize = 13.sp,
   481→                    fontWeight = FontWeight.SemiBold,
   482→                    maxLines = 1
   483→                )
   484→                Spacer(modifier = Modifier.height(2.dp))
   485→            }
   486→
   487→            // Reply reference preview
   488→            if (message.replyToPreview != null) {
   489→                Row(
   490→                    modifier = Modifier
   491→                        .fillMaxWidth()
   492→                        .clip(RoundedCornerShape(8.dp))
   493→                        .background(NeonMintAlpha15)
   494→                        .padding(horizontal = 8.dp, vertical = 4.dp)
   495→                ) {
   496→                    Text(
   497→                        text = message.replyToSenderName ?: "",
   498→                        color = NeonMint,
   499→                        fontSize = 12.sp,
   500→                        fontWeight = FontWeight.SemiBold,
   501→                        maxLines = 1
   502→                    )
   503→                    Spacer(modifier = Modifier.width(6.dp))
   504→                    Text(
   505→                        text = message.replyToPreview ?: "",
   506→                        color = TextSecondary,
   507→                        fontSize = 12.sp,
   508→                        maxLines = 1,
   509→                        overflow = TextOverflow.Ellipsis,
   510→                        modifier = Modifier.weight(1f)
   511→                    )
   512→                }
   513→                Spacer(modifier = Modifier.height(4.dp))
   514→            }
   515→
   516→            // Forwarded label
   517→            if (message.isForwarded) {
   518→                Text(
   519→                    text = "Forwarded",
   520→                    color = TextSecondary,
   521→                    fontSize = 11.sp,
   522→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   523→                )
   524→                Spacer(modifier = Modifier.height(2.dp))
   525→            }
   526→
   527→            // Media thumbnail
   528→            if (message.type == MessageContentType.IMAGE || message.type == MessageContentType.VIDEO) {
   529→                val imageUrl = message.mediaThumbnailUrl ?: message.mediaUrl
   530→                if (!imageUrl.isNullOrBlank()) {
   531→                    AsyncImage(
   532→                        model = imageUrl,
   533→                        contentDescription = "Media",
   534→                        modifier = Modifier
   535→                            .fillMaxWidth()
   536→                            .height(180.dp)
   537→                            .clip(RoundedCornerShape(10.dp)),
   538→                        contentScale = ContentScale.Crop
   539→                    )
   540→                    Spacer(modifier = Modifier.height(4.dp))
   541→                }
   542→                if (message.type == MessageContentType.VIDEO) {
   543→                    Box(
   544→                        modifier = Modifier
   545→                            .size(36.dp)
   546→                            .clip(CircleShape)
   547→                            .background(DarkPetrolCharcoal.copy(alpha = 0.7f)),
   548→                        contentAlignment = Alignment.Center
   549→                    ) {
   550→                        Icon(
   551→                            imageVector = Icons.Filled.PlayArrow,
   552→                            contentDescription = "Play",
   553→                            tint = TextPrimary,
   554→                            modifier = Modifier.size(20.dp)
   555→                        )
   556→                    }
   557→                }
   558→            }
   559→
   560→            // Voice message indicator
   561→            if (message.type == MessageContentType.AUDIO_VOICE) {
   562→                Row(verticalAlignment = Alignment.CenterVertically) {
   563→                    Icon(
   564→                        imageVector = Icons.Filled.PlayArrow,
   565→                        contentDescription = "Voice",
   566→                        tint = NeonMint,
   567→                        modifier = Modifier.size(20.dp)
   568→                    )
   569→                    Spacer(modifier = Modifier.width(6.dp))
   570→                    Text(text = "Voice message", color = TextSecondary, fontSize = 14.sp)
   571→                }
   572→                Spacer(modifier = Modifier.height(4.dp))
   573→            }
   574→
   575→            // Text content
   576→            if (message.content.isNotBlank()) {
   577→                Text(
   578→                    text = message.content,
   579→                    color = TextPrimary,
   580→                    fontSize = 15.sp,
   581→                    lineHeight = 20.sp
   582→                )
   583→            }
   584→
   585→            // Media caption

  1157→/** Formats epoch milliseconds to a readable time string. */
  1158→private fun formatTimestamp(epochMs: Long): String {
  1159→    if (epochMs == 0L) return ""
  1160→    val sdf = java.text.SimpleDateFormat("HH:mm", java.util.Locale.getDefault())
  1161→    return sdf.format(java.util.Date(epochMs))
  1162→}
  1163→
  1164→/** Formats ephemeral timer seconds into a human-readable string. */
  1165→private fun formatEphemeralTimer(seconds: Int): String {
  1166→    return when {
  1167→        seconds >= 86400 -> "${seconds / 86400} day${if (seconds / 86400 > 1) "s" else ""}"
  1168→        seconds >= 3600 -> "${seconds / 3600} hour${if (seconds / 3600 > 1) "s" else ""}"
  1169→        seconds >= 60 -> "${seconds / 60} minute${if (seconds / 60 > 1) "s" else ""}"
  1170→        else -> "$seconds second${if (seconds > 1) "s" else ""}"
  1171→    }
  1172→}

  1174→/** Generates a deterministic color for a sender's name in group chats. */
  1175→private fun senderNameColor(uid: String): Color {
  1176→    val colors = listOf(
  1177→        NeonMint,

  1182→        Color(0xFF26C6DA),
  1183→        Color(0xFFEC407A),

  1186→    val index = uid.hashCode().mod(colors.size).let { if (it < 0) -it else it }
  1187→    return colors[index]
