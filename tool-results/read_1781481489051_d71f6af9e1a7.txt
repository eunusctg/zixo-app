   126→@OptIn(ExperimentalMaterial3Api::class)
   127→@Composable
   128→fun StatusTabScreen(
   129→    viewModel: StatusViewModel = hiltViewModel()
   130→) {
   131→    val statusFeed by viewModel.statusFeed.collectAsStateWithLifecycle()
   132→    val myStatuses by viewModel.myStatuses.collectAsStateWithLifecycle()
   133→    val isUploading by viewModel.isUploading.collectAsStateWithLifecycle()
   134→    val uploadProgress by viewModel.uploadProgress.collectAsStateWithLifecycle()
   135→    val errorMessage by viewModel.errorMessage.collectAsStateWithLifecycle()
   136→    val viewingGroup by viewModel.viewingStatusGroup.collectAsStateWithLifecycle()
   137→    val viewingIndex by viewModel.currentViewingIndex.collectAsStateWithLifecycle()
   138→
   139→    var showAddSheet by remember { mutableStateOf(false) }
   140→    var showTextComposer by remember { mutableStateOf(false) }
   141→    var showMediaComposer by remember { mutableStateOf(false) }
   142→    var isVideoMode by remember { mutableStateOf(false) }
   143→    var replyText by remember { mutableStateOf("") }
   144→
   145→    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
   146→
   147→    // Clear error on consumption
   148→    LaunchedEffect(errorMessage) {
   149→        if (errorMessage != null) {
   150→            delay(3_000L)
   151→            viewModel.clearError()
   152→        }
   153→    }
   154→
   155→    Box(modifier = Modifier.fillMaxSize()) {
   156→        ZixoGlassBackground()
   157→
   158→        Column(modifier = Modifier.fillMaxSize()) {
   159→            // ── Header ─────────────────────────────
   160→            Text(
   161→                text = "Status",
   162→                color = TextPrimary,
   163→                fontSize = 24.sp,
   164→                fontWeight = FontWeight.Bold,
   165→                modifier = Modifier.padding(start = 20.dp, top = 16.dp, end = 20.dp, bottom = 8.dp)
   166→            )
   167→
   168→            // ── My Status Card ─────────────────────
   169→            MyStatusCard(
   170→                myStatuses = myStatuses.myStatuses,
   171→                isUploading = isUploading,
   172→                uploadProgress = uploadProgress,
   173→                onAddClick = { showAddSheet = true }
   174→            )
   175→
   176→            Spacer(modifier = Modifier.height(12.dp))
   177→
   178→            // ── Recent Updates Header ──────────────
   179→            Text(
   180→                text = "Recent Updates",
   181→                color = TextSecondary,
   182→                fontSize = 14.sp,
   183→                fontWeight = FontWeight.SemiBold,
   184→                modifier = Modifier.padding(horizontal = 20.dp, vertical = 8.dp)
   185→            )
   186→
   187→            // ── Status Feed ────────────────────────
   188→            if (statusFeed.isEmpty()) {
   189→                EmptyStatusFeed()
   190→            } else {
   191→                LazyColumn(
   192→                    modifier = Modifier.fillMaxSize(),
   193→                    verticalArrangement = Arrangement.spacedBy(4.dp),
   194→                    contentPadding = PaddingValues(bottom = 80.dp)
   195→                ) {
   196→                    items(
   197→                        items = statusFeed,
   198→                        key = { it.senderUid }
   199→                    ) { group ->
   200→                        StatusGroupItem(
   201→                            group = group,
   202→                            onClick = { viewModel.startViewing(group) }
   203→                        )
   204→                    }
   205→                }
   206→            }
   207→        }
   208→
   209→        // ── Error Snack ──────────────────────────
   210→        AnimatedVisibility(
   211→            visible = errorMessage != null,
   212→            enter = fadeIn() + slideInVertically(initialOffsetY = { -it }),
   213→            exit = fadeOut() + slideOutVertically(targetOffsetY = { -it }),
   214→            modifier = Modifier
   215→                .align(Alignment.TopCenter)
   216→                .padding(16.dp)
   217→        ) {
   218→            errorMessage?.let { msg ->
   219→                Box(
   220→                    modifier = Modifier
   221→                        .fillMaxWidth()
   222→                        .liquidGlassCard()
   223→                        .padding(12.dp)
   224→                ) {
   225→                    Text(text = msg, color = DestructiveText, fontSize = 14.sp)
   226→                }
   227→            }
   228→        }
   229→    }
   230→
   231→    // ── Add Status Bottom Sheet ─────────────────
   232→    if (showAddSheet) {
   233→        AddStatusBottomSheet(
   234→            sheetState = sheetState,
   235→            onDismiss = { showAddSheet = false },
   236→            onTextClick = {
   237→                showAddSheet = false
   238→                showTextComposer = true
   239→            },
   240→            onPhotoClick = {
   241→                showAddSheet = false
   242→                isVideoMode = false
   243→                showMediaComposer = true
   244→            },
   245→            onVideoClick = {
   246→                showAddSheet = false
   247→                isVideoMode = true
   248→                showMediaComposer = true
   249→            }
   250→        )
   251→    }
   252→
   253→    // ── Text Status Composer ────────────────────
   254→    if (showTextComposer) {
   255→        TextStatusComposer(
   256→            isUploading = isUploading,
   257→            onDismiss = { showTextComposer = false },
   258→            onPost = { text, bgColor ->
   259→                viewModel.postTextStatus(text, bgColor)
   260→                showTextComposer = false
   261→            }
   262→        )
   263→    }
   264→
   265→    // ── Media Status Composer ───────────────────
   266→    if (showMediaComposer) {
   267→        MediaStatusComposer(
   268→            isVideoMode = isVideoMode,
   269→            isUploading = isUploading,
   270→            uploadProgress = uploadProgress,
   271→            onDismiss = { showMediaComposer = false },
   272→            onPost = { filePath, caption ->
   273→                if (isVideoMode) {
   274→                    viewModel.postVideoStatus(filePath, caption)
   275→                } else {
   276→                    viewModel.postImageStatus(filePath, caption)
   277→                }
   278→                showMediaComposer = false
   279→            }
   280→        )
   281→    }
   282→
   283→    // ── Full-Screen Status Viewer ───────────────
   284→    if (viewingGroup != null) {
   285→        StatusViewerOverlay(
   286→            group = viewingGroup!!,
   287→            currentIndex = viewingIndex,
   288→            replyText = replyText,
   289→            onReplyTextChange = { replyText = it },
   290→            onPrevious = viewModel::previousStatus,
   291→            onNext = viewModel::nextStatus,
   292→            onClose = viewModel::stopViewing,
   293→            onReaction = { statusId, emoji -> viewModel.addReaction(statusId, emoji) }
   294→        )
   295→    }
   296→}

   446→// ──────────────────────────────────────────────
   447→// Status Ring Avatar
   448→// ──────────────────────────────────────────────
   449→
   450→@Composable
   451→private fun StatusRingAvatar(
   452→    avatarUrl: String,
   453→    name: String,
   454→    size: Dp,
   455→    segmentCount: Int,
   456→    ringColor: Color
   457→) {
   458→    val strokeWidth = 3.dp
   459→    val gapAngle = 6f
   460→    val totalGap = gapAngle * segmentCount
   461→    val segmentSweep = if (segmentCount > 0) (360f - totalGap) / segmentCount else 0f
   462→
   463→    Box(contentAlignment = Alignment.Center) {
   464→        // Colored ring segments
   465→        if (segmentCount > 0) {
   466→            Canvas(modifier = Modifier.size(size + strokeWidth * 2)) {
   467→                var startAngle = -90f
   468→                for (i in 0 until segmentCount) {
   469→                    drawArc(
   470→                        color = ringColor,
   471→                        startAngle = startAngle,
   472→                        sweepAngle = segmentSweep,
   473→                        useCenter = false,
   474→                        style = Stroke(
   475→                            width = strokeWidth.toPx(),
   476→                            cap = StrokeCap.Round
   477→                        )
   478→                    )
   479→                    startAngle += segmentSweep + gapAngle
   480→                }
   481→            }
   482→        }
   483→
   484→        AvatarComponent(
   485→            imageUrl = avatarUrl,
   486→            name = name,
   487→            size = size
   488→        )
   489→    }
   490→}

  1259→private fun formatRelativeTime(timestampMs: Long): String {
  1260→    val now = System.currentTimeMillis()
  1261→    val diff = now - timestampMs
  1262→    val seconds = diff / 1000
  1263→    val minutes = seconds / 60
  1264→    val hours = minutes / 60
  1265→    return when {
  1266→        seconds < 60 -> "just now"
