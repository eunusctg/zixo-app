   379→@Composable
   380→private fun ProfileHeaderBlock(
   381→    displayName: String,
   382→    bio: String,
   383→    avatarUrl: String,
   384→    onQrClick: () -> Unit,
   385→    onProfileClick: () -> Unit
   386→) {
   387→    Box(
   388→        modifier = Modifier
   389→            .fillMaxWidth()
   390→            .diagonalMeshGradient()
   391→            .clickable(
   392→                interactionSource = remember { MutableInteractionSource() },
   393→                indication = null,
   394→                onClick = onProfileClick
   395→            )
   396→            .padding(horizontal = 20.dp, vertical = 24.dp)
   397→    ) {
   398→        Row(
   399→            verticalAlignment = Alignment.CenterVertically
   400→        ) {
   401→            // Avatar
   402→            Box(
   403→                modifier = Modifier
   404→                    .size(80.dp)
   405→                    .clip(CircleShape)
   406→                    .background(Color(0xFF1A2A32))
   407→                    .border(2.dp, Color.White.copy(alpha = 0.3f), CircleShape),
   408→                contentAlignment = Alignment.Center
   409→            ) {
   410→                if (avatarUrl.isNotBlank()) {
   411→                    AsyncImage(
   412→                        model = ImageRequest.Builder(LocalContext.current)
   413→                            .data(avatarUrl)
   414→                            .crossfade(true)
   415→                            .build(),
   416→                        contentDescription = "Profile avatar",
   417→                        modifier = Modifier
   418→                            .size(80.dp)
   419→                            .clip(CircleShape),
   420→                        contentScale = ContentScale.Crop
   421→                    )
   422→                } else {
   423→                    Text(
   424→                        text = displayName.firstOrNull()?.uppercase() ?: "?",
   425→                        color = Color.White,
   426→                        fontSize = 32.sp,
   427→                        fontWeight = FontWeight.Bold
   428→                    )
   429→                }
   430→            }
   431→
   432→            Spacer(modifier = Modifier.width(16.dp))
   433→
   434→            // Name + Bio + QR icon
   435→            Column(modifier = Modifier.weight(1f)) {
   436→                Row(
   437→                    verticalAlignment = Alignment.CenterVertically
   438→                ) {
   439→                    Text(
   440→                        text = displayName.ifBlank { "Set your name" },

   442→                        fontSize = 22.sp,
   443→                        fontWeight = FontWeight.Bold,
   444→                        modifier = Modifier.weight(1f, fill = false)

   556→                    onClick = { /* consume clicks inside popup */ }
   557→                ),
   558→            horizontalAlignment = Alignment.CenterHorizontally

   580→
   581→            Spacer(modifier = Modifier.height(20.dp))
   582→
   583→            // QR code rendered in brand emerald green
   584→            val qrData = "zixo://user/$zixoNumber"
   585→            Box(
   586→                modifier = Modifier
   587→                    .size(220.dp)
   588→                    .background(Color.White, RoundedCornerShape(12.dp))

   608→            if (zixoNumber.isNotBlank()) {
   609→                Text(
   610→                    text = formatZixoNumber(zixoNumber),
   611→                    color = NeonMint,
   612→                    fontSize = 16.sp,
   613→                    fontWeight = FontWeight.Medium,
   614→                    fontFamily = androidx.compose.ui.text.font.FontFamily.Monospace,
   615→                    letterSpacing = 2.sp
   616→                )
   617→            }
   618→

   624→                    .fillMaxWidth()
   625→                    .clip(RoundedCornerShape(12.dp))

   656→// ─────────────────────────────────────────────────────────────────────────────
   657→// Simple QR Canvas (Placeholder — generates a visual QR-like pattern)
   658→// ─────────────────────────────────────────────────────────────────────────────
   659→
   660→/**
   661→ * Draws a placeholder QR-code-like pattern for the account identification URI.
   662→ * In production, this would use a real QR code library (e.g., zxing).
   663→ */
   664→@Composable
   665→private fun SimpleQrCanvas(
   666→    data: String,
   667→    color: Color,
   668→    modifier: Modifier = Modifier
   669→) {
   670→    Canvas(modifier = modifier) {
   671→        val gridSize = 21 // Standard QR Version 1
   672→        val cellSize = size.minDimension / gridSize
   673→        val hash = data.hashCode()
   674→
   675→        for (row in 0 until gridSize) {
   676→            for (col in 0 until gridSize) {
   677→                val isFinderPattern = (
   678→                    (row < 7 && col < 7) ||
   679→                    (row < 7 && col > gridSize - 8) ||
   680→                    (row > gridSize - 8 && col < 7)
   681→                )
   682→                val isFinderBorder = isFinderPattern && (
   683→                    row == 0 || row == 6 || col == 0 || col == 6 ||
   684→                    (row < 7 && col < 7 && (row == 0 || row == 6 || col == 0 || col == 6)) ||
   685→                    (row < 7 && col > gridSize - 8 && (row == 0 || row == 6 || col == gridSize - 7 || col == gridSize - 1)) ||
   686→                    (row > gridSize - 8 && col < 7 && (row == gridSize - 7 || row == gridSize - 1 || col == 0 || col == 6))
   687→                )
   688→                val isFinderInner = isFinderPattern && (
   689→                    (row in 2..4 && col in 2..4) ||
   690→                    (row in 2..4 && col in gridSize - 5..gridSize - 3) ||
   691→                    (row in gridSize - 5..gridSize - 3 && col in 2..4)
   692→                )
   693→
   694→                val cellHash = (hash * 31 + row * 17 + col * 7) and 0xFF
   695→                val shouldFill = isFinderBorder || isFinderInner || (!isFinderPattern && cellHash % 3 != 0)
   696→
   697→                if (shouldFill) {
   698→                    drawRect(
   699→                        color = color,
   700→                        topLeft = androidx.compose.ui.geometry.Offset(
   701→                            x = col * cellSize,
   702→                            y = row * cellSize
   703→                        ),
   704→                        size = Size(cellSize, cellSize)
   705→                    )
   706→                }
   707→            }
   708→        }
   709→    }
   710→}
