   608→private fun VisibilityOption.toIndex(): Int = when (this) {
   609→    VisibilityOption.EVERYONE -> 0
   610→    VisibilityOption.CONTACTS -> 1
   611→    VisibilityOption.NOBODY -> 2
   612→}
   613→
   614→private fun Int.toVisibilityOption(): VisibilityOption = when (this) {
   615→    0 -> VisibilityOption.EVERYONE
   616→    1 -> VisibilityOption.CONTACTS
   617→    else -> VisibilityOption.NOBODY
   618→}
   619→
   620→private fun StatusPrivacyOption.toIndex(): Int = when (this) {
