    41→    41→/**
    42→    42→ * Core WebRTC Client — Crash-Proof Pure WebRTC Implementation (NO LiveKit).
    43→    43→ *
    44→    44→ * ## Rigid Native WebRTC Processing Guardrails:
    45→    45→ *
    46→    46→ * 1. **All Media Isolated on Dispatchers.IO:**
    47→    47→ *    PeerConnectionFactory, PeerConnection, SDP generation, and ICE candidate
    48→    48→ *    discovery are handled exclusively on background thread scopes.
    49→    49→ *    The Android Main Thread is NEVER blocked by WebRTC operations.
    50→    50→ *
    51→    51→ * 2. **Mandatory Foreground Service:**
    52→    52→ *    CallForegroundService (FOREGROUND_SERVICE_TYPE_CAMERA | MICROPHONE | PHONE_CALL)
    53→    53→ *    wraps every WebRTC session so Android does not terminate media streams
    54→    54→ *    when the user backgrounds the application.
    55→    55→ *
    56→    56→ * 3. **Hardware Acoustic Echo & Routing Calibration:**
    57→    57→ *    AudioManager.MODE_IN_COMMUNICATION + AudioFocusRequest(USAGE_VOICE_COMMUNICATION)
    58→    58→ *    eliminates system echo feedback loops and fixes audio routing to
    59→    59→ *    speakerphone or Bluetooth accessories.
    60→    60→ *
    61→    61→ * 4. **EglBase Singleton Scope Preservation:**
    62→    62→ *    The hardware-accelerated root EglBase context is created ONCE as a
    63→    63→ *    DI Singleton and NEVER re-initialized on Composable recomposition.
    64→    64→ *    Re-creating EglBase during recomposition triggers an uncatchable SIGABRT.
    65→    65→ *    SurfaceViewRenderer instances are wrapped inside AndroidView.
    66→    66→ *
    67→    67→ * 5. **Structured try-catch Boundaries:**
    68→    68→ *    Every WebRTC operation has a precise try-catch boundary preventing
    69→    69→ *    any native crash from propagating as an unhandled exception.
    70→    70→ */

   229→   229→    /**
   230→   230→     * Configures the Android audio system for a WebRTC call.
   231→   231→     *
   232→   232→     * This method applies the following critical settings:
   233→   233→     * - **AudioManager.MODE_IN_COMMUNICATION**: Optimizes the audio path for
   234→   234→     *   two-way voice communication, enabling echo cancellation and noise
   235→   235→     *   suppression at the hardware level.
   236→   236→     * - **AudioFocusRequest with USAGE_VOICE_COMMUNICATION**: Requests
   237→   237→     *   permanent audio focus for voice communication, which ensures proper
   238→   238→     *   audio routing and prevents other apps from interrupting the call.
   239→   239→     * - **Speakerphone default OFF**: Routes audio to the earpiece by default
   240→   240→     *   for audio calls, speaker for video calls.
   241→   241→     *
   242→   242→     * This eliminates system echo feedback loops and fixes audio routing
   243→   243→     * to speakerphone or Bluetooth accessories flawlessly.
   244→   244→     */
   245→   245→    private fun configureAudioForCall() {
   246→   246→        try {
   247→   247→            val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
   248→   248→            if (audioManager == null) {
   249→   249→                Timber.w("AudioManager not available, skipping audio calibration")
   250→   250→                return
   251→   251→            }
   252→   252→
   253→   253→            // Set communication mode — enables AEC (Acoustic Echo Cancellation)
   254→   254→            audioManager.mode = AudioManager.MODE_IN_COMMUNICATION
   255→   255→
   256→   256→            // Request audio focus for voice communication
   257→   257→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
   258→   258→                val playbackAttributes = AudioAttributes.Builder()
   259→   259→                    .setUsage(AudioAttributes.USAGE_VOICE_COMMUNICATION)
   260→   260→                    .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
   261→   261→                    .build()
   262→   262→
   263→   263→                audioFocusRequest = AudioFocusRequest.Builder(AudioManager.AUDIOFOCUS_GAIN_TRANSIENT)
   264→   264→                    .setAudioAttributes(playbackAttributes)
   265→   265→                    .setOnAudioFocusChangeListener { focusChange ->
   266→   266→                        Timber.d("Audio focus changed: %d", focusChange)
   267→   267→                    }
   268→   268→                    .build()
   269→   269→
   270→   270→                val result = audioManager.requestAudioFocus(audioFocusRequest!!)
   271→   271→                isAudioFocusHeld = result == AudioManager.AUDIOFOCUS_REQUEST_GRANTED
   272→   272→                Timber.d("Audio focus request result: %s (granted=%s)", result, isAudioFocusHeld)
   273→   273→            } else {
   274→   274→                @Suppress("DEPRECATION")
   275→   275→                val result = audioManager.requestAudioFocus(
   276→   276→                    null,
   277→   277→                    AudioManager.STREAM_VOICE_CALL,
   278→   278→                    AudioManager.AUDIOFOCUS_GAIN_TRANSIENT
   279→   279→                )
   280→   280→                isAudioFocusHeld = result == AudioManager.AUDIOFOCUS_REQUEST_GRANTED
   281→   281→                Timber.d("Audio focus request (legacy) result: %s", result)
   282→   282→            }
   283→   283→
   284→   284→            // Default: earpiece for audio, speaker for video
   285→   285→            audioManager.isSpeakerphoneOn = false
   286→   286→            audioManager.isBluetoothScoOn = false
   287→   287→
   288→   288→            Timber.d("Audio calibrated: MODE_IN_COMMUNICATION, USAGE_VOICE_COMMUNICATION")
   289→   289→        } catch (e: Exception) {
   290→   290→            Timber.e(e, "Failed to configure audio for call — audio may have echo issues")
   291→   291→        }
   292→   292→    }

   300→   300→            val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
   301→   301→            if (audioManager == null) return
   302→   302→
   303→   303→            // Restore normal audio mode

   305→   305→            audioManager.isSpeakerphoneOn = false
   306→   306→            audioManager.isBluetoothScoOn = false

   310→   310→                audioManager.abandonAudioFocusRequest(audioFocusRequest!!)

   312→   312→                @Suppress("DEPRECATION")

   314→   314→            }
   315→   315→            isAudioFocusHeld = false
   316→   316→
   317→   317→            Timber.d("Audio released: MODE_NORMAL, focus abandoned")
   318→   318→        } catch (e: Exception) {
   319→   319→            Timber.e(e, "Failed to release audio from call")
   320→   320→        }
   321→   321→    }

   612→   612→    // ════════════════════════════════════════════════════════════════
   613→   613→
   614→   614→    /**
   615→   615→     * Disconnects the PeerConnection and releases all WebRTC resources.
   616→   616→     *
   617→   617→     * This method is safe to call from any thread. All cleanup operations
   618→   618→     * are wrapped in individual try-catch blocks to ensure that a failure
   619→   619→     * in releasing one resource does not prevent others from being released.
   620→   620→     *
   621→   621→     * Also releases audio focus and restores the AudioManager to normal mode.
   622→   622→     *
   623→   623→     * **IMPORTANT**: Does NOT dispose the EglBase singleton. The EglBase
   624→   624→     * lives for the entire application lifecycle and must never be destroyed
   625→   625→     * during a call disconnect.
   626→   626→     */
   627→   627→    fun disconnect() {
   628→   628→        try {
   629→   629→            videoCapturer?.tryStopCapture()
   630→   630→        } catch (_: Exception) {}
   631→   631→
   632→   632→        try {
   633→   633→            videoCapturer?.dispose()
   634→   634→        } catch (_: Exception) {}
   635→   635→        videoCapturer = null
   636→   636→
   637→   637→        try {
   638→   638→            localVideoSource?.dispose()
   639→   639→        } catch (_: Exception) {}
   640→   640→        localVideoSource = null
   641→   641→
   642→   642→        try {
   643→   643→            localAudioSource?.dispose()
   644→   644→        } catch (_: Exception) {}
   645→   645→        localAudioSource = null
   646→   646→
   647→   647→        try {
   648→   648→            localVideoTrack?.dispose()
   649→   649→        } catch (_: Exception) {}
   650→   650→        localVideoTrack = null
   651→   651→
   652→   652→        try {
   653→   653→            localAudioTrack?.dispose()
   654→   654→        } catch (_: Exception) {}
   655→   655→        localAudioTrack = null
   656→   656→
   657→   657→        try {
   658→   658→            remoteVideoTrack?.dispose()
   659→   659→        } catch (_: Exception) {}
   660→   660→        remoteVideoTrack = null
   661→   661→
   662→   662→        try {
   663→   663→            surfaceTextureHelper?.dispose()
   664→   664→        } catch (_: Exception) {}
   665→   665→        surfaceTextureHelper = null
   666→   666→
   667→   667→        try {
   668→   668→            peerConnection?.close()
   669→   669→        } catch (_: Exception) {}
   670→   670→
   671→   671→        try {
   672→   672→            peerConnection?.dispose()
   673→   673→        } catch (_: Exception) {}
   674→   674→        peerConnection = null
   675→   675→
   676→   676→        // Release audio focus and restore normal mode
   677→   677→        releaseAudioFromCall()
   678→   678→
   679→   679→        // Reset state flags
   680→   680→        isMuted = false
   681→   681→        isCameraOff = false
   682→   682→        isSpeakerOn = false
   683→   683→        onIceCandidateGenerated = null
   684→   684→        onRemoteVideoTrackAdded = null
   685→   685→
   686→   686→        Timber.d("WebRTC disconnected and all resources released (EglBase preserved)")
   687→   687→    }

   706→   706→    private fun createVideoTrack() {
   707→   707→        try {
   708→   708→            val factory = peerConnectionFactory ?: return
   709→   709→            val enumerator = getCameraEnumerator()
   710→   710→            val deviceNames = enumerator.deviceNames
   711→   711→
   712→   712→            // Try to find front camera first
   713→   713→            var capturer: VideoCapturer? = null
   714→   714→            for (name in deviceNames) {
   715→   715→                if (enumerator.isFrontFacing(name)) {
   716→   716→                    try {
   717→   717→                        capturer = enumerator.createCapturer(name, null)
   718→   718→                        break
   719→   719→                    } catch (e: Exception) {
   720→   720→                        Timber.w(e, "Failed to create front camera capturer, trying next")
   721→   721→                    }
   722→   722→                }
   723→   723→            }
   724→   724→            if (capturer == null && deviceNames.isNotEmpty()) {
   725→   725→                try {
   726→   726→                    capturer = enumerator.createCapturer(deviceNames[0], null)
   727→   727→                } catch (e: Exception) {
   728→   728→                    Timber.e(e, "Failed to create any camera capturer")
   729→   729→                }
   730→   730→            }
   731→   731→
   732→   732→            capturer?.let { vc ->
   733→   733→                videoCapturer = vc
   734→   734→                surfaceTextureHelper = SurfaceTextureHelper.create("CaptureThread", _eglBase.eglBaseContext)
   735→   735→                localVideoSource = factory.createVideoSource(vc.isScreencast)
   736→   736→                vc.initialize(surfaceTextureHelper, context.applicationContext, localVideoSource?.capturerObserver)
   737→   737→                vc.startCapture(640, 480, 30)
   738→   738→
   739→   739→                localVideoTrack = factory.createVideoTrack(VIDEO_TRACK_ID, localVideoSource)
   740→   740→                localVideoTrack?.setEnabled(true)
   741→   741→                peerConnection?.addTrack(localVideoTrack)
   742→   742→                Timber.d("Video track created and added to PeerConnection")
   743→   743→            }
   744→   744→        } catch (e: Exception) {
   745→   745→            Timber.e(e, "Failed to create video track")
   746→   746→        }
   747→   747→    }

   749→   749→    /**
   750→   750→     * Creates the PeerConnection observer that handles all WebRTC events.
   751→   751→     *
   752→   752→     * ICE candidates are forwarded to the signaling layer via
   753→   753→     * [onIceCandidateGenerated] callback. Remote video tracks are
   754→   754→     * forwarded via [onRemoteVideoTrackAdded] callback.
   755→   755→     */
   756→   756→    private fun createPeerConnectionObserver() = object : PeerConnection.Observer {
   757→   757→        override fun onSignalingChange(state: PeerConnection.SignalingState?) {
   758→   758→            Timber.d("Signaling state: %s", state)
   759→   759→        }
   760→   760→
   761→   761→        override fun onIceConnectionChange(state: PeerConnection.IceConnectionState?) {
   762→   762→            Timber.d("ICE connection state: %s", state)
   763→   763→            when (state) {
   764→   764→                PeerConnection.IceConnectionState.CONNECTED -> {
   765→   765→                    Timber.d("WebRTC ICE connected")
   766→   766→                }
   767→   767→                PeerConnection.IceConnectionState.DISCONNECTED -> {
   768→   768→                    Timber.w("WebRTC ICE disconnected")
   769→   769→                }
   770→   770→                PeerConnection.IceConnectionState.FAILED -> {
   771→   771→                    Timber.e("WebRTC ICE connection failed")
   772→   772→                }
   773→   773→                else -> {}
   774→   774→            }
   775→   775→        }
   776→   776→
   777→   777→        override fun onIceConnectionReceivingChange(receiving: Boolean) {}
   778→   778→
   779→   779→        override fun onIceGatheringChange(state: PeerConnection.IceGatheringState?) {
   780→   780→            Timber.d("ICE gathering state: %s", state)
   781→   781→        }
   782→   782→
   783→   783→        override fun onIceCandidate(candidate: IceCandidate?) {
   784→   784→            candidate?.let {
   785→   785→                Timber.d("Local ICE candidate generated: %s", it.sdp)
   786→   786→                onIceCandidateGenerated?.invoke(it)
   787→   787→            }
   788→   788→        }
   789→   789→
   790→   790→        override fun onIceCandidatesRemoved(candidates: Array<out IceCandidate>?) {
   791→   791→            Timber.d("ICE candidates removed: %d", candidates?.size ?: 0)
   792→   792→        }
   793→   793→
   794→   794→        override fun onAddStream(stream: MediaStream?) {
   795→   795→            Timber.d("Remote stream added with %d video tracks", stream?.videoTracks?.size ?: 0)
   796→   796→        }
   797→   797→
   798→   798→        override fun onRemoveStream(stream: MediaStream?) {
   799→   799→            Timber.d("Remote stream removed")
   800→   800→        }
   801→   801→
   802→   802→        override fun onDataChannel(channel: DataChannel?) {}
   803→   803→
   804→   804→        override fun onRenegotiationNeeded() {
   805→   805→            Timber.d("Renegotiation needed")
   806→   806→        }
   807→   807→
   808→   808→        override fun onAddTrack(receiver: RtpReceiver?, streams: Array<out MediaStream>?) {
   809→   809→            Timber.d("Remote track added")
   810→   810→        }
   811→   811→
   812→   812→        override fun onTrack(transceiver: RtpTransceiver?) {
   813→   813→            transceiver?.let { t ->
   814→   814→                if (t.isVideoType()) {
   815→   815→                    val track = t.receiver.track() as? VideoTrack
   816→   816→                    if (track != null) {
   817→   817→                        remoteVideoTrack = track
   818→   818→                        Timber.d("Remote video track received via onTrack")
   819→   819→                        onRemoteVideoTrackAdded?.invoke(track)
   820→   820→                    }
   821→   821→                }
   822→   822→            }
   823→   823→        }
   824→   824→    }
