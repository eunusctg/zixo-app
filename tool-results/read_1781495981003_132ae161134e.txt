    84→// Account & Security Screen
    85→// ─────────────────────────────────────────────────────────────────────────────
    86→
    87→/**
    88→ * Full-screen "Account & Security" page featuring:
    89→ *
    90→ * 1. **Biometric Gate** — native BiometricPrompt with comprehensive error handling
    91→ *    (HW unavailable, no enrolled biometrics, locked out).
    92→ * 2. **Two-Step PIN** — 4-digit PIN with confirmation step.
    93→ * 3. **Passkey Management** — register new passkeys, delete existing.
    94→ * 4. **Active Sessions** — view and revoke sessions across devices.
    95→ * 5. **Account Deletion** — destructive action with confirmation dialog.
    96→ * 6. **Security Notifications** — toggle for security change alerts.
    97→ *
    98→ * All cards use [liquidGlassCard] with NeonMint for active states and
    99→ * AmoledBlack backgrounds for the dark glass aesthetic.
   100→ */
   101→@Composable
   102→fun AccountSecurityScreen(
   103→    onBackClick: () -> Unit,
   104→    viewModel: SettingsViewModel = hiltViewModel()
   105→) {
   106→    val settingsState by viewModel.settingsState.collectAsStateWithLifecycle()
   107→    val isPasskeyRegistered by viewModel.isPasskeyRegistered.collectAsStateWithLifecycle()
   108→    val context = LocalContext.current
   109→    val lifecycleOwner = LocalLifecycleOwner.current
   110→
   111→    // ── Local UI state ──
   112→    var showPinSetup by remember { mutableStateOf(false) }
   113→    var pin by remember { mutableStateOf("") }
   114→    var confirmPin by remember { mutableStateOf("") }
   115→    var pinStep by remember { mutableIntStateOf(0) } // 0=hidden, 1=enter, 2=confirm
   116→    var pinError by remember { mutableStateOf<String?>(null) }
   117→
   118→    var showDeleteDialog by remember { mutableStateOf(false) }
   119→    var deleteConfirmText by remember { mutableStateOf("") }
   120→
   121→    var biometricStatus by remember { mutableStateOf<BiometricStatus>(BiometricStatus.Unknown) }
   122→    var biometricAuthResult by remember { mutableStateOf<BiometricAuthResult?>(null) }
   123→
   124→    // ── Active sessions (from ViewModel or placeholder) ──
   125→    var sessions by remember { mutableStateOf(sampleSessions) }
   126→
   127→    // ── Passkey list ──
   128→    var passkeys by remember {
   129→        mutableStateOf(
   130→            if (isPasskeyRegistered) listOf(
   131→                PasskeyEntry(id = "pk_001", name = "Primary Device", createdAt = "Jan 15, 2025"),
   132→                PasskeyEntry(id = "pk_002", name = "Backup Key", createdAt = "Feb 03, 2025")
   133→            ) else emptyList()
   134→        )
   135→    }
   136→    var showDeletePasskeyDialog by remember { mutableStateOf<PasskeyEntry?>(null) }
   137→
   138→    // ── Check biometric availability on composition ──
   139→    LaunchedEffect(Unit) {
   140→        val biometricManager = BiometricManager.from(context)
   141→        val canAuthenticate = biometricManager.canAuthenticate(
   142→            BiometricManager.Authenticators.BIOMETRIC_STRONG or
   143→                    BiometricManager.Authenticators.BIOMETRIC_WEAK
   144→        )
   145→        biometricStatus = when (canAuthenticate) {
   146→            BiometricManager.BIOMETRIC_SUCCESS -> BiometricStatus.Available
   147→            BiometricManager.BIOMETRIC_ERROR_NO_HARDWARE -> BiometricStatus.HwUnavailable
   148→            BiometricManager.BIOMETRIC_ERROR_HW_UNAVAILABLE -> BiometricStatus.HwUnavailable
   149→            BiometricManager.BIOMETRIC_ERROR_NONE_ENROLLED -> BiometricStatus.NoEnrolled
   150→            BiometricManager.BIOMETRIC_ERROR_SECURITY_UPDATE_REQUIRED -> BiometricStatus.HwUnavailable
   151→            BiometricManager.BIOMETRIC_ERROR_UNSUPPORTED -> BiometricStatus.HwUnavailable
   152→            BiometricManager.BIOMETRIC_STATUS_UNKNOWN -> BiometricStatus.Unknown
   153→            else -> BiometricStatus.Unknown
   154→        }
   155→    }
   156→
   157→    // ── Update passkeys when registration state changes ──
   158→    LaunchedEffect(isPasskeyRegistered) {
   159→        if (isPasskeyRegistered && passkeys.isEmpty()) {
   160→            passkeys = listOf(
   161→                PasskeyEntry(id = "pk_001", name = "Primary Device", createdAt = "Today")
   162→            )
   163→        }
   164→    }
   165→
   166→    // ── Delete passkey dialog ──
   167→    if (showDeletePasskeyDialog != null) {
   168→        AlertDialog(
   169→            onDismissRequest = { showDeletePasskeyDialog = null },
   170→            title = {
   171→                Text(
   172→                    text = "Delete Passkey?",
   173→                    color = TextPrimary
   174→                )
   175→            },
   176→            text = {
   177→                Text(
   178→                    text = "Removing \"${showDeletePasskeyDialog!!.name}\" means you will no longer be able to sign in with this passkey. Make sure you have another authentication method available.",
   179→                    color = TextSecondary,
   180→                    fontSize = 14.sp
   181→                )
   182→            },
   183→            confirmButton = {
   184→                TextButton(
   185→                    onClick = {
   186→                        passkeys = passkeys.filter { it.id != showDeletePasskeyDialog!!.id }
   187→                        showDeletePasskeyDialog = null
   188→                    }
   189→                ) {
   190→                    Text("Delete", color = DestructiveText, fontWeight = FontWeight.SemiBold)
   191→                }
   192→            },
   193→            dismissButton = {
   194→                TextButton(onClick = { showDeletePasskeyDialog = null }) {
   195→                    Text("Cancel", color = TextSecondary)
   196→                }
   197→            },
   198→            containerColor = AmoledBlack,
   199→            titleContentColor = TextPrimary,
   200→            textContentColor = TextSecondary
   201→        )
   202→    }
   203→
   204→    // ── Delete account confirmation dialog ──
   205→    if (showDeleteDialog) {
   206→        AlertDialog(
   207→            onDismissRequest = {
   208→                showDeleteDialog = false
   209→                deleteConfirmText = ""
   210→            },
   211→            title = {
   212→                Text(
   213→                    text = "Delete Your Account?",
   214→                    color = DestructiveText,
   215→                    fontWeight = FontWeight.Bold
   216→                )
   217→            },
   218→            text = {
   219→                Column {
   220→                    Text(
   221→                        text = "This action is permanent and cannot be undone. All your messages, media, contacts, and account data will be permanently erased from Zixo servers.",
   222→                        color = TextSecondary,
   223→                        fontSize = 14.sp
   224→                    )
   225→                    Spacer(modifier = Modifier.height(16.dp))
   226→                    Text(
   227→                        text = "Type DELETE to confirm:",
   228→                        color = TextPrimary,
   229→                        fontSize = 13.sp,
   230→                        fontWeight = FontWeight.Medium
   231→                    )
   232→                    Spacer(modifier = Modifier.height(8.dp))
   233→                    GlassOutlinedTextField(
   234→                        value = deleteConfirmText,
   235→                        onValueChange = { deleteConfirmText = it },
   236→                        placeholder = { Text("DELETE", color = TextSecondary.copy(alpha = 0.5f)) },
   237→                        modifier = Modifier.fillMaxWidth(),
   238→                        singleLine = true
   239→                    )
   240→                }
   241→            },
   242→            confirmButton = {
   243→                TextButton(
   244→                    onClick = {
   245→                        viewModel.deleteAccount()
   246→                        showDeleteDialog = false
   247→                        deleteConfirmText = ""
   248→                    },
   249→                    enabled = deleteConfirmText == "DELETE"
   250→                ) {
   251→                    Text(
   252→                        "Delete Account",
   253→                        color = if (deleteConfirmText == "DELETE") DestructiveText else TextSecondary.copy(alpha = 0.3f),
   254→                        fontWeight = FontWeight.SemiBold
   255→                    )
   256→                }
   257→            },
   258→            dismissButton = {
   259→                TextButton(
   260→                    onClick = {
   261→                        showDeleteDialog = false
   262→                        deleteConfirmText = ""
   263→                    }
   264→                ) {
   265→                    Text("Cancel", color = TextSecondary)
   266→                }
   267→            },
   268→            containerColor = AmoledBlack,
   269→            titleContentColor = DestructiveText,
   270→            textContentColor = TextSecondary
   271→        )
   272→    }
   273→
   274→    Box(modifier = Modifier.fillMaxSize()) {
   275→        ZixoGlassBackground()
   276→
   277→        LazyColumn(
   278→            modifier = Modifier.fillMaxSize(),
   279→            contentPadding = PaddingValues(start = 16.dp, end = 16.dp, top = 4.dp, bottom = 32.dp),
   280→            verticalArrangement = Arrangement.spacedBy(12.dp)
   281→        ) {
   282→            // ── Top bar ──
   283→            item {
   284→                ZixoTopBar(
   285→                    title = "Account & Security",
   286→                    showBackButton = true,
   287→                    onBackClick = onBackClick
   288→                )
   289→            }
   290→
   291→            // ── Biometric Authentication ─────────────────────────────────
   292→            item {
   293→                BiometricSection(
   294→                    biometricStatus = biometricStatus,
   295→                    biometricAuthResult = biometricAuthResult,
   296→                    onAuthenticate = {
   297→                        launchBiometricPrompt(
   298→                            context = context,
   299→                            lifecycleOwner = lifecycleOwner,
   300→                            onSuccess = {
   301→                                biometricAuthResult = BiometricAuthResult.Success
   302→                            },
   303→                            onFailed = {
   304→                                biometricAuthResult = BiometricAuthResult.Failed
   305→                            },
   306→                            onError = { errorCode, errorString ->
   307→                                biometricAuthResult = BiometricAuthResult.Error(errorCode, errorString)
   308→                            }
   309→                        )
   310→                    },
   311→                    onDismissResult = { biometricAuthResult = null }
   312→                )
   313→            }
   314→
   315→            // ── Passkey Management ───────────────────────────────────────
   316→            item {
   317→                PasskeyManagementCard(
   318→                    passkeys = passkeys,
   319→                    isPasskeyRegistered = isPasskeyRegistered,
   320→                    isLoading = settingsState.isLoading,
   321→                    onRegisterPasskey = {
   322→                        viewModel.createPasskey(
   323→                            context = context,
   324→                            requestJson = """{"publicKey":{"rp":{"name":"Zixo"},"user":{"name":"user","displayName":"User"},"challenge":"placeholder","pubKeyCredParams":[{"type":"public-key","alg":-7}],"timeout":60000,"attestation":"none"}}"""
   325→                        )
   326→                    },
   327→                    onDeletePasskey = { passkey -> showDeletePasskeyDialog = passkey },
   328→                    errorMessage = if (!settingsState.errorMessage.isNullOrBlank() && !isPasskeyRegistered) settingsState.errorMessage else null
   329→                )
   330→            }
   331→
   332→            // ── Two-Step Verification (4-digit PIN) ─────────────────────
   333→            item {
   334→                PinSetupCard(
   335→                    isPinEnabled = settingsState.isTwoStepEnabled,
   336→                    showPinSetup = showPinSetup,
   337→                    pinStep = pinStep,
   338→                    pin = pin,
   339→                    confirmPin = confirmPin,
   340→                    pinError = pinError,
   341→                    onTogglePinSetup = { showPinSetup = !showPinSetup },
   342→                    onPinChange = { newPin ->
   343→                        pin = newPin
   344→                        pinError = null
   345→                        if (newPin.length == 4 && pinStep == 1) {
   346→                            pinStep = 2
   347→                        }
   348→                    },
   349→                    onConfirmPinChange = { newConfirm ->
   350→                        confirmPin = newConfirm
   351→                        pinError = null
   352→                        if (newConfirm.length == 4) {
   353→                            if (confirmPin == pin) {
   354→                                viewModel.updateTwoStep(true)
   355→                                pinStep = 0
   356→                                showPinSetup = false
   357→                                pin = ""
   358→                                confirmPin = ""
   359→                            } else {
   360→                                pinError = "PINs do not match. Try again."
   361→                                confirmPin = ""
   362→                            }
   363→                        }
   364→                    },
   365→                    onStartPinSetup = {
   366→                        pinStep = 1
   367→                        pin = ""
   368→                        confirmPin = ""
   369→                        pinError = null
   370→                    },
   371→                    onDisablePin = {
   372→                        viewModel.updateTwoStep(false)
   373→                        pin = ""
   374→                        confirmPin = ""
   375→                    },
   376→                    onBackStep = {
   377→                        if (pinStep == 2) {
   378→                            pinStep = 1
   379→                            confirmPin = ""
   380→                        } else {
   381→                            pinStep = 0
   382→                            pin = ""
   383→                            showPinSetup = false
   384→                        }
   385→                    }
   386→                )
   387→            }
   388→
   389→            // ── Security Notifications ───────────────────────────────────
   390→            item {
   391→                Column(
   392→                    modifier = Modifier
   393→                        .fillMaxWidth()
   394→                        .liquidGlassCard()
   395→                        .padding(16.dp)
   396→                ) {
   397→                    SectionLabel("SECURITY")
   398→                    Spacer(modifier = Modifier.height(16.dp))
   399→
   400→                    Row(
   401→                        modifier = Modifier.fillMaxWidth(),
   402→                        verticalAlignment = Alignment.CenterVertically
   403→                    ) {
   404→                        Icon(
   405→                            imageVector = Icons.Default.EnhancedEncryption,
   406→                            contentDescription = null,
   407→                            tint = TextSecondary,
   408→                            modifier = Modifier.size(22.dp)
   409→                        )
   410→                        Spacer(modifier = Modifier.width(12.dp))
   411→                        Column(modifier = Modifier.weight(1f)) {
   412→                            Text(
   413→                                text = "Security Notifications",
   414→                                color = TextPrimary,
   415→                                fontSize = 16.sp,
   416→                                fontWeight = FontWeight.Medium
   417→                            )
   418→                            Text(
   419→                                text = "Get notified when security changes occur",
   420→                                color = TextSecondary,
   421→                                fontSize = 13.sp
   422→                            )
   423→                        }
   424→                        GlassSwitch(
   425→                            checked = settingsState.isSecurityNotificationsEnabled,
   426→                            onCheckedChange = { viewModel.updateSecurityNotifications(it) }
   427→                        )
   428→                    }
   429→                }
   430→            }
   431→
   432→            // ── Active Sessions ──────────────────────────────────────────
   433→            item {
   434→                ActiveSessionsCard(
   435→                    sessions = sessions,
   436→                    onRevokeSession = { session ->
   437→                        sessions = sessions.filter { it.id != session.id }
   438→                        Toast.makeText(context, "Session revoked: ${session.deviceName}", Toast.LENGTH_SHORT).show()
   439→                    },
   440→                    onRevokeAllOther = {
   441→                        val currentSession = sessions.firstOrNull { it.isActive }
   442→                        sessions = if (currentSession != null) listOf(currentSession) else emptyList()
   443→                        Toast.makeText(context, "All other sessions revoked", Toast.LENGTH_SHORT).show()
   444→                    }
   445→                )
   446→            }
   447→
   448→            // ── Account Deletion ─────────────────────────────────────────
   449→            item {
   450→                Column(
   451→                    modifier = Modifier
   452→                        .fillMaxWidth()
   453→                        .liquidGlassCard()
   454→                        .padding(16.dp)
   455→                ) {
   456→                    SectionLabel("DANGER ZONE")
   457→                    Spacer(modifier = Modifier.height(16.dp))
   458→
   459→                    Row(
   460→                        modifier = Modifier.fillMaxWidth(),
   461→                        verticalAlignment = Alignment.CenterVertically
   462→                    ) {
   463→                        Icon(
   464→                            imageVector = Icons.Default.DeleteForever,
   465→                            contentDescription = null,
   466→                            tint = DestructiveText,
   467→                            modifier = Modifier.size(22.dp)
   468→                        )
   469→                        Spacer(modifier = Modifier.width(12.dp))
   470→                        Column(modifier = Modifier.weight(1f)) {
   471→                            Text(
   472→                                text = "Delete My Account",
   473→                                color = DestructiveText,
   474→                                fontSize = 16.sp,
   475→                                fontWeight = FontWeight.Medium
   476→                            )
   477→                            Text(
   478→                                text = "Permanently erase all data and close your account",
   479→                                color = TextSecondary,
   480→                                fontSize = 13.sp
   481→                            )
   482→                        }
   483→                    }
   484→
   485→                    Spacer(modifier = Modifier.height(12.dp))
   486→
   487→                    Row(
   488→                        modifier = Modifier
   489→                            .fillMaxWidth()
   490→                            .clip(RoundedCornerShape(12.dp))
   491→                            .background(DestructiveBackground)
   492→                            .border(1.dp, DestructiveText.copy(alpha = 0.4f), RoundedCornerShape(12.dp))
   493→                            .clickable { showDeleteDialog = true }
   494→                            .padding(vertical = 14.dp),
   495→                        horizontalArrangement = Arrangement.Center,
   496→                        verticalAlignment = Alignment.CenterVertically
   497→                    ) {
   498→                        Icon(
   499→                            imageVector = Icons.Default.DeleteForever,
   500→                            contentDescription = null,
   501→                            tint = DestructiveText,
   502→                            modifier = Modifier.size(18.dp)
   503→                        )
   504→                        Spacer(modifier = Modifier.width(8.dp))
   505→                        Text(
   506→                            text = "Delete Account",
   507→                            color = DestructiveText,
   508→                            fontSize = 16.sp,
   509→                            fontWeight = FontWeight.SemiBold
   510→                        )
   511→                    }
   512→                }
   513→            }
   514→        }
   515→    }

   580→// ─────────────────────────────────────────────────────────────────────────────
