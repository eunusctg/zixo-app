   122→// ──────────────────────────────────────────────────────────────────────────────
   123→// Main Settings Screen
   124→// ──────────────────────────────────────────────────────────────────────────────
   125→
   126→@Composable
   127→fun SettingsScreen(
   128→    navController: NavController,
   129→    viewModel: SettingsViewModel = hiltViewModel()
   130→) {
   131→    val uiState by viewModel.uiState.collectAsState()
   132→
   133→    val backgroundBrush = Brush.verticalGradient(
   134→        colors = listOf(BackgroundGradientStart, BackgroundGradientEnd)
   135→    )
   136→
   137→    Scaffold(
   138→        topBar = {
   139→            ZixoTopBar(
   140→                title = "Settings",
   141→                showBackButton = true,
   142→                onBackClick = { navController.popBackStack() }
   143→            )
   144→        },
   145→        containerColor = Color.Transparent
   146→    ) { innerPadding ->
   147→        Box(
   148→            modifier = Modifier
   149→                .fillMaxSize()
   150→                .background(backgroundBrush)
   151→        ) {
   152→            LazyColumn(
   153→                modifier = Modifier
   154→                    .fillMaxSize()
   155→                    .padding(innerPadding),
   156→                contentPadding = PaddingValues(
   157→                    start = 16.dp,
   158→                    end = 16.dp,
   159→                    top = 8.dp,
   160→                    bottom = 32.dp
   161→                ),
   162→                verticalArrangement = Arrangement.spacedBy(ItemVerticalSpacing)
   163→            ) {
   164→                // ── A. Header / User Profile Section ──────────────────────
   165→                item { UserProfileSection(uiState = uiState, onEditProfile = { navController.navigate(SettingsRoutes.EDIT_PROFILE) }) }
   166→                item { Spacer(modifier = Modifier.height(SectionVerticalSpacing)) }
   167→
   168→                // ── B. Appearance Section ─────────────────────────────────
   169→                item { SectionHeader(title = "Appearance") }
   170→                item { AppearanceSection(uiState = uiState, viewModel = viewModel, onWallpaperClick = { navController.navigate(SettingsRoutes.WALLPAPER) }) }
   171→                item { Spacer(modifier = Modifier.height(SectionVerticalSpacing)) }
   172→
   173→                // ── C. Privacy & Security Section ─────────────────────────
   174→                item { SectionHeader(title = "Privacy & Security") }
   175→                item { PrivacySecuritySection(uiState = uiState, viewModel = viewModel, onEncryptionKeyClick = { navController.navigate(SettingsRoutes.ENCRYPTION_KEY) }, onBlockedContactsClick = { navController.navigate(SettingsRoutes.BLOCKED_CONTACTS) }) }
   176→                item { Spacer(modifier = Modifier.height(SectionVerticalSpacing)) }
   177→
   178→                // ── D. Notifications Section ──────────────────────────────
   179→                item { SectionHeader(title = "Notifications") }
   180→                item { NotificationsSection(uiState = uiState, viewModel = viewModel, onNotificationToneClick = { navController.navigate(SettingsRoutes.NOTIFICATION_TONE) }) }
   181→                item { Spacer(modifier = Modifier.height(SectionVerticalSpacing)) }
   182→
   183→                // ── E. Data & Storage Section ─────────────────────────────
   184→                item { SectionHeader(title = "Data & Storage") }
   185→                item { DataStorageSection(uiState = uiState, viewModel = viewModel) }
   186→                item { Spacer(modifier = Modifier.height(SectionVerticalSpacing)) }
   187→
   188→                // ── F. Call Settings Section ──────────────────────────────
   189→                item { SectionHeader(title = "Call Settings") }
   190→                item { CallSettingsSection(uiState = uiState, viewModel = viewModel) }
   191→                item { Spacer(modifier = Modifier.height(SectionVerticalSpacing)) }
   192→
   193→                // ── G. About & Lifecycle Management Section ───────────────
   194→                item { SectionHeader(title = "About") }
   195→                item { AboutSection(uiState = uiState, onAdvancedNetworkClick = { navController.navigate(SettingsRoutes.ADVANCED_NETWORK) }, onEnterpriseSecurityClick = { navController.navigate(SettingsRoutes.ENTERPRISE_SECURITY) }, onAdvancedDataClick = { navController.navigate(SettingsRoutes.ADVANCED_DATA) }, onLogoutClick = { viewModel.showLogoutDialog() }, onDeleteAccountClick = { viewModel.showDeleteDialog() }) }
   196→            }
   197→
   198→            // ── Logout Confirmation Dialog ───────────────────────────────
   199→            if (uiState.showLogoutDialog) {
   200→                LogoutConfirmationDialog(
   201→                    isLoggingOut = uiState.isLoggingOut,
   202→                    onConfirm = { viewModel.logout() },
   203→                    onDismiss = { viewModel.dismissLogoutDialog() }
   204→                )
   205→            }
   206→
   207→            // ── Delete Account Double-Confirmation Dialog ────────────────
   208→            if (uiState.showDeleteDialog) {
   209→                DeleteAccountDialog(
   210→                    isDeleting = uiState.isDeletingAccount,
   211→                    onConfirm = { viewModel.deleteAccount() },
   212→                    onDismiss = { viewModel.dismissDeleteDialog() }
   213→                )
   214→            }
   215→        }
   216→    }
   217→}

   222→
   223→@Composable
   224→private fun UserProfileSection(
   225→    uiState: SettingsUiState,
   226→    onEditProfile: () -> Unit
   227→) {
   228→    val user = uiState.currentUser
   229→
   230→    Card(
   231→        modifier = Modifier.fillMaxWidth(),
   232→        shape = RoundedCornerShape(CardCornerRadius),
   233→        colors = CardDefaults.cardColors(containerColor = CardBackground),
   234→        elevation = CardDefaults.cardElevation(defaultElevation = 0.dp)
   235→    ) {
   236→        Column(
   237→            modifier = Modifier
   238→                .fillMaxWidth()
   239→                .padding(horizontal = CardHorizontalPadding, vertical = 20.dp),
   240→            horizontalAlignment = Alignment.CenterHorizontally
   241→        ) {
   242→            // Avatar - clickable to edit profile
   243→            AvatarComponent(
   244→                imageUrl = user.photoUrl,
   245→                name = user.displayName,
   246→                size = 80.dp,
   247→                isOnline = user.isOnline,
   248→                modifier = Modifier.clickable(onClick = onEditProfile)
   249→            )
   250→
   251→            Spacer(modifier = Modifier.height(12.dp))
   252→
   253→            // Display Name
   254→            Text(
   255→                text = user.displayName.ifBlank { "Unknown User" },
   256→                color = TextPrimary,
   257→                fontSize = 22.sp,
   258→                fontWeight = FontWeight.Bold,
   259→                textAlign = TextAlign.Center
   260→            )
   261→
   262→            Spacer(modifier = Modifier.height(2.dp))
   263→
   264→            // Username
   265→            if (user.username.isNotBlank()) {
   266→                Text(
   267→                    text = "@${user.username}",
   268→                    color = TextSecondary,
   269→                    fontSize = 15.sp,
   270→                    fontWeight = FontWeight.Normal,
   271→                    textAlign = TextAlign.Center
   272→                )
   273→            }
   274→
   275→            // Bio/Status
   276→            if (!user.bio.isNullOrBlank()) {
   277→                Spacer(modifier = Modifier.height(6.dp))
   278→                Text(
   279→                    text = user.bio,
   280→                    color = TextSecondary,
   281→                    fontSize = 14.sp,
   282→                    fontWeight = FontWeight.Normal,
   283→                    textAlign = TextAlign.Center
   284→                )
   285→            }
   286→
   287→            Spacer(modifier = Modifier.height(16.dp))
   288→
   289→            // Zixo Number Badge
   290→            if (user.zixoNumber.isNotBlank()) {
   291→                ZixoNumberBadge(zixoNumber = user.zixoNumber)
   292→            }
   293→
   294→            Spacer(modifier = Modifier.height(12.dp))
   295→
   296→            // QR Code Shortcut Row
   297→            Row(
   298→                modifier = Modifier
   299→                    .fillMaxWidth()
   300→                    .clip(RoundedCornerShape(10.dp))
   301→                    .background(CardBackground.copy(alpha = 0.5f))
   302→                    .clickable { /* Navigate to QR code screen */ }
   303→                    .padding(horizontal = 16.dp, vertical = 12.dp),
   304→                verticalAlignment = Alignment.CenterVertically,
   305→                horizontalArrangement = Arrangement.SpaceBetween
   306→            ) {
   307→                Row(verticalAlignment = Alignment.CenterVertically) {
   308→                    Icon(
   309→                        imageVector = Icons.Filled.QrCode2,
   310→                        contentDescription = "QR Code",
   311→                        tint = AccentGreen,
   312→                        modifier = Modifier.size(22.dp)
   313→                    )
   314→                    Spacer(modifier = Modifier.width(10.dp))
   315→                    Text(
   316→                        text = "My QR Code",
   317→                        color = TextPrimary,
   318→                        fontSize = 15.sp,
   319→                        fontWeight = FontWeight.Medium
   320→                    )
   321→                }
   322→
   323→                Icon(
   324→                    imageVector = Icons.Filled.Share,
   325→                    contentDescription = "Share",
   326→                    tint = TextSecondary,
   327→                    modifier = Modifier.size(20.dp)
   328→                )
   329→            }
   330→        }
   331→    }
   332→}
