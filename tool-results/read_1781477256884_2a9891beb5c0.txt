   109→/**
   110→ * Applies a premium "Liquid Glass" container effect to any composable.
   111→ *
   112→ * This modifier creates a frosted-glass appearance by combining:
   113→ * - A semi-transparent dark surface background
   114→ * - A 24.dp blur effect (on API 31+; gracefully degrades to background-only on older devices)
   115→ * - A thin 1.dp frosted white border
   116→ * - 20.dp rounded corners
   117→ * - Clipping to ensure the blur stays within bounds
   118→ *
   119→ * Use this for large content containers such as settings panels or main content sections.
   120→ *
   121→ * @return A [Modifier] with the liquid glass container styling applied.

   124→    cornerRadius = 20.dp,
   125→    blurRadius = 24.dp

   182→// ZixoGlassBackground – Animated Full-Screen Background
   183→// ──────────────────────────────────────────────
   184→
   185→/**
   186→ * Full-screen animated background composable that renders the signature
   187→ * Zixo "Liquid Glass" atmosphere.
   188→ *
   189→ * The background consists of:
   190→ * - A deep dark slate/teal linear gradient base (#0B1519 → #111E24)
   191→ * - Three slowly-moving, soft-blurred radial gradient "blobs" underneath:
   192→ *     1. **Mint/Emerald** (300dp) – drifts across the top-left region
   193→ *     2. **Teal/Cyan** (250dp) – drifts across the bottom-right region
   194→ *     3. **Purple/Indigo** (200dp) – drifts through the center region
   195→ *
   196→ * Each blob follows a gentle Lissajous oscillation pattern using
   197→ * [infiniteRepeatable] animations with 15–25 second durations,
   198→ * creating a living, breathing dark-mode atmosphere.
   199→ *
   200→ * Example usage:
   201→ * ```kotlin
   202→ * Box(modifier = Modifier.fillMaxSize()) {
   203→ *     ZixoGlassBackground()
   204→ *     // Your content here…
   205→ * }
   206→ * ```
   207→ */
   208→@Composable
   209→fun ZixoGlassBackground(modifier: Modifier = Modifier) {
   210→    val infiniteTransition = rememberInfiniteTransition(label = "glass_bg_transition")
   211→
   212→    // ── Blob 1: Mint / Emerald – slow top-left drift ──
   213→    val blob1X by infiniteTransition.animateFloat(
   214→        initialValue = 0f,
   215→        targetValue = 1f,
   216→        animationSpec = infiniteRepeatable(
   217→            animation = tween(durationMillis = 20_000),
   218→            repeatMode = RepeatMode.Reverse
   219→        ),
   220→        label = "blob1_x"
   221→    )
   222→    val blob1Y by infiniteTransition.animateFloat(
   223→        initialValue = 0f,
   224→        targetValue = 1f,
   225→        animationSpec = infiniteRepeatable(
   226→            animation = tween(durationMillis = 17_000),
   227→            repeatMode = RepeatMode.Reverse
   228→        ),
   229→        label = "blob1_y"
   230→    )
   231→
   232→    // ── Blob 2: Teal / Cyan – slow bottom-right drift ──
   233→    val blob2X by infiniteTransition.animateFloat(
   234→        initialValue = 1f,
   235→        targetValue = 0f,
   236→        animationSpec = infiniteRepeatable(
   237→            animation = tween(durationMillis = 22_000),
   238→            repeatMode = RepeatMode.Reverse
   239→        ),
   240→        label = "blob2_x"
   241→    )
   242→    val blob2Y by infiniteTransition.animateFloat(
   243→        initialValue = 1f,
   244→        targetValue = 0f,
   245→        animationSpec = infiniteRepeatable(
   246→            animation = tween(durationMillis = 18_000),
   247→            repeatMode = RepeatMode.Reverse
   248→        ),
   249→        label = "blob2_y"
   250→    )
   251→
   252→    // ── Blob 3: Purple / Indigo – slow center drift ──
   253→    val blob3X by infiniteTransition.animateFloat(
   254→        initialValue = 0.3f,
   255→        targetValue = 0.7f,
   256→        animationSpec = infiniteRepeatable(
   257→            animation = tween(durationMillis = 25_000),
   258→            repeatMode = RepeatMode.Reverse
   259→        ),
   260→        label = "blob3_x"
   261→    )
   262→    val blob3Y by infiniteTransition.animateFloat(
   263→        initialValue = 0.5f,
   264→        targetValue = 0.3f,
   265→        animationSpec = infiniteRepeatable(
   266→            animation = tween(durationMillis = 15_000),
   267→            repeatMode = RepeatMode.Reverse
   268→        ),
   269→        label = "blob3_y"
   270→    )
   271→
   272→    BoxWithConstraints(modifier = modifier.fillMaxSize()) {
   273→        val widthPx = maxWidth
   274→        val heightPx = maxHeight
   275→
   276→        // Base gradient
   277→        Box(
   278→            modifier = Modifier
   279→                .fillMaxSize()
   280→                .background(
   281→                    brush = Brush.linearGradient(
   282→                        colors = listOf(BackgroundGradientStart, BackgroundGradientEnd),
   283→                        start = Offset.Zero,
   284→                        end = Offset(
   285→                            x = widthPx.value * 2f,
   286→                            y = heightPx.value * 2f
   287→                        )
   288→                    )
   289→                )
   290→        )
   291→
   292→        // ── Blob 1: Mint / Emerald ──
   293→        val blob1OffsetX = (blob1X * 0.35f + 0.02f) * widthPx
   294→        val blob1OffsetY = (blob1Y * 0.30f + 0.02f) * heightPx
   295→        Box(
   296→            modifier = Modifier
   297→                .offset(x = blob1OffsetX, y = blob1OffsetY)
   298→                .size(300.dp)
   299→                .blur(80.dp)
   300→        ) {
   301→            BlobCanvas(color = BlobMintTint, radiusFraction = 1f)
   302→        }
   303→
   304→        // ── Blob 2: Teal / Cyan ──
   305→        val blob2OffsetX = (blob2X * 0.35f + 0.55f) * widthPx
   306→        val blob2OffsetY = (blob2Y * 0.30f + 0.50f) * heightPx
   307→        Box(
   308→            modifier = Modifier
   309→                .offset(x = blob2OffsetX, y = blob2OffsetY)
   310→                .size(250.dp)
   311→                .blur(80.dp)
   312→        ) {
   313→            BlobCanvas(color = BlobTealTint, radiusFraction = 1f)
   314→        }
   315→
   316→        // ── Blob 3: Purple / Indigo ──
   317→        val blob3OffsetX = (blob3X * 0.40f + 0.25f) * widthPx
   318→        val blob3OffsetY = (blob3Y * 0.35f + 0.25f) * heightPx
   319→        Box(
   320→            modifier = Modifier
   321→                .offset(x = blob3OffsetX, y = blob3OffsetY)
   322→                .size(200.dp)
   323→                .blur(80.dp)
   324→        ) {
   325→            BlobCanvas(color = BlobPurpleTint, radiusFraction = 1f)
   326→        }
   327→    }
   328→}

   330→/**
   331→ * Draws a soft radial gradient blob with the given [color] at full opacity
   332→ * in the center, fading to transparent at the edges.
   333→ *
   334→ * @param color       The tint color for the center of the blob.
   335→ * @param radiusFraction Fraction of the canvas size to use as the gradient radius (0..1).
   336→ */
   337→@Composable
   338→private fun BlobCanvas(color: Color, radiusFraction: Float = 1f) {
   339→    Canvas(modifier = Modifier.fillMaxSize()) {
   340→        val canvasSize = size.minDimension
   341→        val radius = canvasSize * radiusFraction * 0.5f
   342→        drawCircle(
   343→            brush = Brush.radialGradient(
   344→                colors = listOf(
   345→                    color.copy(alpha = 0.45f),
   346→                    color.copy(alpha = 0.20f),
   347→                    color.copy(alpha = 0.05f),
   348→                    Color.Transparent
   349→                ),
   350→                center = center,
   351→                radius = radius
   352→            ),
   353→            radius = radius,
   354→            center = center
   355→        )
   356→    }
   357→}

   534→/**
   535→ * A segmented control picker with the Zixo glass aesthetic.
   536→ *
   537→ * Displays a list of [options] inside a horizontal glass panel.
   538→ * The selected option is highlighted with a bright accent green (#00E676)
   539→ * background and white text, while unselected options appear transparent
   540→ * with secondary text color. The selection indicator animates smoothly
   541→ * using [animateDpAsState] and [animateColorAsState].
   542→ *
   543→ * @param options           The list of string labels for each segment.
   544→ * @param selectedIndex     The index of the currently selected option.
   545→ * @param onOptionSelected  Callback invoked with the new index when the user selects an option.
   546→ * @param modifier          Optional modifier for the picker.
   547→ */
   548→@Composable
   549→fun GlassSegmentedPicker(
   550→    options: List<String>,
   551→    selectedIndex: Int,
   552→    onOptionSelected: (Int) -> Unit,
   553→    modifier: Modifier = Modifier
   554→) {
   555→    val outerShape = RoundedCornerShape(12.dp)
   556→
   557→    Row(
   558→        modifier = modifier
   559→            .fillMaxWidth()
   560→            .clip(outerShape)
   561→            .background(SegmentPanelBackground)
   562→            .border(
   563→                width = 1.dp,
   564→                color = GlassBorderColor,
   565→                shape = outerShape
   566→            )
   567→            .padding(3.dp),
   568→        horizontalArrangement = Arrangement.spacedBy(3.dp),
   569→        verticalAlignment = Alignment.CenterVertically
   570→    ) {
   571→        options.forEachIndexed { index, option ->
   572→            val isSelected = index == selectedIndex
   573→            val segmentShape = RoundedCornerShape(10.dp)
   574→
   575→            val backgroundColor by animateColorAsState(
   576→                targetValue = if (isSelected) AccentGreen else Color.Transparent,
   577→                animationSpec = tween(durationMillis = 250),
   578→                label = "segment_bg_$index"
   579→            )
   580→            val textColor by animateColorAsState(
   581→                targetValue = if (isSelected) Color.White else TextSecondary,
   582→                animationSpec = tween(durationMillis = 250),
   583→                label = "segment_text_$index"
   584→            )
   585→            val fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Normal
   586→
   587→            Row(
   588→                modifier = Modifier
   589→                    .weight(1f)
   590→                    .clip(segmentShape)
   591→                    .background(backgroundColor)
   592→                    .clickable(
   593→                        interactionSource = remember { MutableInteractionSource() },
   594→                        indication = null,
   595→                        onClick = { onOptionSelected(index) }
   596→                    )
   597→                    .padding(vertical = 10.dp),
   598→                horizontalArrangement = Arrangement.Center,
   599→                verticalAlignment = Alignment.CenterVertically
   600→            ) {
   601→                Text(
   602→                    text = option,
   603→                    color = textColor,
   604→                    fontSize = 14.sp,
   605→                    fontWeight = fontWeight
   606→                )
   607→            }
   608→        }
   609→    }
