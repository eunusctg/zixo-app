    33→/**
    34→ * Concrete implementation of [ChatRepository].
    35→ *
    36→ * Manages chat threads and messages using Firestore with continuous
    37→ * snapshot listeners for real-time sync. Uses Firebase Realtime Database
    38→ * for presence and typing indicators, and Firebase Storage for media uploads.
    39→ *
    40→ * Zero-trust enforcement: All send operations verify mutual contact status
    41→ * through [ContactRepository.checkCommunicationGate] before writing.
    42→ * Non-contact messages are blocked at this boundary.
    43→ */
    44→@Singleton
    45→class ChatRepositoryImpl @Inject constructor(
    46→    private val firestore: FirebaseFirestore,
    47→    private val realtimeDb: FirebaseDatabase,
    48→    private val storage: FirebaseStorage,
    49→    private val firebaseAuth: FirebaseAuth,
    50→    private val contactRepository: ContactRepository
    51→) : ChatRepository {
    52→
    53→    private val currentUid: String?
    54→        get() = firebaseAuth.currentUser?.uid

    56→    private val threadsCollection get() = firestore.collection("threads")

    58→    private fun messagesCollection(threadId: String) =
    59→        threadsCollection.document(threadId).collection("messages")

    62→        realtimeDb.getReference("presence").child(uid)

    65→        realtimeDb.getReference("typing").child(threadId).child(uid)

    67→    // ── Observe Threads ───────────────────────────────────────────────────────
    68→
    69→    override fun observeThreads(): Flow<List<ChatThreadModel>> = callbackFlow {
    70→        val myUid = currentUid
    71→        if (myUid == null) {
    72→            trySend(emptyList())
    73→            close()
    74→            return@callbackFlow
    75→        }
    76→
    77→        val subscription = threadsCollection
    78→            .whereArrayContains("participantUids", myUid)
    79→            .orderBy("lastMessageTimestamp", Query.Direction.DESCENDING)
    80→            .addSnapshotListener { snapshot, error ->
    81→                if (error != null) {
    82→                    Timber.e(error, "Error observing threads")
    83→                    trySend(emptyList())
    84→                    return@addSnapshotListener
    85→                }
    86→
    87→                val threads = snapshot?.documents?.mapNotNull { doc ->
    88→                    try {
    89→                        mapToChatThreadModel(doc)
    90→                    } catch (e: Exception) {
    91→                        Timber.e(e, "Failed to map thread document: %s", doc.id)
    92→                        null
    93→                    }
    94→                } ?: emptyList()
    95→
    96→                trySend(threads)
    97→            }
    98→
    99→        awaitClose { subscription.remove() }
   100→    }.flowOn(Dispatchers.IO)
   101→
   102→    override fun observeThread(threadId: String): Flow<ChatThreadModel?> = callbackFlow {
   103→        val subscription = threadsCollection.document(threadId)
   104→            .addSnapshotListener { snapshot, error ->
   105→                if (error != null) {
   106→                    Timber.e(error, "Error observing thread: %s", threadId)
   107→                    trySend(null)
   108→                    return@addSnapshotListener
   109→                }
   110→
   111→                trySend(
   112→                    if (snapshot != null && snapshot.exists()) {
   113→                        try {
   114→                            mapToChatThreadModel(snapshot)
   115→                        } catch (e: Exception) {
   116→                            Timber.e(e, "Failed to map thread: %s", threadId)
   117→                            null
   118→                        }
   119→                    } else {
   120→                        null
   121→                    }
   122→                )
   123→            }
   124→
   125→        awaitClose { subscription.remove() }
   126→    }.flowOn(Dispatchers.IO)

   130→    override fun observeMessages(threadId: String): Flow<List<MessageModel>> = callbackFlow {
   131→        val subscription = messagesCollection(threadId)
   132→            .orderBy("timestamp", Query.Direction.ASCENDING)
   133→            .addSnapshotListener { snapshot, error ->
   134→                if (error != null) {
   135→                    Timber.e(error, "Error observing messages in thread: %s", threadId)
   136→                    trySend(emptyList())
   137→                    return@addSnapshotListener
   138→                }
   139→
   140→                val messages = snapshot?.documents?.mapNotNull { doc ->
   141→                    try {
   142→                        mapToMessageModel(doc)
   143→                    } catch (e: Exception) {
   144→                        Timber.e(e, "Failed to map message document: %s", doc.id)
   145→                        null
   146→                    }
   147→                } ?: emptyList()
   148→
   149→                trySend(messages)
   150→            }
   151→
   152→        awaitClose { subscription.remove() }
   153→    }.flowOn(Dispatchers.IO)

   155→    // ── Send Text Message ─────────────────────────────────────────────────────
   156→
   157→    override fun sendTextMessage(
   158→        threadId: String,
   159→        content: String,
   160→        replyToMessageId: String?
   161→    ): Flow<Result<MessageModel>> = flow {
   162→        val myUid = currentUid
   163→        if (myUid == null) {
   164→            emit(Result.failure(IllegalStateException("Not authenticated")))
   165→            return@flow
   166→        }
   167→
   168→        try {
   169→            val threadDoc = threadsCollection.document(threadId).get().await()
   170→            if (!threadDoc.exists()) {
   171→                emit(Result.failure(IllegalStateException("Thread not found")))
   172→                return@flow
   173→            }
   174→
   175→            @Suppress("UNCHECKED_CAST")
   176→            val participantUids = (threadDoc.get("participantUids") as? List<String>)
   177→                ?: emptyList()
   178→
   179→            for (uid in participantUids) {
   180→                if (uid == myUid) continue
   181→                val gate = contactRepository.checkCommunicationGate(uid)
   182→                if (gate is CommunicationGate.Blocked || gate is CommunicationGate.Error) {
   183→                    emit(Result.failure(
   184→                        SecurityException("Communication blocked for participant: $uid")
   185→                    ))
   186→                    return@flow
   187→                }
   188→            }
   189→
   190→            val myProfile = firestore.collection("users").document(myUid).get().await()
   191→            val now = System.currentTimeMillis()
   192→            val messageId = UUID.randomUUID().toString()
   193→
   194→            val messageData = hashMapOf(
   195→                "id" to messageId,
   196→                "threadId" to threadId,
   197→                "senderUid" to myUid,
   198→                "senderDisplayName" to (myProfile.getString("displayName") ?: ""),
   199→                "senderAvatarUrl" to (myProfile.getString("photoUrl") ?: ""),
   200→                "content" to content,
   201→                "timestamp" to now,
   202→                "type" to MessageContentType.TEXT.name,
   203→                "mediaUrl" to null,
   204→                "mediaThumbnailUrl" to null,
   205→                "mediaFileSize" to 0L,
   206→                "mediaMimeType" to "",
   207→                "isRead" to false,
   208→                "readByUids" to listOf(myUid),
   209→                "deliveredToUids" to listOf(myUid),
   210→                "replyToMessageId" to replyToMessageId,
   211→                "replyToPreview" to null,
   212→                "replyToSenderName" to null,
   213→                "forwardedFromUid" to null,
   214→                "forwardedFromName" to null,
   215→                "isForwarded" to false,
   216→                "reactions" to emptyList<Map<String, Any>>(),
   217→                "isDeletedForMe" to false,
   218→                "isDeletedForEveryone" to false,
   219→                "isEdited" to false,
   220→                "editedAt" to null,
   221→                "ephemeralExpiresAt" to null,
   222→                "caption" to null
   223→            )
   224→
   225→            val batch = firestore.batch()
   226→            batch.set(messagesCollection(threadId).document(messageId), messageData)
   227→
   228→            val lastMessage = LastMessageInfo(
   229→                senderUid = myUid,
   230→                senderDisplayName = myProfile.getString("displayName") ?: "",
   231→                content = content,
   232→                type = MessageContentType.TEXT,
   233→                timestamp = now,
   234→                isRead = false
   235→            )
   236→            batch.update(
   237→                threadsCollection.document(threadId),
   238→                mapOf(
   239→                    "lastMessage" to lastMessage.toFirestoreMap(),
   240→                    "lastMessageTimestamp" to now
   241→                )
   242→            )
   243→            batch.commit().await()
   244→
   245→            val message = MessageModel(
   246→                id = messageId,
   247→                threadId = threadId,
   248→                senderUid = myUid,
   249→                senderDisplayName = myProfile.getString("displayName") ?: "",
   250→                senderAvatarUrl = myProfile.getString("photoUrl") ?: "",
   251→                content = content,
   252→                timestamp = now,
   253→                type = MessageContentType.TEXT,
   254→                replyToMessageId = replyToMessageId
   255→            )
   256→
   257→            emit(Result.success(message))
   258→        } catch (e: Exception) {
   259→            Timber.e(e, "Failed to send text message")
   260→            emit(Result.failure(e))
   261→        }
   262→    }.flowOn(Dispatchers.IO)

   394→    // ── Forward Message ───────────────────────────────────────────────────────
   395→
   396→    override fun forwardMessage(
   397→        messageId: String,
   398→        targetThreadIds: List<String>
   399→    ): Flow<Result<Unit>> = flow {
   400→        val myUid = currentUid
   401→        if (myUid == null) {
   402→            emit(Result.failure(IllegalStateException("Not authenticated")))
   403→            return@flow
   404→        }
   405→
   406→        try {
   407→            var sourceMessage: MessageModel? = null
   408→            for (threadId in threadsCollection.get().await().documents) {
   409→                val msgDoc = messagesCollection(threadId.id).document(messageId).get().await()
   410→                if (msgDoc.exists()) {
   411→                    sourceMessage = mapToMessageModel(msgDoc)
   412→                    break
   413→                }
   414→            }
   415→
   416→            if (sourceMessage == null) {
   417→                emit(Result.failure(IllegalStateException("Source message not found")))
   418→                return@flow
   419→            }
   420→
   421→            for (targetThreadId in targetThreadIds) {
   422→                val threadDoc = threadsCollection.document(targetThreadId).get().await()
   423→                if (!threadDoc.exists()) continue
   424→
   425→                @Suppress("UNCHECKED_CAST")
   426→                val participantUids = (threadDoc.get("participantUids") as? List<String>)
   427→                    ?: emptyList()
   428→
   429→                var blocked = false
   430→                for (uid in participantUids) {
   431→                    if (uid == myUid) continue
   432→                    val gate = contactRepository.checkCommunicationGate(uid)
   433→                    if (gate is CommunicationGate.Blocked || gate is CommunicationGate.Error) {
   434→                        blocked = true
   435→                        break
   436→                    }
   437→                }
   438→                if (blocked) continue
   439→
   440→                val myProfile = firestore.collection("users").document(myUid).get().await()
   441→                val now = System.currentTimeMillis()
   442→                val newMessageId = UUID.randomUUID().toString()
   443→
   444→                val messageData = hashMapOf(
   445→                    "id" to newMessageId,
   446→                    "threadId" to targetThreadId,
   447→                    "senderUid" to myUid,
   448→                    "senderDisplayName" to (myProfile.getString("displayName") ?: ""),
   449→                    "senderAvatarUrl" to (myProfile.getString("photoUrl") ?: ""),
   450→                    "content" to sourceMessage.content,
   451→                    "timestamp" to now,
   452→                    "type" to sourceMessage.type.name,
   453→                    "mediaUrl" to sourceMessage.mediaUrl,
   454→                    "forwardedFromUid" to sourceMessage.senderUid,
   455→                    "forwardedFromName" to sourceMessage.senderDisplayName,
   456→                    "isForwarded" to true,
   457→                    "isRead" to false,
   458→                    "readByUids" to listOf(myUid),
   459→                    "deliveredToUids" to listOf(myUid),
   460→                    "reactions" to emptyList<Map<String, Any>>(),
   461→                    "isDeletedForMe" to false,
   462→                    "isDeletedForEveryone" to false,
   463→                    "isEdited" to false,
   464→                    "caption" to sourceMessage.caption
   465→                )
   466→
   467→                messagesCollection(targetThreadId).document(newMessageId)
   468→                    .set(messageData).await()
   469→            }
   470→
   471→            emit(Result.success(Unit))
   472→        } catch (e: Exception) {
   473→            Timber.e(e, "Failed to forward message")
   474→            emit(Result.failure(e))
   475→        }
   476→    }.flowOn(Dispatchers.IO)

   513→    override suspend fun addReaction(
   514→        threadId: String,
   515→        messageId: String,
   516→        emoji: String,
   517→        isThreeD: Boolean
   518→    ) {
   519→        try {
   520→            val myUid = currentUid ?: return

   526→                ?: emptyList()
   527→
   528→            val filtered = existingReactions.filterNot { it["uid"] == myUid }
   529→            val newReaction = mapOf(
   530→                "uid" to myUid,
   531→                "emoji" to emoji,
   532→                "customStickerId" to null,
   533→                "timestamp" to System.currentTimeMillis(),
   534→                "isThreeD" to isThreeD
   535→            )
   536→
   537→            val updatedReactions = filtered + newReaction
   538→            messagesCollection(threadId).document(messageId)
   539→                .update("reactions", updatedReactions)
   540→                .await()
   541→
   542→            Timber.d("Reaction added: %s to message: %s", emoji, messageId)
   543→        } catch (e: Exception) {
   544→            Timber.e(e, "Failed to add reaction")
   545→        }

   562→    // ── Create Thread ─────────────────────────────────────────────────────────
   563→
   564→    override fun createSingleThread(contactUid: String): Flow<Result<ChatThreadModel>> = flow {
   565→        val myUid = currentUid
   566→        if (myUid == null) {
   567→            emit(Result.failure(IllegalStateException("Not authenticated")))
   568→            return@flow
   569→        }
   570→
   571→        try {
   572→            val gate = contactRepository.checkCommunicationGate(contactUid)
   573→            if (gate is CommunicationGate.Blocked || gate is CommunicationGate.Error) {
   574→                emit(Result.failure(
   575→                    SecurityException("Cannot create thread: not a mutual contact")
   576→                ))
   577→                return@flow
   578→            }
   579→
   580→            val existingThread = threadsCollection
   581→                .whereArrayContains("participantUids", myUid)
   582→                .get().await()
   583→
   584→            for (doc in existingThread.documents) {
   585→                @Suppress("UNCHECKED_CAST")
   586→                val uids = (doc.get("participantUids") as? List<String>) ?: emptyList()
   587→                val typeStr = doc.getString("type") ?: ""
   588→                if (uids.contains(contactUid) && typeStr == ThreadType.SINGLE.name) {
   589→                    val thread = mapToChatThreadModel(doc)
   590→                    emit(Result.success(thread))
   591→                    return@flow
   592→                }
   593→            }
   594→

   596→            val contactProfile = firestore.collection("users")

   619→            val threadData = hashMapOf(
   620→                "id" to threadId,

   622→                "participantUids" to listOf(myUid, contactUid),
   623→                "participantProfiles" to mapOf(
   624→                    myUid to myParticipant.toFirestoreMap(),
   625→                    contactUid to contactParticipant.toFirestoreMap()
   626→                ),
   627→                "groupName" to null,

   629→                "groupDescription" to null,

   631→                "createdByUid" to myUid,
   632→                "createdAt" to now,
   633→                "lastMessage" to null,

   638→                "isArchived" to false,

   653→                createdByUid = myUid,

   655→            )
   656→
   657→            emit(Result.success(thread))
   658→        } catch (e: Exception) {
   659→            Timber.e(e, "Failed to create single thread")
   660→            emit(Result.failure(e))
   661→        }

   664→    override fun createGroupThread(
   665→        groupName: String,
   666→        participantUids: List<String>
   667→    ): Flow<Result<ChatThreadModel>> = flow {
   668→        val myUid = currentUid
   669→        if (myUid == null) {
   670→            emit(Result.failure(IllegalStateException("Not authenticated")))
   671→            return@flow
   672→        }
   673→
   674→        try {
   675→            val verifiedUids = mutableListOf(myUid)
   676→            for (uid in participantUids) {
   677→                if (uid == myUid) continue
   678→                val gate = contactRepository.checkCommunicationGate(uid)
   679→                if (gate is CommunicationGate.Allowed) {
   680→                    verifiedUids.add(uid)
   681→                }
   682→            }
   683→
   684→            val now = System.currentTimeMillis()
   685→            val threadId = UUID.randomUUID().toString()
   686→
   687→            val participantProfiles = mutableMapOf<String, Map<String, Any?>>()
   688→            val domainParticipants = mutableMapOf<String, ThreadParticipant>()
   689→
   690→            for (uid in verifiedUids) {
   691→                val profile = firestore.collection("users").document(uid).get().await()
   692→                val participant = ThreadParticipant(
   693→                    uid = uid,
   694→                    displayName = profile.getString("displayName") ?: "",
   695→                    avatarUrl = profile.getString("photoUrl") ?: "",
   696→                    zixoNumber = profile.getString("zixoNumber") ?: "",
   697→                    role = if (uid == myUid) ParticipantRole.ADMIN else ParticipantRole.MEMBER,
   698→                    joinedAt = now
   699→                )
   700→                participantProfiles[uid] = participant.toFirestoreMap()
   701→                domainParticipants[uid] = participant
   702→            }
   703→
   704→            val threadData = hashMapOf(
   705→                "id" to threadId,
   706→                "type" to ThreadType.GROUP.name,
   707→                "participantUids" to verifiedUids,
   708→                "participantProfiles" to participantProfiles,
   709→                "groupName" to groupName,
   710→                "groupAvatarUrl" to null,
   711→                "groupDescription" to null,
   712→                "groupAdminUids" to listOf(myUid),
   713→                "createdByUid" to myUid,
   714→                "createdAt" to now,
   715→                "lastMessage" to null,
   716→                "lastMessageTimestamp" to now,
   717→                "unreadCount" to verifiedUids.associateWith { 0 },
   718→                "isPinned" to false,
   719→                "isMuted" to false,
   720→                "isArchived" to false,
   721→                "ephemeralTimerSeconds" to 0,
   722→                "wallpaperUrl" to null
   723→            )
   724→
   725→            threadsCollection.document(threadId).set(threadData).await()
   726→
   727→            val thread = ChatThreadModel(
   728→                id = threadId,
   729→                type = ThreadType.GROUP,
   730→                participantUids = verifiedUids.toSet(),
   731→                participantProfiles = domainParticipants,
   732→                groupName = groupName,
   733→                groupAdminUids = setOf(myUid),
   734→                createdByUid = myUid,
   735→                createdAt = now
   736→            )
   737→
   738→            emit(Result.success(thread))
   739→        } catch (e: Exception) {
   740→            Timber.e(e, "Failed to create group thread")
   741→            emit(Result.failure(e))
   742→        }
   743→    }.flowOn(Dispatchers.IO)

   781→
   782→    override suspend fun handleActionResult(action: MessageActionResult) {
   783→        try {
   784→            when (action) {
   785→                is MessageActionResult.DeleteForMe -> {
   786→                    val threadId = findThreadForMessage(action.messageId) ?: return
   787→                    deleteMessageForMe(threadId, action.messageId)
   788→                }
   789→                is MessageActionResult.DeleteForEveryone -> {
   790→                    val threadId = findThreadForMessage(action.messageId) ?: return
   791→                    deleteMessageForEveryone(threadId, action.messageId)
   792→                }
   793→                is MessageActionResult.React -> {
   794→                    val threadId = findThreadForMessage(action.messageId) ?: return
   795→                    addReaction(threadId, action.messageId, action.emoji, action.isThreeD)
   796→                }
   797→                is MessageActionResult.Forward ->
   798→                    forwardMessage(action.messageId, action.targetThreadIds)
   799→                else -> Timber.d("Unhandled action result: %s", action::class.simpleName)
   800→            }
   801→        } catch (e: Exception) {
   802→            Timber.e(e, "Failed to handle action result")
   803→        }
   804→    }

   806→    private suspend fun findThreadForMessage(messageId: String): String? {
   807→        return try {
   808→            val myUid = currentUid ?: return null
   809→            val threadSnapshot = threadsCollection
   810→                .whereArrayContains("participantUids", myUid)
   811→                .get().await()
   812→
   813→            for (threadDoc in threadSnapshot.documents) {
   814→                val msgDoc = messagesCollection(threadDoc.id)
   815→                    .document(messageId).get().await()
   816→                if (msgDoc.exists()) {
   817→                    return threadDoc.id
   818→                }
   819→            }
   820→            null
   821→        } catch (e: Exception) {
   822→            Timber.e(e, "Failed to find thread for message: %s", messageId)
   823→            null
   824→        }
   825→    }

   829→    private fun mapToChatThreadModel(
   830→        doc: com.google.firebase.firestore.DocumentSnapshot
   831→    ): ChatThreadModel {
   832→        @Suppress("UNCHECKED_CAST")
   833→        val participantUids = (doc.get("participantUids") as? List<String>)
   834→            ?.toSet() ?: emptySet()
   835→
   836→        @Suppress("UNCHECKED_CAST")
   837→        val profilesRaw = doc.get("participantProfiles") as? Map<String, Map<String, Any>>
   838→            ?: emptyMap()
   839→
   840→        val participantProfiles = profilesRaw.mapValues { (_, data) ->
   841→            ThreadParticipant(
   842→                uid = data["uid"] as? String ?: "",
   843→                displayName = data["displayName"] as? String ?: "",
   844→                avatarUrl = data["avatarUrl"] as? String ?: "",
   845→                zixoNumber = data["zixoNumber"] as? String ?: "",
   846→                role = try {
   847→                    ParticipantRole.valueOf(data["role"] as? String ?: "MEMBER")
   848→                } catch (_: Exception) {
   849→                    ParticipantRole.MEMBER
   850→                },
   851→                joinedAt = data["joinedAt"] as? Long ?: 0L,
   852→                isOnline = data["isOnline"] as? Boolean ?: false
   853→            )
   854→        }
   855→
   856→        val lastMessageRaw = doc.get("lastMessage") as? Map<String, Any>
   857→        val lastMessage = lastMessageRaw?.let { data ->
   858→            LastMessageInfo(
   859→                senderUid = data["senderUid"] as? String ?: "",
   860→                senderDisplayName = data["senderDisplayName"] as? String ?: "",
   861→                content = data["content"] as? String ?: "",
   862→                type = try {
   863→                    MessageContentType.valueOf(data["type"] as? String ?: "TEXT")
   864→                } catch (_: Exception) {
   865→                    MessageContentType.TEXT
   866→                },
   867→                timestamp = data["timestamp"] as? Long ?: 0L,
   868→                isRead = data["isRead"] as? Boolean ?: false
   869→            )
   870→        }
   871→
   872→        val typeStr = doc.getString("type") ?: "SINGLE"
   873→        val threadType = try {
   874→            ThreadType.valueOf(typeStr)
   875→        } catch (_: Exception) {
   876→            ThreadType.SINGLE
   877→        }
   878→
   879→        @Suppress("UNCHECKED_CAST")
   880→        val adminUids = (doc.get("groupAdminUids") as? List<String>)
   881→            ?.toSet() ?: emptySet()
   882→
   883→        return ChatThreadModel(
   884→            id = doc.id,
   885→            type = threadType,
   886→            participantUids = participantUids,
   887→            participantProfiles = participantProfiles,
   888→            groupName = doc.getString("groupName"),
   889→            groupAvatarUrl = doc.getString("groupAvatarUrl"),
   890→            groupDescription = doc.getString("groupDescription"),
   891→            groupAdminUids = adminUids,
   892→            createdByUid = doc.getString("createdByUid") ?: "",
   893→            createdAt = doc.getLong("createdAt") ?: 0L,
   894→            lastMessage = lastMessage,
   895→            isPinned = doc.getBoolean("isPinned") ?: false,
   896→            isMuted = doc.getBoolean("isMuted") ?: false,
   897→            isArchived = doc.getBoolean("isArchived") ?: false,
   898→            ephemeralTimerSeconds = (doc.getLong("ephemeralTimerSeconds") ?: 0L).toInt(),
   899→            wallpaperUrl = doc.getString("wallpaperUrl")
   900→        )
   901→    }
