   170→                text = "Status",
   171→                color = TextPrimary,
   172→                fontSize = 24.sp,

   180→                isUploading = isUploading,
   181→                uploadProgress = uploadProgress,

   183→            )
   184→
   185→            Spacer(modifier = Modifier.height(12.dp))
   186→
   187→            // ── Recent Updates Header ──────────────

   301→            onClose = viewModel::stopViewing,
   302→            onReaction = { statusId, emoji -> viewModel.addReaction(statusId, emoji) }
   303→        )
   304→    }
   305→}

   455→// ──────────────────────────────────────────────
   456→// Status Ring Avatar
   457→// ──────────────────────────────────────────────
   458→
   459→@Composable
   460→private fun StatusRingAvatar(
   461→    avatarUrl: String,
   462→    name: String,
   463→    size: Dp,
   464→    segmentCount: Int,
   465→    ringColor: Color
   466→) {
   467→    val strokeWidth = 3.dp
   468→    val gapAngle = 6f
   469→    val totalGap = gapAngle * segmentCount
   470→    val segmentSweep = if (segmentCount > 0) (360f - totalGap) / segmentCount else 0f
   471→
   472→    Box(contentAlignment = Alignment.Center) {
   473→        // Colored ring segments
   474→        if (segmentCount > 0) {
   475→            Canvas(modifier = Modifier.size(size + strokeWidth * 2)) {
   476→                var startAngle = -90f
   477→                for (i in 0 until segmentCount) {
   478→                    drawArc(
   479→                        color = ringColor,
   480→                        startAngle = startAngle,
   481→                        sweepAngle = segmentSweep,
   482→                        useCenter = false,
   483→                        style = Stroke(
   484→                            width = strokeWidth.toPx(),
   485→                            cap = StrokeCap.Round
   486→                        )
   487→                    )
   488→                    startAngle += segmentSweep + gapAngle
   489→                }
   490→            }
   491→        }
   492→
   493→        AvatarComponent(
   494→            imageUrl = avatarUrl,
   495→            name = name,
   496→            size = size
   497→        )
   498→    }
   499→}

   984→@Composable

   986→    group: StatusGroupModel,
   987→    currentIndex: Int,
   988→    replyText: String,
   989→    onReplyTextChange: (String) -> Unit,
   990→    onPrevious: () -> Unit,
   991→    onNext: () -> Unit,
   992→    onClose: () -> Unit,
   993→    onReaction: (statusId: String, emoji: String) -> Unit
   994→) {
   995→    val currentStatus = group.statuses.getOrNull(currentIndex) ?: return
   996→
   997→    // Auto-advance timer
   998→    LaunchedEffect(currentIndex, currentStatus.id) {
   999→        val duration = if (currentStatus.type == StatusContentType.VIDEO) {
  1000→            30_000L // Longer for video; real duration would come from media metadata
  1001→        } else {
  1002→            STATUS_AUTO_ADVANCE_MS
  1003→        }
  1004→        delay(duration)
  1005→        onNext()
  1006→    }

  1010→            .fillMaxSize()
  1011→            .background(Color.Black)

  1016→                .fillMaxWidth()

  1025→                        .weight(1f)

  1267→    val now = System.currentTimeMillis()
  1268→    val diff = now - timestampMs
  1269→    val seconds = diff / 1000
  1270→    val minutes = seconds / 60
  1271→    val hours = minutes / 60
  1272→    return when {
  1273→        seconds < 60 -> "just now"
  1274→        minutes < 60 -> "${minutes}m ago"
  1275→        hours < 24 -> "${hours}h ago"
  1276→        else -> "${hours / 24}d ago"

  1278→}
