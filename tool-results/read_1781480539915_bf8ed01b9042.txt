    33→    33→/**
    34→    34→ * Concrete implementation of [ChatRepository].
    35→    35→ *
    36→    36→ * Manages chat threads and messages using Firestore with continuous
    37→    37→ * snapshot listeners for real-time sync. Uses Firebase Realtime Database
    38→    38→ * for presence and typing indicators, and Firebase Storage for media uploads.
    39→    39→ *
    40→    40→ * Zero-trust enforcement: All send operations verify mutual contact status
    41→    41→ * through [ContactRepository.checkCommunicationGate] before writing.
    42→    42→ * Non-contact messages are blocked at this boundary.
    43→    43→ */
    44→    44→@Singleton

    46→    46→    private val firestore: FirebaseFirestore,
    47→    47→    private val realtimeDb: FirebaseDatabase,

    67→    67→    // ── Observe Threads ───────────────────────────────────────────────────────
    68→    68→
    69→    69→    override fun observeThreads(): Flow<List<ChatThreadModel>> = callbackFlow {
    70→    70→        val myUid = currentUid
    71→    71→        if (myUid == null) {
    72→    72→            trySend(emptyList())
    73→    73→            close()
    74→    74→            return@callbackFlow
    75→    75→        }
    76→    76→
    77→    77→        val subscription = threadsCollection
    78→    78→            .whereArrayContains("participantUids", myUid)
    79→    79→            .orderBy("lastMessageTimestamp", Query.Direction.DESCENDING)
    80→    80→            .addSnapshotListener { snapshot, error ->
    81→    81→                if (error != null) {
    82→    82→                    Timber.e(error, "Error observing threads")
    83→    83→                    trySend(emptyList())
    84→    84→                    return@addSnapshotListener
    85→    85→                }
    86→    86→
    87→    87→                val threads = snapshot?.documents?.mapNotNull { doc ->
    88→    88→                    try {
    89→    89→                        mapToChatThreadModel(doc)
    90→    90→                    } catch (e: Exception) {
    91→    91→                        Timber.e(e, "Failed to map thread document: %s", doc.id)
    92→    92→                        null
    93→    93→                    }
    94→    94→                } ?: emptyList()
    95→    95→
    96→    96→                trySend(threads)
    97→    97→            }
    98→    98→
    99→    99→        awaitClose { subscription.remove() }
   100→   100→    }.flowOn(Dispatchers.IO)
   101→   101→
   102→   102→    override fun observeThread(threadId: String): Flow<ChatThreadModel?> = callbackFlow {
   103→   103→        val subscription = threadsCollection.document(threadId)
   104→   104→            .addSnapshotListener { snapshot, error ->
   105→   105→                if (error != null) {
   106→   106→                    Timber.e(error, "Error observing thread: %s", threadId)
   107→   107→                    trySend(null)
   108→   108→                    return@addSnapshotListener
   109→   109→                }
   110→   110→
   111→   111→                trySend(
   112→   112→                    if (snapshot != null && snapshot.exists()) {
   113→   113→                        try {
   114→   114→                            mapToChatThreadModel(snapshot)
   115→   115→                        } catch (e: Exception) {
   116→   116→                            Timber.e(e, "Failed to map thread: %s", threadId)
   117→   117→                            null
   118→   118→                        }
   119→   119→                    } else {
   120→   120→                        null
   121→   121→                    }
   122→   122→                )
   123→   123→            }
   124→   124→
   125→   125→        awaitClose { subscription.remove() }
   126→   126→    }.flowOn(Dispatchers.IO)

   130→   130→    override fun observeMessages(threadId: String): Flow<List<MessageModel>> = callbackFlow {

   136→   136→                    trySend(emptyList())

   138→   138→                }
   139→   139→
   140→   140→                val messages = snapshot?.documents?.mapNotNull { doc ->
   141→   141→                    try {
   142→   142→                        mapToMessageModel(doc)
   143→   143→                    } catch (e: Exception) {
   144→   144→                        Timber.e(e, "Failed to map message document: %s", doc.id)
   145→   145→                        null
   146→   146→                    }
   147→   147→                } ?: emptyList()
   148→   148→
   149→   149→                trySend(messages)
   150→   150→            }
   151→   151→
   152→   152→        awaitClose { subscription.remove() }
   153→   153→    }.flowOn(Dispatchers.IO)

   164→   164→            emit(Result.failure(IllegalStateException("Not authenticated")))
   165→   165→            return@flow
   166→   166→        }

   168→   168→        try {
   169→   169→            val threadDoc = threadsCollection.document(threadId).get().await()
   170→   170→            if (!threadDoc.exists()) {
   171→   171→                emit(Result.failure(IllegalStateException("Thread not found")))
   172→   172→                return@flow
   173→   173→            }
   174→   174→
   175→   175→            @Suppress("UNCHECKED_CAST")
   176→   176→            val participantUids = (threadDoc.get("participantUids") as? List<String>)
   177→   177→                ?: emptyList()
   178→   178→
   179→   179→            for (uid in participantUids) {
   180→   180→                if (uid == myUid) continue
   181→   181→                val gate = contactRepository.checkCommunicationGate(uid)
   182→   182→                if (gate is CommunicationGate.Blocked || gate is CommunicationGate.Error) {
   183→   183→                    emit(Result.failure(
   184→   184→                        SecurityException("Communication blocked for participant: $uid")
   185→   185→                    ))
   186→   186→                    return@flow
   187→   187→                }
   188→   188→            }
   189→   189→
   190→   190→            val myProfile = firestore.collection("users").document(myUid).get().await()
   191→   191→            val now = System.currentTimeMillis()
   192→   192→            val messageId = UUID.randomUUID().toString()
   193→   193→
   194→   194→            val messageData = hashMapOf(
   195→   195→                "id" to messageId,
   196→   196→                "threadId" to threadId,
   197→   197→                "senderUid" to myUid,
   198→   198→                "senderDisplayName" to (myProfile.getString("displayName") ?: ""),
   199→   199→                "senderAvatarUrl" to (myProfile.getString("photoUrl") ?: ""),
   200→   200→                "content" to content,

   205→   205→                "mediaFileSize" to 0L,
   206→   206→                "mediaMimeType" to "",
   207→   207→                "isRead" to false,

   209→   209→                "deliveredToUids" to listOf(myUid),
   210→   210→                "replyToMessageId" to replyToMessageId,
   211→   211→                "replyToPreview" to null,
   212→   212→                "replyToSenderName" to null,
   213→   213→                "forwardedFromUid" to null,
   214→   214→                "forwardedFromName" to null,
   215→   215→                "isForwarded" to false,
   216→   216→                "reactions" to emptyList<Map<String, Any>>(),
   217→   217→                "isDeletedForMe" to false,
   218→   218→                "isDeletedForEveryone" to false,
   219→   219→                "isEdited" to false,
   220→   220→                "editedAt" to null,
   221→   221→                "ephemeralExpiresAt" to null,
   222→   222→                "caption" to null
   223→   223→            )
   224→   224→
   225→   225→            val batch = firestore.batch()
   226→   226→            batch.set(messagesCollection(threadId).document(messageId), messageData)
   227→   227→
   228→   228→            val lastMessage = LastMessageInfo(
   229→   229→                senderUid = myUid,
   230→   230→                senderDisplayName = myProfile.getString("displayName") ?: "",
   231→   231→                content = content,
   232→   232→                type = MessageContentType.TEXT,
   233→   233→                timestamp = now,
   234→   234→                isRead = false
   235→   235→            )
   236→   236→            batch.update(
   237→   237→                threadsCollection.document(threadId),
   238→   238→                mapOf(
   239→   239→                    "lastMessage" to lastMessage.toFirestoreMap(),
   240→   240→                    "lastMessageTimestamp" to now
   241→   241→                )
   242→   242→            )
   243→   243→            batch.commit().await()
   244→   244→
   245→   245→            val message = MessageModel(
   246→   246→                id = messageId,
   247→   247→                threadId = threadId,
   248→   248→                senderUid = myUid,
   249→   249→                senderDisplayName = myProfile.getString("displayName") ?: "",
   250→   250→                senderAvatarUrl = myProfile.getString("photoUrl") ?: "",
   251→   251→                content = content,
   252→   252→                timestamp = now,
   253→   253→                type = MessageContentType.TEXT,
   254→   254→                replyToMessageId = replyToMessageId
   255→   255→            )
   256→   256→
   257→   257→            emit(Result.success(message))
   258→   258→        } catch (e: Exception) {
   259→   259→            Timber.e(e, "Failed to send text message")
   260→   260→            emit(Result.failure(e))
   261→   261→        }
   262→   262→    }.flowOn(Dispatchers.IO)

   394→   394→    // ── Forward Message ───────────────────────────────────────────────────────
   395→   395→
   396→   396→    override fun forwardMessage(
   397→   397→        messageId: String,
   398→   398→        targetThreadIds: List<String>
   399→   399→    ): Flow<Result<Unit>> = flow {
   400→   400→        val myUid = currentUid
   401→   401→        if (myUid == null) {
   402→   402→            emit(Result.failure(IllegalStateException("Not authenticated")))
   403→   403→            return@flow
   404→   404→        }
   405→   405→
   406→   406→        try {
   407→   407→            var sourceMessage: MessageModel? = null
   408→   408→            for (threadId in threadsCollection.get().await().documents) {
   409→   409→                val msgDoc = messagesCollection(threadId.id).document(messageId).get().await()
   410→   410→                if (msgDoc.exists()) {
   411→   411→                    sourceMessage = mapToMessageModel(msgDoc)
   412→   412→                    break
   413→   413→                }
   414→   414→            }
   415→   415→
   416→   416→            if (sourceMessage == null) {
   417→   417→                emit(Result.failure(IllegalStateException("Source message not found")))
   418→   418→                return@flow
   419→   419→            }
   420→   420→
   421→   421→            for (targetThreadId in targetThreadIds) {
   422→   422→                val threadDoc = threadsCollection.document(targetThreadId).get().await()
   423→   423→                if (!threadDoc.exists()) continue
   424→   424→
   425→   425→                @Suppress("UNCHECKED_CAST")
   426→   426→                val participantUids = (threadDoc.get("participantUids") as? List<String>)
   427→   427→                    ?: emptyList()
   428→   428→
   429→   429→                var blocked = false
   430→   430→                for (uid in participantUids) {
   431→   431→                    if (uid == myUid) continue
   432→   432→                    val gate = contactRepository.checkCommunicationGate(uid)
   433→   433→                    if (gate is CommunicationGate.Blocked || gate is CommunicationGate.Error) {
   434→   434→                        blocked = true
   435→   435→                        break
   436→   436→                    }
   437→   437→                }
   438→   438→                if (blocked) continue
   439→   439→
   440→   440→                val myProfile = firestore.collection("users").document(myUid).get().await()
   441→   441→                val now = System.currentTimeMillis()
   442→   442→                val newMessageId = UUID.randomUUID().toString()
   443→   443→
   444→   444→                val messageData = hashMapOf(
   445→   445→                    "id" to newMessageId,
   446→   446→                    "threadId" to targetThreadId,
   447→   447→                    "senderUid" to myUid,
   448→   448→                    "senderDisplayName" to (myProfile.getString("displayName") ?: ""),
   449→   449→                    "senderAvatarUrl" to (myProfile.getString("photoUrl") ?: ""),
   450→   450→                    "content" to sourceMessage.content,
   451→   451→                    "timestamp" to now,
   452→   452→                    "type" to sourceMessage.type.name,
   453→   453→                    "mediaUrl" to sourceMessage.mediaUrl,
   454→   454→                    "forwardedFromUid" to sourceMessage.senderUid,
   455→   455→                    "forwardedFromName" to sourceMessage.senderDisplayName,
   456→   456→                    "isForwarded" to true,
   457→   457→                    "isRead" to false,
   458→   458→                    "readByUids" to listOf(myUid),
   459→   459→                    "deliveredToUids" to listOf(myUid),
   460→   460→                    "reactions" to emptyList<Map<String, Any>>(),
   461→   461→                    "isDeletedForMe" to false,
   462→   462→                    "isDeletedForEveryone" to false,
   463→   463→                    "isEdited" to false,
   464→   464→                    "caption" to sourceMessage.caption
   465→   465→                )
   466→   466→
   467→   467→                messagesCollection(targetThreadId).document(newMessageId)
   468→   468→                    .set(messageData).await()
   469→   469→            }
   470→   470→
   471→   471→            emit(Result.success(Unit))
   472→   472→        } catch (e: Exception) {
   473→   473→            Timber.e(e, "Failed to forward message")
   474→   474→            emit(Result.failure(e))
   475→   475→        }
   476→   476→    }.flowOn(Dispatchers.IO)

   478→   478→    // ── Delete Message ────────────────────────────────────────────────────────

   480→   480→    override suspend fun deleteMessageForMe(threadId: String, messageId: String) {
   481→   481→        try {
   482→   482→            val myUid = currentUid ?: return
   483→   483→            messagesCollection(threadId).document(messageId)

   485→   485→                .await()

   781→   781→
   782→   782→    override suspend fun handleActionResult(action: MessageActionResult) {
   783→   783→        try {
   784→   784→            when (action) {
   785→   785→                is MessageActionResult.DeleteForMe -> {
   786→   786→                    val threadId = findThreadForMessage(action.messageId) ?: return
   787→   787→                    deleteMessageForMe(threadId, action.messageId)
   788→   788→                }
   789→   789→                is MessageActionResult.DeleteForEveryone -> {
   790→   790→                    val threadId = findThreadForMessage(action.messageId) ?: return
   791→   791→                    deleteMessageForEveryone(threadId, action.messageId)
   792→   792→                }
   793→   793→                is MessageActionResult.React -> {
   794→   794→                    val threadId = findThreadForMessage(action.messageId) ?: return
   795→   795→                    addReaction(threadId, action.messageId, action.emoji, action.isThreeD)
   796→   796→                }
   797→   797→                is MessageActionResult.Forward ->
   798→   798→                    forwardMessage(action.messageId, action.targetThreadIds)
   799→   799→                else -> Timber.d("Unhandled action result: %s", action::class.simpleName)
   800→   800→            }
   801→   801→        } catch (e: Exception) {
   802→   802→            Timber.e(e, "Failed to handle action result")
   803→   803→        }
   804→   804→    }
