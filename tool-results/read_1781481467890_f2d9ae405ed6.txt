   544→ * Steps when enabling:

   551→ */
   552→@OptIn(ExperimentalMaterial3Api::class)
   553→@Composable
   554→private fun TwoStepVerificationSheet(

   578→    ) {
   579→        Column(
   580→            modifier = Modifier
   581→                .fillMaxWidth()

   585→        ) {
   586→            Text(
   587→                text = "Two-Step Verification",
   588→                color = TextPrimary,
   589→                fontSize = 20.sp,
   590→                fontWeight = FontWeight.Bold
   591→            )
   592→
   593→            Spacer(modifier = Modifier.height(4.dp))
   594→
   595→            Text(
   596→                text = when (currentStep) {
   597→                    STEP_MANAGE -> "Manage your two-step verification"
   598→                    STEP_ENTER_PIN -> "Enter a 6-digit PIN"
   599→                    STEP_CONFIRM_PIN -> "Confirm your PIN"
   600→                    else -> "Add a backup email address"
   601→                },
   602→                color = TextSecondary,
   603→                fontSize = 14.sp,
   604→                textAlign = TextAlign.Center
   605→            )
   606→
   607→            Spacer(modifier = Modifier.height(24.dp))
   608→
   609→            when (currentStep) {
   610→                STEP_MANAGE -> ManageStep(onDisable = onDisable)
   611→                STEP_ENTER_PIN -> EnterPinStep(
   612→                    pin = pin,
   613→                    onPinChange = { newPin ->
   614→                        val filtered = newPin.filter { it.isDigit() }
   615→                        if (filtered.length <= 6) pin = filtered
   616→                        pinError = null
   617→                    },
   618→                    pinVisible = pinVisible,
   619→                    onToggleVisibility = { pinVisible = !pinVisible },
   620→                    pinError = pinError,
   621→                    onNext = {
   622→                        keyboardController?.hide()
   623→                        currentStep = STEP_CONFIRM_PIN
   624→                    }
   625→                )
   626→                STEP_CONFIRM_PIN -> ConfirmPinStep(
   627→                    confirmPin = confirmPin,
   628→                    onConfirmPinChange = { newPin ->
   629→                        val filtered = newPin.filter { it.isDigit() }
   630→                        if (filtered.length <= 6) confirmPin = filtered
   631→                        pinError = null
   632→                    },
   633→                    confirmPinVisible = confirmPinVisible,
   634→                    onToggleVisibility = { confirmPinVisible = !confirmPinVisible },
   635→                    pinError = pinError,
   636→                    onPinMismatch = {
   637→                        pinError = "PINs do not match. Try again."
   638→                        confirmPin = ""
   639→                    },
   640→                    onNext = {
   641→                        keyboardController?.hide()
   642→                        currentStep = STEP_EMAIL
   643→                    }
   644→                )
   645→                STEP_EMAIL -> EmailStep(
   646→                    email = backupEmail,
   647→                    onEmailChange = { backupEmail = it },
   648→                    onEnable = {
   649→                        keyboardController?.hide()
   650→                        onEnable(pin, backupEmail)
   651→                    }
   652→                )
   653→            }
   654→
   655→            Spacer(modifier = Modifier.height(24.dp))
   656→        }
   657→    }
   658→}

   819→/**
   820→ * A prominent call-to-action button rendered with the glass aesthetic.
   821→ *
   822→ * @param label       Uppercase label text displayed inside the button.
   823→ * @param enabled     Whether the button is interactive.
   824→ * @param destructive When true, renders in the destructive red palette instead of green.
   825→ * @param onClick     Callback invoked on tap.
   826→ */
   827→@Composable
   828→private fun GlassActionButton(
   829→    label: String,
   830→    enabled: Boolean,
   831→    destructive: Boolean = false,
   832→    onClick: () -> Unit
   833→) {
   834→    val activeColor = if (destructive) DestructiveText else NeonMint
   835→    val bgColor = if (enabled) activeColor else activeColor.copy(alpha = 0.3f)
   836→    val textColor = if (enabled) Color(0xFF0B1519) else Color(0xFF0B1519).copy(alpha = 0.5f)
   837→
   838→    Box(
   839→        modifier = Modifier
   840→            .fillMaxWidth()
   841→            .clip(RoundedCornerShape(12.dp))
   842→            .background(bgColor)
   843→            .then(if (enabled) Modifier.clickable(onClick = onClick) else Modifier)
   844→            .padding(vertical = 14.dp),
   845→        contentAlignment = Alignment.Center
   846→    ) {
   847→        Text(
   848→            text = label,
   849→            color = textColor,
   850→            fontSize = 14.sp,
   851→            fontWeight = FontWeight.Bold,
   852→            letterSpacing = 1.sp
   853→        )
   854→    }
   855→}
