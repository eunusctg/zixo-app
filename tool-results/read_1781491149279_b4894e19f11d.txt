   522→// ─────────────────────────────────────────────────────────────────────────────
   523→// QR Code Popup
   524→// ─────────────────────────────────────────────────────────────────────────────
   525→
   526→/**
   527→ * Frosted liquid glass modal overlay displaying the user's dynamically computed
   528→ * neon green QR bitmap centered inside an ultra-glossy frosted glass panel.
   529→ *
   530→ * The QR code is generated in real-time by [SettingsViewModel.generateRealtimeQrMatrix]
   531→ * using the ZXing library, encoding the secure URI `zixo://profile/{zixoNumber}`.
   532→ * It is rendered in brand emerald green (#00E676) for high-contrast visibility.
   533→ *
   534→ * Features:
   535→ * - Real bitmap rendered from ZXing (not a placeholder pattern)
   536→ * - Frosted glass container with liquid glass styling
   537→ * - "Copy Invite Link" action row at the baseline
   538→ * - "Share" button for system share sheet
   539→ * - Close button in the top-right corner
   540→ */
   541→@Composable
   542→private fun QrCodePopup(
   543→    displayName: String,
   544→    zixoNumber: String,
   545→    qrBitmap: Bitmap?,
   546→    inviteLink: String,
   547→    onClose: () -> Unit
   548→) {
   549→    val context = LocalContext.current
   550→    val clipboardManager = androidx.compose.ui.platform.LocalClipboardManager.current
   551→
   552→    Box(
   553→        modifier = Modifier
   554→            .fillMaxSize()
   555→            .background(Color.Black.copy(alpha = 0.6f))
   556→            .clickable(
   557→                interactionSource = remember { MutableInteractionSource() },
   558→                indication = null,
   559→                onClick = onClose
   560→            ),
   561→        contentAlignment = Alignment.Center
   562→    ) {
   563→        Column(
   564→            modifier = Modifier
   565→                .fillMaxWidth(0.85f)
   566→                .liquidGlassContainer()
   567→                .padding(24.dp)
   568→                .clickable(
   569→                    interactionSource = remember { MutableInteractionSource() },
   570→                    indication = null,
   571→                    onClick = { /* consume clicks inside popup */ }
   572→                ),
   573→            horizontalAlignment = Alignment.CenterHorizontally
   574→        ) {
   575→            // Close button row
   576→            Row(
   577→                modifier = Modifier.fillMaxWidth(),
   578→                horizontalArrangement = Arrangement.SpaceBetween,
   579→                verticalAlignment = Alignment.CenterVertically
   580→            ) {
   581→                Text(
   582→                    text = "Your QR Code",
   583→                    color = TextPrimary,
   584→                    fontSize = 18.sp,
   585→                    fontWeight = FontWeight.Bold
   586→                )
   587→                IconButton(onClick = onClose) {
   588→                    Icon(
   589→                        imageVector = Icons.Default.Close,
   590→                        contentDescription = "Close",
   591→                        tint = TextSecondary
   592→                    )
   593→                }
   594→            }
   595→
   596→            Spacer(modifier = Modifier.height(20.dp))
   597→
   598→            // QR code — Real bitmap from ZXing or fallback
   599→            Box(
   600→                modifier = Modifier
   601→                    .size(220.dp)
   602→                    .background(Color.White, RoundedCornerShape(12.dp))
   603→                    .padding(12.dp),
   604→                contentAlignment = Alignment.Center
   605→            ) {
   606→                if (qrBitmap != null) {
   607→                    Image(
   608→                        painter = BitmapPainter(qrBitmap.asImageBitmap()),
   609→                        contentDescription = "QR Code",
   610→                        modifier = Modifier
   611→                            .fillMaxSize()
   612→                            .clip(RoundedCornerShape(8.dp)),
   613→                        contentScale = ContentScale.Fit
   614→                    )
   615→                } else {
   616→                    // Fallback: placeholder while bitmap generates
   617→                    Text(
   618→                        text = "Generating…",
   619→                        color = TextSecondary,
   620→                        fontSize = 14.sp
