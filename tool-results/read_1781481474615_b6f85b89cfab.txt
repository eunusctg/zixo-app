   494→   494→private fun generateQrBitmapThumbnail(data: String, sizePx: Int): Bitmap {
   495→   495→    val writer = QRCodeWriter()
   496→   496→    val bitMatrix = writer.encode(data, BarcodeFormat.QR_CODE, sizePx, sizePx)
   497→   497→    val width = bitMatrix.width
   498→   498→    val height = bitMatrix.height
   499→   499→    val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.RGB_565)
   500→   500→    for (x in 0 until width) {
   501→   501→        for (y in 0 until height) {
   502→   502→            bitmap.setPixel(
   503→   503→                x, y,
   504→   504→                if (bitMatrix[x, y]) AndroidColor.BLACK else AndroidColor.WHITE
   505→   505→            )
   506→   506→        }
   507→   507→    }
   508→   508→    return bitmap
   509→   509→}
