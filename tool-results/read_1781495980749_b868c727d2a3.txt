   565→    title: String,
   566→    subtitle: String,
   567→    checked: Boolean,
   568→    onCheckedChange: (Boolean) -> Unit,
   569→    icon: androidx.compose.ui.graphics.vector.ImageVector? = null

   574→    ) {

   576→            Icon(
   577→                imageVector = icon,
   578→                contentDescription = null,
   579→                tint = TextSecondary,
   580→                modifier = Modifier.size(22.dp)
   581→            )
   582→            Spacer(modifier = Modifier.width(12.dp))
   583→        }
   584→        Column(modifier = Modifier.weight(1f)) {
   585→            Text(
   586→                text = title,
   587→                color = TextPrimary,
   588→                fontSize = 16.sp,
   589→                fontWeight = FontWeight.Medium
   590→            )
   591→            Text(
   592→                text = subtitle,
   593→                color = TextSecondary,
   594→                fontSize = 13.sp
   595→            )

   621→private fun resolveToneName(context: android.content.Context, uriString: String): String {
   622→    if (uriString.isBlank()) return "Default"
   623→    return try {
   624→        val uri = android.net.Uri.parse(uriString)
   625→        RingtoneManager.getRingtone(context, uri)?.getTitle(context) ?: "Default"
   626→    } catch (_: Exception) {
   627→        "Default"
   628→    }
   629→}
   630→
   631→private fun parseUriOrNull(uriString: String): android.net.Uri? {
   632→    return if (uriString.isNotBlank()) {
   633→        try { android.net.Uri.parse(uriString) } catch (_: Exception) { null }
   634→    } else {
   635→        null
   636→    }
   637→}
