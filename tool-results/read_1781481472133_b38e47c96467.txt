    91→    91→/**
    92→    92→ * Full chat message screen with keyboard avoidance and Liquid Glass design.
    93→    93→ *
    94→    94→ * Features:
    95→    95→ * - Wrapped in [Scaffold] with `Modifier.imePadding()` for keyboard avoidance
    96→    96→ * - [LazyColumn] with `reverseLayout = true` for messages — newest at bottom
    97→    97→ * - Auto-scroll on new messages via [LaunchedEffect]
    98→    98→ * - 74dp liquid glass pill-shaped input tray
    99→    99→ * - Outgoing messages: glowing neon mint transparency background
   100→   100→ * - Incoming messages: translucent dark petrol grey curves
   101→   101→ * - Long-press action menu with 3D reactions, reply, forward, delete
   102→   102→ * - Reply preview bar above input tray
   103→   103→ * - Call state overlay when active
   104→   104→ * - Top bar with contact info and call icons
   105→   105→ * - Media messages with Coil [AsyncImage] thumbnails
   106→   106→ *
   107→   107→ * @param threadId The ID of the chat thread to display.
   108→   108→ * @param onBack Callback for the back navigation button.
   109→   109→ */
   110→   110→@OptIn(ExperimentalFoundationApi::class)
   111→   111→@Composable
   112→   112→fun ChatMessageScreen(
   113→   113→    threadId: String,
   114→   114→    onBack: () -> Unit = {},
   115→   115→    viewModel: ChatViewModel = hiltViewModel()
   116→   116→) {
   117→   117→    val messages by viewModel.messages.collectAsState()
   118→   118→    val thread by viewModel.thread.collectAsState()
   119→   119→    val inputText by viewModel.inputText.collectAsState()
   120→   120→    val isSending by viewModel.isSending.collectAsState()
   121→   121→    val callState by viewModel.callState.collectAsState()
   122→   122→    val replyToMessage by viewModel.replyToMessage.collectAsState()
   123→   123→    val showActionMenu by viewModel.showActionMenu.collectAsState()
   124→   124→    val currentUserId by viewModel.currentUserId.collectAsState()
   125→   125→    val errorMessage by viewModel.errorMessage.collectAsState()
   126→   126→
   127→   127→    val snackbarHostState = remember { SnackbarHostState() }
   128→   128→    val listState = rememberLazyListState()
   129→   129→
   130→   130→    // Load thread data on first composition
   131→   131→    LaunchedEffect(threadId) {
   132→   132→        viewModel.loadThread(threadId)
   133→   133→    }
   134→   134→
   135→   135→    // Auto-scroll to bottom on new messages
   136→   136→    LaunchedEffect(messages.size) {
   137→   137→        if (messages.isNotEmpty()) {
   138→   138→            listState.animateScrollToItem(0)
   139→   139→        }
   140→   140→    }
   141→   141→
   142→   142→    // Display error messages as snackbars
   143→   143→    LaunchedEffect(errorMessage) {
   144→   144→        errorMessage?.let {
   145→   145→            snackbarHostState.showSnackbar(it)
   146→   146→            viewModel.clearError()
   147→   147→        }
   148→   148→    }
   149→   149→
   150→   150→    // Determine contact info from thread
   151→   151→    val otherParticipant = thread?.participantProfiles?.values?.firstOrNull {
   152→   152→        it.uid != currentUserId
   153→   153→    }
   154→   154→    val contactName = otherParticipant?.displayName ?: "Chat"
   155→   155→    val contactAvatar = otherParticipant?.avatarUrl ?: ""
   156→   156→    val isOnline = otherParticipant?.isOnline ?: false
   157→   157→
   158→   158→    Scaffold(
   159→   159→        snackbarHost = { SnackbarHost(snackbarHostState) },
   160→   160→        containerColor = Color.Transparent,
   161→   161→        modifier = Modifier.fillMaxSize()
   162→   162→    ) { innerPadding ->
   163→   163→        Box(modifier = Modifier.fillMaxSize()) {
   164→   164→            // Animated glass background
   165→   165→            ZixoGlassBackground()
   166→   166→
   167→   167→            Column(
   168→   168→                modifier = Modifier
   169→   169→                    .fillMaxSize()
   170→   170→                    .imePadding()
   171→   171→                    .padding(innerPadding)
   172→   172→            ) {
   173→   173→                // ── Top Bar ────────────────────────────────────────────────
   174→   174→                ChatTopBar(
   175→   175→                    contactName = contactName,
   176→   176→                    isOnline = isOnline,
   177→   177→                    onBackClick = onBack,
   178→   178→                    onAudioCallClick = { viewModel.startAudioCall() },
   179→   179→                    onVideoCallClick = { viewModel.startVideoCall() }
   180→   180→                )
   181→   181→
   182→   182→                // ── Message List ───────────────────────────────────────────
   183→   183→                LazyColumn(
   184→   184→                    state = listState,
   185→   185→                    reverseLayout = true,
   186→   186→                    modifier = Modifier
   187→   187→                        .weight(1f)
   188→   188→                        .fillMaxWidth()
   189→   189→                        .padding(horizontal = 12.dp),
   190→   190→                    verticalArrangement = Arrangement.spacedBy(6.dp)
   191→   191→                ) {
   192→   192→                    items(
   193→   193→                        count = messages.size,
   194→   194→                        key = { index -> messages[index].id }
   195→   195→                    ) { index ->
   196→   196→                        val message = messages[index]
   197→   197→                        val isOwnMessage = message.senderUid == currentUserId
   198→   198→
   199→   199→                        MessageBubble(
   200→   200→                            message = message,
   201→   201→                            isOwnMessage = isOwnMessage,
   202→   202→                            onLongClick = { viewModel.showActionMenu.value = message }
   203→   203→                        )
   204→   204→                    }
   205→   205→                }
   206→   206→
   207→   207→                // ── Reply Preview Bar ──────────────────────────────────────
   208→   208→                AnimatedVisibility(
   209→   209→                    visible = replyToMessage != null,
   210→   210→                    enter = slideInVertically() + fadeIn(),
   211→   211→                    exit = slideOutVertically() + fadeOut()
   212→   212→                ) {
   213→   213→                    replyToMessage?.let { replied ->
   214→   214→                        ReplyPreviewBar(
   215→   215→                            senderName = replied.replyToSenderName ?: replied.senderDisplayName,
   216→   216→                            previewText = replied.replyToPreview ?: replied.content,
   217→   217→                            onDismiss = { viewModel.clearReply() }
   218→   218→                        )
   219→   219→                    }
   220→   220→                }
   221→   221→
   222→   222→                // ── Input Tray (74dp Liquid Glass Pill) ────────────────────
   223→   223→                InputTray(
   224→   224→                    value = inputText,
   225→   225→                    onValueChange = { viewModel.inputText.value = it },
   226→   226→                    onSendClick = { viewModel.sendMessage() },
   227→   227→                    isSending = isSending,
   228→   228→                    modifier = Modifier.fillMaxWidth()
   229→   229→                )
   230→   230→            }
   231→   231→
   232→   232→            // ── Action Menu Overlay ────────────────────────────────────────
   233→   233→            showActionMenu?.let { actionMessage ->
   234→   234→                ActionMenuOverlay(
   235→   235→                    message = actionMessage,
   236→   236→                    isOwnMessage = actionMessage.senderUid == currentUserId,
   237→   237→                    onReact = { emoji, isThreeD ->
   238→   238→                        viewModel.addReaction(actionMessage.id, emoji, isThreeD)
   239→   239→                    },
   240→   240→                    onReply = {
   241→   241→                        viewModel.replyToMessage.value = actionMessage
   242→   242→                        viewModel.dismissActionMenu()
   243→   243→                    },
   244→   244→                    onForward = {
   245→   245→                        viewModel.forwardMessage(actionMessage.id, emptyList())
   246→   246→                    },
   247→   247→                    onDeleteForMe = {
   248→   248→                        viewModel.deleteForMe(actionMessage.id)
   249→   249→                    },
   250→   250→                    onDeleteForEveryone = {
   251→   251→                        viewModel.deleteForEveryone(actionMessage.id)
   252→   252→                    },
   253→   253→                    onDismiss = { viewModel.dismissActionMenu() }
   254→   254→                )
   255→   255→            }
   256→   256→
   257→   257→            // ── Call State Overlay ─────────────────────────────────────────
   258→   258→            if (callState !is CallState.IDLE) {
   259→   259→                CallStateOverlay(
   260→   260→                    callState = callState,
   261→   261→                    contactName = contactName,
   262→   262→                    contactAvatar = contactAvatar
   263→   263→                )
   264→   264→            }
   265→   265→        }
   266→   266→    }
   267→   267→}

   331→   331→@OptIn(ExperimentalFoundationApi::class)
   332→   332→@Composable
   333→   333→private fun MessageBubble(
   334→   334→    message: MessageModel,
   335→   335→    isOwnMessage: Boolean,
   336→   336→    onLongClick: () -> Unit
   337→   337→) {
   338→   338→    // Deleted placeholder
   339→   339→    if (message.isDeletedForEveryone) {
   340→   340→        Row(
   341→   341→            modifier = Modifier
   342→   342→                .fillMaxWidth()
   343→   343→                .padding(
   344→   344→                    start = if (isOwnMessage) 48.dp else 0.dp,
   345→   345→                    end = if (!isOwnMessage) 48.dp else 0.dp
   346→   346→                ),
   347→   347→            horizontalArrangement = if (isOwnMessage) Arrangement.End else Arrangement.Start
   348→   348→        ) {
   349→   349→            Box(
   350→   350→                modifier = Modifier
   351→   351→                    .clip(RoundedCornerShape(BubbleCornerSize))
   352→   352→                    .background(IncomingBubbleColor)
   353→   353→                    .padding(horizontal = 14.dp, vertical = 10.dp)
   354→   354→            ) {
   355→   355→                Text(
   356→   356→                    text = "This message was deleted",
   357→   357→                    color = TextSecondary,
   358→   358→                    fontSize = 14.sp,
   359→   359→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   360→   360→                )
   361→   361→            }
   362→   362→        }
   363→   363→        return
   364→   364→    }
   365→   365→
   366→   366→    // Soft-deleted for me — invisible
   367→   367→    if (message.isDeletedForMe) return
   368→   368→
   369→   369→    val bubbleColor = if (isOwnMessage) NeonMintAlpha15 else IncomingBubbleColor
   370→   370→    val alignment = if (isOwnMessage) Arrangement.End else Arrangement.Start
   371→   371→    val shape = RoundedCornerShape(
   372→   372→        topStart = BubbleCornerSize,
   373→   373→        topEnd = BubbleCornerSize,
   374→   374→        bottomStart = if (isOwnMessage) BubbleCornerSize else 4.dp,
   375→   375→        bottomEnd = if (isOwnMessage) 4.dp else BubbleCornerSize
   376→   376→    )
   377→   377→
   378→   378→    Row(
   379→   379→        modifier = Modifier
   380→   380→            .fillMaxWidth()
   381→   381→            .padding(
   382→   382→                start = if (isOwnMessage) 48.dp else 0.dp,
   383→   383→                end = if (!isOwnMessage) 48.dp else 0.dp
   384→   384→            ),
   385→   385→        horizontalArrangement = alignment
   386→   386→    ) {
   387→   387→        Column(
   388→   388→            modifier = Modifier
   389→   389→                .clip(shape)
   390→   390→                .background(bubbleColor)
   391→   391→                .combinedClickable(
   392→   392→                    interactionSource = remember { MutableInteractionSource() },
   393→   393→                    indication = null,
   394→   394→                    onClick = {},
   395→   395→                    onLongClick = onLongClick
   396→   396→                )
   397→   397→                .padding(horizontal = 14.dp, vertical = 10.dp)
   398→   398→        ) {
   399→   399→            // Reply reference preview
   400→   400→            if (message.replyToPreview != null) {
   401→   401→                Row(
   402→   402→                    modifier = Modifier
   403→   403→                        .fillMaxWidth()
   404→   404→                        .clip(RoundedCornerShape(8.dp))
   405→   405→                        .background(NeonMintAlpha15)
   406→   406→                        .padding(horizontal = 8.dp, vertical = 4.dp)
   407→   407→                ) {
   408→   408→                    Text(
   409→   409→                        text = message.replyToSenderName ?: "",
   410→   410→                        color = NeonMint,
   411→   411→                        fontSize = 12.sp,
   412→   412→                        fontWeight = FontWeight.SemiBold,
   413→   413→                        maxLines = 1
   414→   414→                    )
   415→   415→                    Spacer(modifier = Modifier.width(6.dp))
   416→   416→                    Text(
   417→   417→                        text = message.replyToPreview ?: "",
   418→   418→                        color = TextSecondary,
   419→   419→                        fontSize = 12.sp,
   420→   420→                        maxLines = 1,
   421→   421→                        overflow = TextOverflow.Ellipsis,
   422→   422→                        modifier = Modifier.weight(1f)
   423→   423→                    )
   424→   424→                }
   425→   425→                Spacer(modifier = Modifier.height(4.dp))
   426→   426→            }
   427→   427→
   428→   428→            // Forwarded label
   429→   429→            if (message.isForwarded) {
   430→   430→                Text(
   431→   431→                    text = "Forwarded",
   432→   432→                    color = TextSecondary,
   433→   433→                    fontSize = 11.sp,
   434→   434→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   435→   435→                )
   436→   436→                Spacer(modifier = Modifier.height(2.dp))
   437→   437→            }
   438→   438→
   439→   439→            // Media thumbnail
   440→   440→            if (message.type == MessageContentType.IMAGE || message.type == MessageContentType.VIDEO) {
   441→   441→                val imageUrl = message.mediaThumbnailUrl ?: message.mediaUrl
   442→   442→                if (!imageUrl.isNullOrBlank()) {
   443→   443→                    AsyncImage(
   444→   444→                        model = imageUrl,
   445→   445→                        contentDescription = "Media",
   446→   446→                        modifier = Modifier
   447→   447→                            .fillMaxWidth()
   448→   448→                            .height(180.dp)
   449→   449→                            .clip(RoundedCornerShape(10.dp)),
   450→   450→                        contentScale = ContentScale.Crop
   451→   451→                    )
   452→   452→                    Spacer(modifier = Modifier.height(4.dp))
   453→   453→                }
   454→   454→                // Play icon overlay for video
   455→   455→                if (message.type == MessageContentType.VIDEO) {
   456→   456→                    Box(
   457→   457→                        modifier = Modifier
   458→   458→                            .size(36.dp)
   459→   459→                            .clip(CircleShape)
   460→   460→                            .background(DarkPetrolCharcoal.copy(alpha = 0.7f)),
   461→   461→                        contentAlignment = Alignment.Center
   462→   462→                    ) {
   463→   463→                        Icon(
   464→   464→                            imageVector = Icons.Filled.PlayArrow,
   465→   465→                            contentDescription = "Play",
   466→   466→                            tint = TextPrimary,
   467→   467→                            modifier = Modifier.size(20.dp)
   468→   468→                        )
   469→   469→                    }
   470→   470→                }
   471→   471→            }
   472→   472→
   473→   473→            // Voice message indicator
   474→   474→            if (message.type == MessageContentType.AUDIO_VOICE) {
   475→   475→                Row(verticalAlignment = Alignment.CenterVertically) {
   476→   476→                    Icon(
   477→   477→                        imageVector = Icons.Filled.PlayArrow,
   478→   478→                        contentDescription = "Voice",
   479→   479→                        tint = NeonMint,
   480→   480→                        modifier = Modifier.size(20.dp)
   481→   481→                    )
   482→   482→                    Spacer(modifier = Modifier.width(6.dp))
   483→   483→                    Text(
   484→   484→                        text = "Voice message",
   485→   485→                        color = TextSecondary,

   852→   852→/** Formats epoch milliseconds to a readable time string. */
   853→   853→private fun formatTimestamp(epochMs: Long): String {
   854→   854→    if (epochMs == 0L) return ""
   855→   855→    val sdf = java.text.SimpleDateFormat("HH:mm", java.util.Locale.getDefault())
   856→   856→    return sdf.format(java.util.Date(epochMs))
   857→   857→}
