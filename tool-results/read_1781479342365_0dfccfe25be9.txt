    87→// ── Screen Entry Point ────────────────────────────────────────────────────────

    89→/**
    90→ * Full chat message screen with keyboard avoidance and Liquid Glass design.
    91→ *
    92→ * Features:
    93→ * - Wrapped in [Scaffold] with `Modifier.imePadding()` for keyboard avoidance
    94→ * - [LazyColumn] with `reverseLayout = true` for messages — newest at bottom
    95→ * - Auto-scroll on new messages via [LaunchedEffect]
    96→ * - 74dp liquid glass pill-shaped input tray
    97→ * - Outgoing messages: glowing neon mint transparency background
    98→ * - Incoming messages: translucent dark petrol grey curves
    99→ * - Long-press action menu with 3D reactions, reply, forward, delete
   100→ * - Reply preview bar above input tray
   101→ * - Call state overlay when active
   102→ * - Top bar with contact info and call icons
   103→ * - Media messages with Coil [AsyncImage] thumbnails
   104→ *
   105→ * @param threadId The ID of the chat thread to display.
   106→ * @param onBack Callback for the back navigation button.
   107→ */
   108→@OptIn(ExperimentalFoundationApi::class)
   109→@Composable
   110→fun ChatMessageScreen(
   111→    threadId: String,
   112→    onBack: () -> Unit = {},
   113→    viewModel: ChatViewModel = hiltViewModel()
   114→) {
   115→    val messages by viewModel.messages.collectAsState()
   116→    val thread by viewModel.thread.collectAsState()
   117→    val inputText by viewModel.inputText.collectAsState()
   118→    val isSending by viewModel.isSending.collectAsState()
   119→    val callState by viewModel.callState.collectAsState()
   120→    val replyToMessage by viewModel.replyToMessage.collectAsState()
   121→    val showActionMenu by viewModel.showActionMenu.collectAsState()
   122→    val currentUserId by viewModel.currentUserId.collectAsState()
   123→    val errorMessage by viewModel.errorMessage.collectAsState()
   124→
   125→    val snackbarHostState = remember { SnackbarHostState() }
   126→    val listState = rememberLazyListState()
   127→
   128→    // Load thread data on first composition
   129→    LaunchedEffect(threadId) {
   130→        viewModel.loadThread(threadId)
   131→    }
   132→
   133→    // Auto-scroll to bottom on new messages
   134→    LaunchedEffect(messages.size) {
   135→        if (messages.isNotEmpty()) {
   136→            listState.animateScrollToItem(0)
   137→        }
   138→    }
   139→
   140→    // Display error messages as snackbars
   141→    LaunchedEffect(errorMessage) {
   142→        errorMessage?.let {
   143→            snackbarHostState.showSnackbar(it)
   144→            viewModel.clearError()
   145→        }
   146→    }
   147→
   148→    // Determine contact info from thread
   149→    val otherParticipant = thread?.participantProfiles?.values?.firstOrNull {
   150→        it.uid != currentUserId
   151→    }
   152→    val contactName = otherParticipant?.displayName ?: "Chat"
   153→    val contactAvatar = otherParticipant?.avatarUrl ?: ""
   154→    val isOnline = otherParticipant?.isOnline ?: false
   155→
   156→    Scaffold(
   157→        snackbarHost = { SnackbarHost(snackbarHostState) },
   158→        containerColor = Color.Transparent,
   159→        modifier = Modifier.fillMaxSize()
   160→    ) { innerPadding ->
   161→        Box(modifier = Modifier.fillMaxSize()) {
   162→            // Animated glass background
   163→            ZixoGlassBackground()
   164→
   165→            Column(
   166→                modifier = Modifier
   167→                    .fillMaxSize()
   168→                    .padding(innerPadding)
   169→            ) {
   170→                // ── Top Bar ────────────────────────────────────────────────
   171→                ChatTopBar(
   172→                    contactName = contactName,
   173→                    isOnline = isOnline,
   174→                    onBackClick = onBack,
   175→                    onAudioCallClick = { viewModel.startAudioCall() },
   176→                    onVideoCallClick = { viewModel.startVideoCall() }
   177→                )
   178→
   179→                // ── Message List ───────────────────────────────────────────
   180→                LazyColumn(
   181→                    state = listState,
   182→                    reverseLayout = true,
   183→                    modifier = Modifier
   184→                        .weight(1f)
   185→                        .fillMaxWidth()
   186→                        .padding(horizontal = 12.dp),
   187→                    verticalArrangement = Arrangement.spacedBy(6.dp)
   188→                ) {
   189→                    items(
   190→                        count = messages.size,
   191→                        key = { index -> messages[index].id }
   192→                    ) { index ->
   193→                        val message = messages[index]
   194→                        val isOwnMessage = message.senderUid == currentUserId
   195→
   196→                        MessageBubble(
   197→                            message = message,
   198→                            isOwnMessage = isOwnMessage,
   199→                            onLongClick = { viewModel.showActionMenu.value = message }
   200→                        )
   201→                    }
   202→                }
   203→
   204→                // ── Reply Preview Bar ──────────────────────────────────────
   205→                AnimatedVisibility(
   206→                    visible = replyToMessage != null,
   207→                    enter = slideInVertically() + fadeIn(),
   208→                    exit = slideOutVertically() + fadeOut()
   209→                ) {
   210→                    replyToMessage?.let { replied ->
   211→                        ReplyPreviewBar(
   212→                            senderName = replied.replyToSenderName ?: replied.senderDisplayName,
   213→                            previewText = replied.replyToPreview ?: replied.content,
   214→                            onDismiss = { viewModel.clearReply() }
   215→                        )
   216→                    }
   217→                }
   218→
   219→                // ── Input Tray (74dp Liquid Glass Pill) ────────────────────
   220→                InputTray(
   221→                    value = inputText,
   222→                    onValueChange = { viewModel.inputText.value = it },
   223→                    onSendClick = { viewModel.sendMessage() },
   224→                    isSending = isSending,
   225→                    modifier = Modifier.fillMaxWidth()
   226→                )
   227→            }
   228→
   229→            // ── Action Menu Overlay ────────────────────────────────────────
   230→            showActionMenu?.let { actionMessage ->
   231→                ActionMenuOverlay(
   232→                    message = actionMessage,
   233→                    isOwnMessage = actionMessage.senderUid == currentUserId,
   234→                    onReact = { emoji, isThreeD ->
   235→                        viewModel.addReaction(actionMessage.id, emoji, isThreeD)
   236→                    },
   237→                    onReply = {
   238→                        viewModel.replyToMessage.value = actionMessage
   239→                        viewModel.dismissActionMenu()
   240→                    },
   241→                    onForward = {
   242→                        viewModel.forwardMessage(actionMessage.id, emptyList())
   243→                    },
   244→                    onDeleteForMe = {
   245→                        viewModel.deleteForMe(actionMessage.id)
   246→                    },
   247→                    onDeleteForEveryone = {
   248→                        viewModel.deleteForEveryone(actionMessage.id)
   249→                    },
   250→                    onDismiss = { viewModel.dismissActionMenu() }
   251→                )
   252→            }
   253→
   254→            // ── Call State Overlay ─────────────────────────────────────────
   255→            if (callState !is CallState.IDLE) {
   256→                CallStateOverlay(
   257→                    callState = callState,
   258→                    contactName = contactName,
   259→                    contactAvatar = contactAvatar
   260→                )
   261→            }
   262→        }
   263→    }
   264→}

   328→@OptIn(ExperimentalFoundationApi::class)
   329→@Composable
   330→private fun MessageBubble(
   331→    message: MessageModel,
   332→    isOwnMessage: Boolean,
   333→    onLongClick: () -> Unit
   334→) {
   335→    // Deleted placeholder
   336→    if (message.isDeletedForEveryone) {
   337→        Row(
   338→            modifier = Modifier
   339→                .fillMaxWidth()
   340→                .padding(
   341→                    start = if (isOwnMessage) 48.dp else 0.dp,
   342→                    end = if (!isOwnMessage) 48.dp else 0.dp
   343→                ),
   344→            horizontalArrangement = if (isOwnMessage) Arrangement.End else Arrangement.Start
   345→        ) {
   346→            Box(
   347→                modifier = Modifier
   348→                    .clip(RoundedCornerShape(BubbleCornerSize))
   349→                    .background(IncomingBubbleColor)
   350→                    .padding(horizontal = 14.dp, vertical = 10.dp)
   351→            ) {
   352→                Text(
   353→                    text = "This message was deleted",
   354→                    color = TextSecondary,
   355→                    fontSize = 14.sp,
   356→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   357→                )
   358→            }
   359→        }
   360→        return
   361→    }
   362→
   363→    // Soft-deleted for me — invisible
   364→    if (message.isDeletedForMe) return
   365→
   366→    val bubbleColor = if (isOwnMessage) NeonMintAlpha15 else IncomingBubbleColor
   367→    val alignment = if (isOwnMessage) Arrangement.End else Arrangement.Start
   368→    val shape = RoundedCornerShape(
   369→        topStart = BubbleCornerSize,
   370→        topEnd = BubbleCornerSize,
   371→        bottomStart = if (isOwnMessage) BubbleCornerSize else 4.dp,
   372→        bottomEnd = if (isOwnMessage) 4.dp else BubbleCornerSize
   373→    )
   374→
   375→    Row(
   376→        modifier = Modifier
   377→            .fillMaxWidth()
   378→            .padding(
   379→                start = if (isOwnMessage) 48.dp else 0.dp,
   380→                end = if (!isOwnMessage) 48.dp else 0.dp
   381→            ),
   382→        horizontalArrangement = alignment
   383→    ) {
   384→        Column(
   385→            modifier = Modifier
   386→                .clip(shape)
   387→                .background(bubbleColor)
   388→                .combinedClickable(
   389→                    interactionSource = remember { MutableInteractionSource() },
   390→                    indication = null,
   391→                    onClick = {},
   392→                    onLongClick = onLongClick
   393→                )
   394→                .padding(horizontal = 14.dp, vertical = 10.dp)
   395→        ) {
   396→            // Reply reference preview
   397→            if (message.replyToPreview != null) {
   398→                Row(
   399→                    modifier = Modifier
   400→                        .fillMaxWidth()
   401→                        .clip(RoundedCornerShape(8.dp))
   402→                        .background(NeonMintAlpha15)
   403→                        .padding(horizontal = 8.dp, vertical = 4.dp)
   404→                ) {
   405→                    Text(
   406→                        text = message.replyToSenderName ?: "",
   407→                        color = NeonMint,
   408→                        fontSize = 12.sp,
   409→                        fontWeight = FontWeight.SemiBold,
   410→                        maxLines = 1
   411→                    )
   412→                    Spacer(modifier = Modifier.width(6.dp))
   413→                    Text(
   414→                        text = message.replyToPreview ?: "",
   415→                        color = TextSecondary,
   416→                        fontSize = 12.sp,
   417→                        maxLines = 1,
   418→                        overflow = TextOverflow.Ellipsis,
   419→                        modifier = Modifier.weight(1f)
   420→                    )
   421→                }
   422→                Spacer(modifier = Modifier.height(4.dp))
   423→            }
   424→
   425→            // Forwarded label
   426→            if (message.isForwarded) {
   427→                Text(
   428→                    text = "Forwarded",
   429→                    color = TextSecondary,
   430→                    fontSize = 11.sp,
   431→                    fontStyle = androidx.compose.ui.text.font.FontStyle.Italic
   432→                )
   433→                Spacer(modifier = Modifier.height(2.dp))
   434→            }
   435→
   436→            // Media thumbnail
   437→            if (message.type == MessageContentType.IMAGE || message.type == MessageContentType.VIDEO) {
   438→                val imageUrl = message.mediaThumbnailUrl ?: message.mediaUrl
   439→                if (!imageUrl.isNullOrBlank()) {
   440→                    AsyncImage(
   441→                        model = imageUrl,
   442→                        contentDescription = "Media",
   443→                        modifier = Modifier
   444→                            .fillMaxWidth()
   445→                            .height(180.dp)
   446→                            .clip(RoundedCornerShape(10.dp)),
   447→                        contentScale = ContentScale.Crop
   448→                    )
   449→                    Spacer(modifier = Modifier.height(4.dp))
   450→                }
   451→                // Play icon overlay for video
   452→                if (message.type == MessageContentType.VIDEO) {
   453→                    Box(
   454→                        modifier = Modifier
   455→                            .size(36.dp)
   456→                            .clip(CircleShape)
   457→                            .background(DarkPetrolCharcoal.copy(alpha = 0.7f)),
   458→                        contentAlignment = Alignment.Center
   459→                    ) {
   460→                        Icon(
   461→                            imageVector = Icons.Filled.PlayArrow,
   462→                            contentDescription = "Play",
   463→                            tint = TextPrimary,
   464→                            modifier = Modifier.size(20.dp)
   465→                        )
   466→                    }
   467→                }
   468→            }
   469→
   470→            // Voice message indicator
   471→            if (message.type == MessageContentType.AUDIO_VOICE) {
   472→                Row(verticalAlignment = Alignment.CenterVertically) {
   473→                    Icon(
   474→                        imageVector = Icons.Filled.PlayArrow,
   475→                        contentDescription = "Voice",
   476→                        tint = NeonMint,
   477→                        modifier = Modifier.size(20.dp)
   478→                    )
   479→                    Spacer(modifier = Modifier.width(6.dp))
   480→                    Text(
   481→                        text = "Voice message",
   482→                        color = TextSecondary,
   483→                        fontSize = 14.sp
   484→                    )
   485→                }
   486→                Spacer(modifier = Modifier.height(4.dp))
   487→            }
   488→
   489→            // Text content
   490→            if (message.content.isNotBlank()) {
   491→                Text(
   492→                    text = message.content,
   493→                    color = TextPrimary,
   494→                    fontSize = 15.sp,
   495→                    lineHeight = 20.sp
   496→                )
   497→            }
   498→
   499→            // Media caption
   500→            if (!message.caption.isNullOrBlank() && message.type != MessageContentType.TEXT) {
   501→                Spacer(modifier = Modifier.height(4.dp))
   502→                Text(
   503→                    text = message.caption,
   504→                    color = TextPrimary,
   505→                    fontSize = 14.sp
   506→                )
   507→            }
   508→
   509→            // Timestamp and read receipt row
   510→            Row(
   511→                modifier = Modifier.fillMaxWidth(),
   512→                horizontalArrangement = Arrangement.End,
   513→                verticalAlignment = Alignment.CenterVertically
   514→            ) {
   515→                Text(
   516→                    text = formatTimestamp(message.timestamp),
   517→                    color = TextSecondary,
   518→                    fontSize = 11.sp
   519→                )
   520→                if (isOwnMessage) {
   521→                    Spacer(modifier = Modifier.width(4.dp))
   522→                    val tickColor = if (message.isRead) NeonMint else TextSecondary
   523→                    Text(text = if (message.isRead) "✓✓" else "✓", color = tickColor, fontSize = 11.sp)
   524→                }
   525→            }
   526→
   527→            // Reactions row
   528→            if (message.reactions.isNotEmpty()) {
   529→                Spacer(modifier = Modifier.height(4.dp))
   530→                Row(horizontalArrangement = Arrangement.spacedBy(2.dp)) {
   531→                    message.reactions.distinctBy { it.emoji }.forEach { reaction ->
   532→                        Box(
   533→                            modifier = Modifier
   534→                                .clip(RoundedCornerShape(12.dp))
   535→                                .background(GlassBorderColor)
   536→                                .padding(horizontal = 6.dp, vertical = 2.dp)
   537→                        ) {
   538→                            Text(
   539→                                text = "${reaction.emoji} ${message.reactions.count { it.emoji == reaction.emoji }}",
   540→                                fontSize = 12.sp
   541→                            )
   542→                        }
   543→                    }
   544→                }
   545→            }
   546→        }
   547→    }
   548→}

   600→// ── Input Tray ────────────────────────────────────────────────────────────────
   601→
   602→@Composable
   603→private fun InputTray(
   604→    value: String,

   849→/** Formats epoch milliseconds to a readable time string. */
   850→private fun formatTimestamp(epochMs: Long): String {
   851→    if (epochMs == 0L) return ""
   852→    val sdf = java.text.SimpleDateFormat("HH:mm", java.util.Locale.getDefault())
   853→    return sdf.format(java.util.Date(epochMs))
   854→}
