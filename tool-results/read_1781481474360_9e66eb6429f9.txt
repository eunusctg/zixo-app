   532→   532→// QR Code Image Composable (brand emerald green on frosted glass)
   533→   533→// ════════════════════════════════════════════════════════════════
   534→   534→
   535→   535→@Composable
   536→   536→private fun QrCodeImage(
   537→   537→    data: String,
   538→   538→    size: androidx.compose.ui.unit.Dp,
   539→   539→    qrColor: Color = NeonMint,
   540→   540→    backgroundColor: Color = Color.White,
   541→   541→    modifier: Modifier = Modifier
   542→   542→) {
   543→   543→    if (data.isBlank()) return
   544→   544→
   545→   545→    val density = LocalDensity.current
   546→   546→    val sizePx = with(density) { size.roundToPx() }
   547→   547→
   548→   548→    val qrBitmap = remember(data, sizePx) {
   549→   549→        generateQrBitmap(data, sizePx, qrColor, backgroundColor)
   550→   550→    }
   551→   551→
   552→   552→    Box(
   553→   553→        modifier = modifier
   554→   554→            .size(size)
   555→   555→            .clip(RoundedCornerShape(16.dp))
   556→   556→            .background(Color(0x0D1A2A32)) // Very subtle dark glass behind QR
   557→   557→            .border(
   558→   558→                width = 1.dp,
   559→   559→                color = Color(0x1AFFFFFF),
   560→   560→                shape = RoundedCornerShape(16.dp)
   561→   561→            )
   562→   562→            .padding(12.dp),
   563→   563→        contentAlignment = Alignment.Center
   564→   564→    ) {
   565→   565→        androidx.compose.foundation.Image(
   566→   566→            bitmap = qrBitmap.asImageBitmap(),
   567→   567→            contentDescription = "QR Code",
   568→   568→            modifier = Modifier
   569→   569→                .size(size - 24.dp)
   570→   570→                .clip(RoundedCornerShape(8.dp)),
   571→   571→            contentScale = ContentScale.Fit
   572→   572→        )
   573→   573→    }

   575→   575→
   576→   576→/**
   577→   577→ * Generates a QR code bitmap with custom foreground/background colors.
   578→   578→ */
   579→   579→private fun generateQrBitmap(
   580→   580→    data: String,
   581→   581→    sizePx: Int,
   582→   582→    fgColor: Color,
   583→   583→    bgColor: Color
   584→   584→): Bitmap {
   585→   585→    val writer = QRCodeWriter()
   586→   586→    val bitMatrix = writer.encode(data, BarcodeFormat.QR_CODE, sizePx, sizePx)
   587→   587→    val width = bitMatrix.width
   588→   588→    val height = bitMatrix.height
   589→   589→    val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
   590→   590→    val fgArgb = fgColor.copy(alpha = 1f).let {
   591→   591→        AndroidColor.argb(255, (it.red * 255).toInt(), (it.green * 255).toInt(), (it.blue * 255).toInt())
   592→   592→    }
   593→   593→    val bgArgb = bgColor.let {
   594→   594→        AndroidColor.argb(
   595→   595→            (it.alpha * 255).toInt(),
   596→   596→            (it.red * 255).toInt(),
   597→   597→            (it.green * 255).toInt(),
   598→   598→            (it.blue * 255).toInt()
   599→   599→        )
   600→   600→    }
   601→   601→    for (x in 0 until width) {
   602→   602→        for (y in 0 until height) {
   603→   603→            bitmap.setPixel(x, y, if (bitMatrix[x, y]) fgArgb else bgArgb)
   604→   604→        }
   605→   605→    }
   606→   606→    return bitmap
   607→   607→}
