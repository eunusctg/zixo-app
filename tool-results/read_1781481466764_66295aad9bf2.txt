   447→   447→// Status Ring Avatar
   448→   448→// ──────────────────────────────────────────────
   449→   449→
   450→   450→@Composable
   451→   451→private fun StatusRingAvatar(
   452→   452→    avatarUrl: String,
   453→   453→    name: String,
   454→   454→    size: Dp,
   455→   455→    segmentCount: Int,
   456→   456→    ringColor: Color
   457→   457→) {
   458→   458→    val strokeWidth = 3.dp
   459→   459→    val gapAngle = 6f
   460→   460→    val totalGap = gapAngle * segmentCount
   461→   461→    val segmentSweep = if (segmentCount > 0) (360f - totalGap) / segmentCount else 0f
   462→   462→
   463→   463→    Box(contentAlignment = Alignment.Center) {
   464→   464→        // Colored ring segments
   465→   465→        if (segmentCount > 0) {
   466→   466→            Canvas(modifier = Modifier.size(size + strokeWidth * 2)) {
   467→   467→                var startAngle = -90f
   468→   468→                for (i in 0 until segmentCount) {
   469→   469→                    drawArc(
   470→   470→                        color = ringColor,
   471→   471→                        startAngle = startAngle,
   472→   472→                        sweepAngle = segmentSweep,
   473→   473→                        useCenter = false,
   474→   474→                        style = Stroke(
   475→   475→                            width = strokeWidth.toPx(),
   476→   476→                            cap = StrokeCap.Round
   477→   477→                        )
   478→   478→                    )
   479→   479→                    startAngle += segmentSweep + gapAngle
   480→   480→                }
   481→   481→            }
   482→   482→        }
   483→   483→
   484→   484→        AvatarComponent(
   485→   485→            imageUrl = avatarUrl,
   486→   486→            name = name,
   487→   487→            size = size
   488→   488→        )
   489→   489→    }
   490→   490→}

  1014→  1014→                Box(

  1259→  1259→private fun formatRelativeTime(timestampMs: Long): String {
  1260→  1260→    val now = System.currentTimeMillis()
  1261→  1261→    val diff = now - timestampMs
  1262→  1262→    val seconds = diff / 1000
  1263→  1263→    val minutes = seconds / 60
  1264→  1264→    val hours = minutes / 60
  1265→  1265→    return when {
  1266→  1266→        seconds < 60 -> "just now"
  1267→  1267→        minutes < 60 -> "${minutes}m ago"
  1268→  1268→        hours < 24 -> "${hours}h ago"
  1269→  1269→        else -> "${hours / 24}d ago"
  1270→  1270→    }
  1271→  1271→}

  1273→  1273→private fun colorToHex(color: Color): String {
  1274→  1274→    val alpha = (color.alpha * 255).toInt()
  1275→  1275→    val red = (color.red * 255).toInt()
  1276→  1276→    val green = (color.green * 255).toInt()
  1277→  1277→    val blue = (color.blue * 255).toInt()
  1278→  1278→    return "#${String.format("%02X", alpha)}${String.format("%02X", red)}${String.format("%02X", green)}${String.format("%02X", blue)}"
  1279→  1279→}

  1282→  1282→    return try {

  1286→  1286→    } catch (_: Exception) {

  1288→  1288→    }
