    41→/**
    42→ * Core WebRTC Client — Crash-Proof Pure WebRTC Implementation (NO LiveKit).
    43→ *
    44→ * ## Rigid Native WebRTC Processing Guardrails:
    45→ *
    46→ * 1. **All Media Isolated on Dispatchers.IO:**
    47→ *    PeerConnectionFactory, PeerConnection, SDP generation, and ICE candidate
    48→ *    discovery are handled exclusively on background thread scopes.
    49→ *    The Android Main Thread is NEVER blocked by WebRTC operations.
    50→ *
    51→ * 2. **Mandatory Foreground Service:**
    52→ *    CallForegroundService (FOREGROUND_SERVICE_TYPE_CAMERA | MICROPHONE | PHONE_CALL)
    53→ *    wraps every WebRTC session so Android does not terminate media streams
    54→ *    when the user backgrounds the application.
    55→ *
    56→ * 3. **Hardware Acoustic Echo & Routing Calibration:**
    57→ *    AudioManager.MODE_IN_COMMUNICATION + AudioFocusRequest(USAGE_VOICE_COMMUNICATION)
    58→ *    eliminates system echo feedback loops and fixes audio routing to
    59→ *    speakerphone or Bluetooth accessories.
    60→ *
    61→ * 4. **EglBase Singleton Scope Preservation:**
    62→ *    The hardware-accelerated root EglBase context is created ONCE as a
    63→ *    DI Singleton and NEVER re-initialized on Composable recomposition.
    64→ *    Re-creating EglBase during recomposition triggers an uncatchable SIGABRT.
    65→ *    SurfaceViewRenderer instances are wrapped inside AndroidView.
    66→ *
    67→ * 5. **Structured try-catch Boundaries:**
    68→ *    Every WebRTC operation has a precise try-catch boundary preventing
    69→ *    any native crash from propagating as an unhandled exception.
    70→ */
    71→@Singleton

    73→    private val context: Context

   143→    /**
   144→     * Initializes the PeerConnectionFactory on Dispatchers.IO.
   145→     * Must be called before any other WebRTC operation.
   146→     *
   147→     * This method is idempotent — calling it multiple times is safe.
   148→     */
   149→    suspend fun initialize() = withContext(Dispatchers.IO) {
   150→        if (peerConnectionFactory != null) {
   151→            Timber.d("PeerConnectionFactory already initialized, skipping")
   152→            return@withContext
   153→        }
   154→
   155→        try {
   156→            val initializationOptions = PeerConnectionFactory.InitializationOptions
   157→                .builder(context.applicationContext)
   158→                .setEnableInternalTracer(false)
   159→                .createInitializationOptions()
   160→
   161→            PeerConnectionFactory.initialize(initializationOptions)
   162→
   163→            val encoderFactory = DefaultVideoEncoderFactory(_eglBase.eglBaseContext, true, true)
   164→            val decoderFactory = DefaultVideoDecoderFactory(_eglBase.eglBaseContext)
   165→
   166→            peerConnectionFactory = PeerConnectionFactory.builder()
   167→                .setVideoEncoderFactory(encoderFactory)
   168→                .setVideoDecoderFactory(decoderFactory)
   169→                .createPeerConnectionFactory()
   170→
   171→            Timber.d("PeerConnectionFactory initialized successfully on IO thread")
   172→        } catch (e: Exception) {
   173→            Timber.e(e, "CRITICAL: Failed to initialize PeerConnectionFactory")
   174→            throw e
   175→        }
   176→    }

   178→    /**
   179→     * Creates a PeerConnection with STUN/TURN servers and initializes
   180→     * local media tracks (audio and optionally video).
   181→     *
   182→     * Also configures the hardware audio pipeline:
   183→     * - AudioManager.MODE_IN_COMMUNICATION
   184→     * - AudioFocusRequest with USAGE_VOICE_COMMUNICATION
   185→     *
   186→     * @param isVideoCall Whether to include video tracks.
   187→     */
   188→    suspend fun initializePeerConnection(isVideoCall: Boolean) = withContext(Dispatchers.IO) {
   189→        try {
   190→            initialize()
   191→            disconnect() // Clean up any existing connection
   192→

   194→                sdpSemantics = PeerConnection.SdpSemantics.UNIFIED_PLAN
   195→                continualGatheringPolicy = PeerConnection.ContinualGatheringPolicy.GATHER_CONTINUALLY
   196→            }
   197→
   198→            peerConnection = peerConnectionFactory?.createPeerConnection(
   199→                rtcConfig,
   200→                createPeerConnectionObserver()
   201→            )
   202→
   203→            if (peerConnection == null) {
   204→                throw IllegalStateException("Failed to create PeerConnection — factory returned null")
   205→            }
   206→
   207→            // Add audio track
   208→            createAudioTrack()
   209→
   210→            // Add video track if video call
   211→            if (isVideoCall) {
   212→                createVideoTrack()
   213→            }
   214→
   215→            // Configure hardware audio pipeline
   216→            configureAudioForCall()
   217→
   218→            Timber.d("PeerConnection created (video=%s) with audio calibration", isVideoCall)
   219→        } catch (e: Exception) {
   220→            Timber.e(e, "CRITICAL: Failed to create PeerConnection")
   221→            throw e
   222→        }
   223→    }

   229→    /**
   230→     * Configures the Android audio system for a WebRTC call.
   231→     *
   232→     * This method applies the following critical settings:
   233→     * - **AudioManager.MODE_IN_COMMUNICATION**: Optimizes the audio path for
   234→     *   two-way voice communication, enabling echo cancellation and noise
   235→     *   suppression at the hardware level.
   236→     * - **AudioFocusRequest with USAGE_VOICE_COMMUNICATION**: Requests
   237→     *   permanent audio focus for voice communication, which ensures proper
   238→     *   audio routing and prevents other apps from interrupting the call.
   239→     * - **Speakerphone default OFF**: Routes audio to the earpiece by default
   240→     *   for audio calls, speaker for video calls.
   241→     *
   242→     * This eliminates system echo feedback loops and fixes audio routing
   243→     * to speakerphone or Bluetooth accessories flawlessly.
   244→     */
   245→    private fun configureAudioForCall() {
   246→        try {
   247→            val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
   248→            if (audioManager == null) {
   249→                Timber.w("AudioManager not available, skipping audio calibration")
   250→                return
   251→            }
   252→
   253→            // Set communication mode — enables AEC (Acoustic Echo Cancellation)
   254→            audioManager.mode = AudioManager.MODE_IN_COMMUNICATION
   255→
   256→            // Request audio focus for voice communication
   257→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
   258→                val playbackAttributes = AudioAttributes.Builder()
   259→                    .setUsage(AudioAttributes.USAGE_VOICE_COMMUNICATION)
   260→                    .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
   261→                    .build()
   262→
   263→                audioFocusRequest = AudioFocusRequest.Builder(AudioManager.AUDIOFOCUS_GAIN_TRANSIENT)
   264→                    .setAudioAttributes(playbackAttributes)
   265→                    .setOnAudioFocusChangeListener { focusChange ->
   266→                        Timber.d("Audio focus changed: %d", focusChange)
   267→                    }
   268→                    .build()
   269→
   270→                val result = audioManager.requestAudioFocus(audioFocusRequest!!)
   271→                isAudioFocusHeld = result == AudioManager.AUDIOFOCUS_REQUEST_GRANTED
   272→                Timber.d("Audio focus request result: %s (granted=%s)", result, isAudioFocusHeld)
   273→            } else {
   274→                @Suppress("DEPRECATION")
   275→                val result = audioManager.requestAudioFocus(
   276→                    null,
   277→                    AudioManager.STREAM_VOICE_CALL,
   278→                    AudioManager.AUDIOFOCUS_GAIN_TRANSIENT
   279→                )
   280→                isAudioFocusHeld = result == AudioManager.AUDIOFOCUS_REQUEST_GRANTED
   281→                Timber.d("Audio focus request (legacy) result: %s", result)
   282→            }
   283→
   284→            // Default: earpiece for audio, speaker for video
   285→            audioManager.isSpeakerphoneOn = false
   286→            audioManager.isBluetoothScoOn = false
   287→
   288→            Timber.d("Audio calibrated: MODE_IN_COMMUNICATION, USAGE_VOICE_COMMUNICATION")
   289→        } catch (e: Exception) {
   290→            Timber.e(e, "Failed to configure audio for call — audio may have echo issues")
   291→        }
   292→    }

   294→    /**
   295→     * Releases audio focus and restores the audio manager to normal mode.
   296→     * Must be called when a call ends.
   297→     */
   298→    private fun releaseAudioFromCall() {
   299→        try {
   300→            val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
   301→            if (audioManager == null) return
   302→
   303→            // Restore normal audio mode
   304→            audioManager.mode = AudioManager.MODE_NORMAL
   305→            audioManager.isSpeakerphoneOn = false
   306→            audioManager.isBluetoothScoOn = false
   307→
   308→            // Release audio focus
   309→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O && audioFocusRequest != null) {
   310→                audioManager.abandonAudioFocusRequest(audioFocusRequest!!)
   311→            } else {
   312→                @Suppress("DEPRECATION")
   313→                audioManager.abandonAudioFocus(null)
   314→            }
   315→            isAudioFocusHeld = false
   316→
   317→            Timber.d("Audio released: MODE_NORMAL, focus abandoned")
   318→        } catch (e: Exception) {
   319→            Timber.e(e, "Failed to release audio from call")
   320→        }
   321→    }

   327→    /**
   328→     * Creates an SDP offer for initiating a call.
   329→     * Uses suspendCancellableCoroutine for clean async handling.
   330→     * Must be called on Dispatchers.IO.
   331→     *
   332→     * @return The SDP offer string, or null on failure.
   333→     */
   334→    suspend fun createOffer(): String? = withContext(Dispatchers.IO) {
   335→        try {
   336→            val connection = peerConnection
   337→                ?: throw IllegalStateException("No PeerConnection available for createOffer")
   338→
   339→            val constraints = MediaConstraints().apply {
   340→                mandatory.add(MediaConstraints.KeyValuePair("OfferToReceiveAudio", "true"))
   341→                mandatory.add(MediaConstraints.KeyValuePair("OfferToReceiveVideo", "true"))
   342→            }
   343→
   344→            suspendCancellableCoroutine { cont ->
   345→                connection.createOffer(object : SdpObserver {
   346→                    override fun onCreateSuccess(description: SessionDescription) {
   347→                        connection.setLocalDescription(SimpleSdpObserver(), description)
   348→                        cont.resume(description.description)
   349→                    }
   350→
   351→                    override fun onSetSuccess() {}
   352→                    override fun onCreateFailure(error: String) {
   353→                        Timber.e("Create offer failed: %s", error)
   354→                        cont.resume(null)
   355→                    }
   356→
   357→                    override fun onSetFailure(error: String) {
   358→                        Timber.e("Set local description failed: %s", error)
   359→                        cont.resume(null)
   360→                    }
   361→                }, constraints)
   362→            }
   363→        } catch (e: Exception) {
   364→            Timber.e(e, "CRITICAL: Failed to create offer")
   365→            null
   366→        }
   367→    }

   458→    /**

   571→     * When speaker is ON: audio routes through the loudspeaker.
   572→     * When speaker is OFF: audio routes through the earpiece or Bluetooth.
   573→     */
   574→    fun setSpeakerOn(speakerOn: Boolean) {
   575→        try {
   576→            isSpeakerOn = speakerOn
   577→            val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
   578→            audioManager?.isSpeakerphoneOn = speakerOn
   579→
   580→            // If turning speaker off, try Bluetooth SCO if available
   581→            if (!speakerOn) {
   582→                audioManager?.isBluetoothScoOn = false
   583→            }
   584→
   585→            Timber.d("Speaker on: %s", speakerOn)
   586→        } catch (e: Exception) {
   587→            Timber.e(e, "Failed to set speaker state")
   588→        }
   589→    }

   591→    /**
   592→     * Switches between front and back camera.
   593→     */
   594→    fun switchCamera() {
   595→        try {
   596→            videoCapturer?.let { capturer ->
   597→                val cameraEnumerator = getCameraEnumerator()
   598→                if (cameraEnumerator is Camera1Enumerator) {
   599→                    Camera1Enumerator.switchCamera(capturer)
   600→                } else {
   601→                    Camera2Enumerator(context).switchCamera(capturer)
   602→                }
   603→            }
   604→            Timber.d("Camera switched")
   605→        } catch (e: Exception) {
   606→            Timber.e(e, "Failed to switch camera")
   607→        }
   608→    }

   612→    // ════════════════════════════════════════════════════════════════

   614→    /**
   615→     * Disconnects the PeerConnection and releases all WebRTC resources.
   616→     *
   617→     * This method is safe to call from any thread. All cleanup operations
   618→     * are wrapped in individual try-catch blocks to ensure that a failure
   619→     * in releasing one resource does not prevent others from being released.
   620→     *
   621→     * Also releases audio focus and restores the AudioManager to normal mode.
   622→     *
   623→     * **IMPORTANT**: Does NOT dispose the EglBase singleton. The EglBase
   624→     * lives for the entire application lifecycle and must never be destroyed
   625→     * during a call disconnect.
   626→     */
   627→    fun disconnect() {
   628→        try {
   629→            videoCapturer?.tryStopCapture()
   630→        } catch (_: Exception) {}
   631→
   632→        try {
   633→            videoCapturer?.dispose()
   634→        } catch (_: Exception) {}
   635→        videoCapturer = null
   636→
   637→        try {
   638→            localVideoSource?.dispose()
   639→        } catch (_: Exception) {}
   640→        localVideoSource = null
   641→
   642→        try {
   643→            localAudioSource?.dispose()
   644→        } catch (_: Exception) {}
   645→        localAudioSource = null
   646→
   647→        try {
   648→            localVideoTrack?.dispose()
   649→        } catch (_: Exception) {}
   650→        localVideoTrack = null
   651→
   652→        try {
   653→            localAudioTrack?.dispose()
   654→        } catch (_: Exception) {}
   655→        localAudioTrack = null
   656→
   657→        try {
   658→            remoteVideoTrack?.dispose()
   659→        } catch (_: Exception) {}
   660→        remoteVideoTrack = null
   661→
   662→        try {
   663→            surfaceTextureHelper?.dispose()
   664→        } catch (_: Exception) {}
   665→        surfaceTextureHelper = null
   666→
   667→        try {
   668→            peerConnection?.close()
   669→        } catch (_: Exception) {}
   670→
   671→        try {
   672→            peerConnection?.dispose()
   673→        } catch (_: Exception) {}
   674→        peerConnection = null
   675→
   676→        // Release audio focus and restore normal mode
   677→        releaseAudioFromCall()
   678→
   679→        // Reset state flags
   680→        isMuted = false
   681→        isCameraOff = false
   682→        isSpeakerOn = false
   683→        onIceCandidateGenerated = null
   684→        onRemoteVideoTrackAdded = null
   685→
   686→        Timber.d("WebRTC disconnected and all resources released (EglBase preserved)")
   687→    }

   693→    private fun createAudioTrack() {
   694→        try {
   695→            val factory = peerConnectionFactory ?: return
   696→            localAudioSource = factory.createAudioSource(MediaConstraints())
   697→            localAudioTrack = factory.createAudioTrack(AUDIO_TRACK_ID, localAudioSource)
   698→            localAudioTrack?.setEnabled(true)
   699→            peerConnection?.addTrack(localAudioTrack)
   700→            Timber.d("Audio track created and added to PeerConnection")
   701→        } catch (e: Exception) {
   702→            Timber.e(e, "Failed to create audio track")
   703→        }
   704→    }
   705→
   706→    private fun createVideoTrack() {
   707→        try {
   708→            val factory = peerConnectionFactory ?: return
   709→            val enumerator = getCameraEnumerator()
   710→            val deviceNames = enumerator.deviceNames
   711→
   712→            // Try to find front camera first
   713→            var capturer: VideoCapturer? = null
   714→            for (name in deviceNames) {
   715→                if (enumerator.isFrontFacing(name)) {
   716→                    try {
   717→                        capturer = enumerator.createCapturer(name, null)
   718→                        break
   719→                    } catch (e: Exception) {
   720→                        Timber.w(e, "Failed to create front camera capturer, trying next")
   721→                    }
   722→                }
   723→            }
   724→            if (capturer == null && deviceNames.isNotEmpty()) {
   725→                try {
   726→                    capturer = enumerator.createCapturer(deviceNames[0], null)
   727→                } catch (e: Exception) {
   728→                    Timber.e(e, "Failed to create any camera capturer")
   729→                }
   730→            }
   731→
   732→            capturer?.let { vc ->
   733→                videoCapturer = vc
   734→                surfaceTextureHelper = SurfaceTextureHelper.create("CaptureThread", _eglBase.eglBaseContext)
   735→                localVideoSource = factory.createVideoSource(vc.isScreencast)
   736→                vc.initialize(surfaceTextureHelper, context.applicationContext, localVideoSource?.capturerObserver)
   737→                vc.startCapture(640, 480, 30)
   738→
   739→                localVideoTrack = factory.createVideoTrack(VIDEO_TRACK_ID, localVideoSource)
   740→                localVideoTrack?.setEnabled(true)
   741→                peerConnection?.addTrack(localVideoTrack)
   742→                Timber.d("Video track created and added to PeerConnection")
   743→            }
   744→        } catch (e: Exception) {
   745→            Timber.e(e, "Failed to create video track")
   746→        }
   747→    }

   749→    /**
   750→     * Creates the PeerConnection observer that handles all WebRTC events.
   751→     *
   752→     * ICE candidates are forwarded to the signaling layer via
   753→     * [onIceCandidateGenerated] callback. Remote video tracks are
   754→     * forwarded via [onRemoteVideoTrackAdded] callback.
   755→     */
   756→    private fun createPeerConnectionObserver() = object : PeerConnection.Observer {
   757→        override fun onSignalingChange(state: PeerConnection.SignalingState?) {
   758→            Timber.d("Signaling state: %s", state)
   759→        }
   760→
   761→        override fun onIceConnectionChange(state: PeerConnection.IceConnectionState?) {
   762→            Timber.d("ICE connection state: %s", state)
   763→            when (state) {
   764→                PeerConnection.IceConnectionState.CONNECTED -> {
   765→                    Timber.d("WebRTC ICE connected")
   766→                }
   767→                PeerConnection.IceConnectionState.DISCONNECTED -> {
   768→                    Timber.w("WebRTC ICE disconnected")
   769→                }
   770→                PeerConnection.IceConnectionState.FAILED -> {
   771→                    Timber.e("WebRTC ICE connection failed")
   772→                }
   773→                else -> {}
   774→            }
   775→        }
   776→
   777→        override fun onIceConnectionReceivingChange(receiving: Boolean) {}
   778→
   779→        override fun onIceGatheringChange(state: PeerConnection.IceGatheringState?) {
   780→            Timber.d("ICE gathering state: %s", state)
   781→        }
   782→
   783→        override fun onIceCandidate(candidate: IceCandidate?) {
   784→            candidate?.let {
   785→                Timber.d("Local ICE candidate generated: %s", it.sdp)
   786→                onIceCandidateGenerated?.invoke(it)
   787→            }
   788→        }
   789→
   790→        override fun onIceCandidatesRemoved(candidates: Array<out IceCandidate>?) {
   791→            Timber.d("ICE candidates removed: %d", candidates?.size ?: 0)
   792→        }
   793→
   794→        override fun onAddStream(stream: MediaStream?) {
   795→            Timber.d("Remote stream added with %d video tracks", stream?.videoTracks?.size ?: 0)
   796→        }
   797→
   798→        override fun onRemoveStream(stream: MediaStream?) {
   799→            Timber.d("Remote stream removed")
   800→        }
   801→
   802→        override fun onDataChannel(channel: DataChannel?) {}
   803→
   804→        override fun onRenegotiationNeeded() {
   805→            Timber.d("Renegotiation needed")
   806→        }
   807→
   808→        override fun onAddTrack(receiver: RtpReceiver?, streams: Array<out MediaStream>?) {
   809→            Timber.d("Remote track added")
   810→        }
   811→
   812→        override fun onTrack(transceiver: RtpTransceiver?) {
   813→            transceiver?.let { t ->
   814→                if (t.isVideoType()) {
   815→                    val track = t.receiver.track() as? VideoTrack
   816→                    if (track != null) {
   817→                        remoteVideoTrack = track
   818→                        Timber.d("Remote video track received via onTrack")
   819→                        onRemoteVideoTrackAdded?.invoke(track)
   820→                    }
   821→                }
   822→            }
   823→        }
   824→    }

   853→     * Detects whether an RtpTransceiver carries video or audio.
   854→     * Uses RtpTransceiverType enum from the WebRTC library.
   855→     */
   856→    private fun RtpTransceiver?.isVideoType(): Boolean {
   857→        return try {
   858→            this?.let { transceiver ->
   859→                val type = RtpTransceiver.RtpTransceiverType.values()
   860→                    .getOrElse(transceiver.mediaType) { RtpTransceiver.RtpTransceiverType.AUDIO }
   861→                type == RtpTransceiver.RtpTransceiverType.VIDEO
   862→            } ?: false
   863→        } catch (_: Exception) { false }
   864→    }
