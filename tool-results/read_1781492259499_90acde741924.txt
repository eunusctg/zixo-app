   725→// ─────────────────────────────────────────────────────────────────────────────
   726→
   727→/**
   728→ * An expandable FAQ item with a question header and animated answer body.
   729→ *
   730→ * Tapping the row toggles the answer visibility with a smooth vertical
   731→ * expand/collapse animation. A chevron icon rotates to indicate the
   732→ * expansion state.
   733→ *
   734→ * @param question    The FAQ question text.
   735→ * @param answer      The FAQ answer text revealed on expansion.
   736→ * @param isExpanded  Whether the answer is currently visible.
   737→ * @param onToggle    Callback invoked when the row is tapped.
   738→ */
   739→@Composable
   740→private fun FaqItem(
   741→    question: String,
   742→    answer: String,
   743→    isExpanded: Boolean,
   744→    onToggle: () -> Unit
   745→) {
   746→    Column(
   747→        modifier = Modifier
   748→            .fillMaxWidth()
   749→            .clip(RoundedCornerShape(12.dp))
   750→            .background(Color(0x0DFFFFFF))
   751→            .clickable(
   752→                interactionSource = remember { MutableInteractionSource() },
   753→                indication = null,
   754→                onClick = onToggle
   755→            )
   756→            .padding(14.dp)
   757→    ) {
   758→        Row(
   759→            modifier = Modifier.fillMaxWidth(),
   760→            verticalAlignment = Alignment.CenterVertically
   761→        ) {
   762→            Column(
   763→                modifier = Modifier.weight(1f)
   764→            ) {
   765→                Text(
   766→                    text = question,
   767→                    color = TextPrimary,
   768→                    fontSize = 14.sp,
   769→                    fontWeight = FontWeight.SemiBold,
   770→                    lineHeight = 20.sp
   771→                )
   772→            }
   773→
   774→            Spacer(modifier = Modifier.width(8.dp))
   775→
   776→            Icon(
   777→                imageVector = if (isExpanded) Icons.Default.ArrowDropUp else Icons.Default.ArrowDropDown,
   778→                contentDescription = if (isExpanded) "Collapse" else "Expand",
   779→                tint = NeonMint,
   780→                modifier = Modifier.size(24.dp)
   781→            )
   782→        }
   783→
   784→        AnimatedVisibility(
   785→            visible = isExpanded,
   786→            enter = fadeIn() + expandVertically(),
   787→            exit = fadeOut() + shrinkVertically()
   788→        ) {
   789→            Column {
   790→                Spacer(modifier = Modifier.height(8.dp))
   791→                Text(
   792→                    text = answer,
   793→                    color = TextSecondary,
   794→                    fontSize = 13.sp,
   795→                    lineHeight = 19.sp
   796→                )
   797→            }
   798→        }
   799→    }
   800→}
